48 /**
49 * @brief Download a bound diffuse texture and write it next to the model
50 * file, so AssimpModelWriter's "diffuse_path" submesh attribute has
51 * somewhere to point.
52 *
53 * Named "<model-stem>_diffuse.png", or "<model-stem>_<suffix>_diffuse.png"
54 * when more than one texture is being written alongside the same model
55 * file (one per MeshNetwork slot).
56 *
57 * @return The sibling path on success, empty string if there is nothing
58 * to write or the download/encode fails.
59 */
60 std::string pack_diffuse_texture(
61 const std::shared_ptr<Core::VKImage>& texture,
62 const std::string& model_filepath,
63 const std::string& suffix)
64 {
65 if (!texture) {
66 return {};
67 }
68
69 const std::filesystem::path model_path(model_filepath);
70 std::string name = model_path.stem().string() + "_diffuse";
71 if (!suffix.empty()) {
72 name += "_" + suffix;
73 }
74 name += ".png";
75
76 const auto image_path = (model_path.parent_path() / name).string();
77 if (!save_image(texture, image_path)) {
78 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
79 "save_mesh: failed to write diffuse texture to '{}'", image_path);
80 return {};
81 }
82 return image_path;
83 }

85 /**
86 * @brief Give mesh_data a single full-mesh submesh Region if it has none,
87 * so a caller-supplied diffuse_path has an attribute to attach to.
88 */
89 void ensure_submesh_region(Kakshya::MeshData& mesh_data, const std::string& name)
90 {
91 if (mesh_data.submeshes && !mesh_data.submeshes->regions.empty()) {
92 return;
93 }
94
95 Kakshya::MeshSubrange sub;
96 sub.index_start = 0;
97 sub.index_count = mesh_data.face_count() * 3;
98 sub.name = name;
99
100 Kakshya::RegionGroup rg("submeshes");
101 rg.add_region(sub.to_region());
102 mesh_data.submeshes = std::move(rg);
103 }

105 /**
106 * @brief Pack one MeshWriterNode's current vertices/indices into a
107 * MeshData, baking @p world into position/normal/tangent.
108 * @return nullopt if the node has no vertices yet, or insertion fails.
109 */
110 std::optional<Kakshya::MeshData> pack_node(
111 const std::shared_ptr<Nodes::GpuSync::MeshWriterNode>& node,
112 const glm::mat4& world,
113 const std::string& name,
114 const std::string& diffuse_path = {})
115 {
116 auto mesh_data = node->get_mesh_data();
117 if (!mesh_data) {
118 return std::nullopt;
119 }
120
121 const glm::mat3 normal_matrix(world);
122
123 auto& vertex_bytes = std::get<std::vector<uint8_t>>(mesh_data->vertex_variant);
124 std::vector<Kakshya::MeshVertex> verts(vertex_bytes.size() / sizeof(Kakshya::MeshVertex));
125 std::memcpy(verts.data(), vertex_bytes.data(), vertex_bytes.size());
126 for (auto& v : verts) {
127 v.position = glm::vec3(world * glm::vec4(v.position, 1.0F));
128 v.normal = glm::normalize(normal_matrix * v.normal);
129 v.tangent = glm::normalize(normal_matrix * v.tangent);
130 }
131 std::memcpy(vertex_bytes.data(), verts.data(), vertex_bytes.size());
132
133 const auto index_count = std::get<std::vector<uint32_t>>(mesh_data->index_variant).size();
134
135 Kakshya::MeshSubrange sub;
136 sub.index_start = 0;
137 sub.index_count = static_cast<uint32_t>(index_count);
138 sub.name = name;
139 sub.diffuse_path = diffuse_path;
140 Kakshya::RegionGroup rg("submeshes");
141 rg.add_region(sub.to_region());
142 mesh_data->submeshes = std::move(rg);
143
144 return mesh_data;
145 }

157bool save_mesh(
158 const std::shared_ptr<Buffers::MeshBuffer>& buffer,
159 const std::string& filepath,
160 const ModelWriteOptions& options)
161{
162 if (!buffer) {
163 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO, "save_mesh: null buffer");
164 return false;
165 }
166
167 auto mesh_data = buffer->get_mesh_data();
168 if (buffer->has_diffuse_texture()) {
169 const auto image_path = pack_diffuse_texture(buffer->get_diffuse_texture(), filepath, "");
170 if (!image_path.empty()) {
171 ensure_submesh_region(mesh_data, "mesh");
172 mesh_data.submeshes->regions.front().set_attribute("diffuse_path", image_path);
173 }
174 }
175
176 return write_via_registry(filepath, { mesh_data }, options);

179bool save_mesh(
180 const std::shared_ptr<Nodes::Network::MeshNetwork>& network,
181 const std::string& filepath,
182 const ModelWriteOptions& options)
183{
184 if (!network) {
185 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO, "save_mesh: null network");
186 return false;
187 }
188
189 std::vector<Kakshya::MeshData> meshes;
190 meshes.reserve(network->slots().size());
191 for (const auto& slot : network->slots()) {
192 if (!slot.node) {
193 continue;
194 }
195 std::string diffuse_path;
196 if (slot.diffuse_texture) {
197 const auto suffix = !slot.name.empty() ? slot.name : std::to_string(slot.index);
198 diffuse_path = pack_diffuse_texture(slot.diffuse_texture, filepath, suffix);
199 }
200 if (auto packed = pack_node(slot.node, slot.world_transform, slot.name, diffuse_path)) {
201 meshes.push_back(std::move(*packed));
202 }
203 }
204
205 if (meshes.empty()) {
206 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
207 "save_mesh: no exportable slots in network");
208 return false;
209 }
210
211 return write_via_registry(filepath, meshes, options);
212}
