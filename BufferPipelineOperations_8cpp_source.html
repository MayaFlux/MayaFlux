19 std::optional<Kakshya::DataVariant> select_operation_data(
20 const std::unordered_map<BufferOperation*, Kakshya::DataVariant>& operation_data,
21 BufferOperation& op,
22 bool graphics_read_pending)
23 {
24 if (const auto own = operation_data.find(&op); own != operation_data.end()) {
25 return own->second;
26 }
27
28 if (!operation_data.empty()) {
29 return operation_data.begin()->second;
30 }
31
32 if (graphics_read_pending) {
33 return std::nullopt;
34 }
35
36 return Kakshya::DataVariant {};

47void BufferPipeline::process_operation(BufferOperation& op, uint64_t cycle)
48{
49 try {
50 switch (op.get_type()) {
51 case BufferOperation::OpType::CAPTURE:
52 capture_operation(op, cycle);
53 break;
54
55 case BufferOperation::OpType::TRANSFORM:
56 process_transform(op, cycle);
57 break;
58
59 case BufferOperation::OpType::ROUTE:
60 process_route(op);
61 break;
62
63 case BufferOperation::OpType::LOAD:
64 process_load(op);
65 break;
66
67 case BufferOperation::OpType::FUSE:
68 process_fuse(op, cycle);
69 break;
70
71 case BufferOperation::OpType::DISPATCH:
72 process_dispatch(op, cycle);
73 break;
74
75 case BufferOperation::OpType::MODIFY:
76 process_modify(op, cycle);
77 break;
78
79 case BufferOperation::OpType::CONDITION:
80 break;
81
82 default:
83 MF_ERROR(Journal::Component::Kriya,
84 Journal::Context::CoroutineScheduling,
85 "Unknown operation type in pipeline : {} : {}",
86 Reflect::enum_to_string(op.get_type()), std::to_string(static_cast<int>(op.get_type())));
87 break;
88 }
89 } catch (const std::exception& e) {
90 error_rethrow(Journal::Component::Kriya,
91 Journal::Context::CoroutineScheduling,
92 std::source_location::current(),
93 "Error processing operation in BufferPipeline: {}",
94 e.what());
95 }

98void BufferPipeline::process_transform(BufferOperation& op, uint64_t cycle)
99{
100 auto selected = select_operation_data(m_operation_data, op, !m_readers.empty());
101 if (!selected) {
102 return;
103 }
104 auto input_data = std::move(*selected);
105
106 if (op.m_transformer) {
107 auto transformed = op.m_transformer(input_data, cycle);
108 m_operation_data[&op] = transformed;
109
110 const bool has_downstream_route = std::ranges::any_of(
111 m_operations,
112 [](const BufferOperation& o) {
113 return o.get_type() == BufferOperation::OpType::ROUTE;
114 });
115
116 if (!has_downstream_route) {
117 for (auto& candidate : std::ranges::reverse_view(m_operations)) {
118 if (&candidate == &op)
119 continue;
120 if (candidate.get_type() != BufferOperation::OpType::CAPTURE) {
121 continue;
122 }
123
124 if (const auto buf = candidate.m_capture.get_audio_buffer()) {
125 if (std::holds_alternative<std::vector<double>>(transformed)
126 && std::get<std::vector<double>>(transformed).size() == buf->get_data().size()) {
127 detail::write_to_buffer(buf, transformed);
128 }
129 break;
130 }
131
132 if (const auto target = candidate.m_capture.get_graphics_buffer()) {
133 const auto captured = m_operation_data.find(&candidate);
134 if (captured != m_operation_data.end()
135 && detail::accepts_raw_write(target)
136 && same_extent(captured->second, transformed)) {
137 queue_graphics_write(op, target, transformed);
138 }
139 break;
140 }
141 }
142 }
143 }

146void BufferPipeline::process_route(BufferOperation& op)
147{
148 if (!op.m_source_images.empty()) {
149 uint32_t layer = 0;
150 for (const auto& image : op.m_source_images) {
151 detail::write_layer(op.m_target_graphics_buffer, layer++, image);
152 }
153 return;
154 }
155
156 if (op.m_source_graphics_buffer) {
157 auto frame = read_graphics_buffer(op, op.m_source_graphics_buffer);
158 if (!frame) {
159 return;
160 }
161
162 detail::write_to_container(op.m_target_graphics_stream, *frame);
163 m_operation_data[&op] = std::move(*frame);
164 return;
165 }
166
167 const auto selected = select_operation_data(m_operation_data, op, !m_readers.empty());
168 if (!selected) {
169 return;
170 }
171 const auto& data_to_route = *selected;
172
173 if (op.m_target_audio_buffer) {
174 if (!m_buffer_manager) {
175 error<std::invalid_argument>(Journal::Component::Kriya,
176 Journal::Context::CoroutineScheduling,
177 std::source_location::current(),
178 "BufferPipeline has no BufferManager for ROUTE-to-buffer operation");
179 }
180
181 if (!op.m_attached_processor) {
182 auto writer = std::make_shared<Buffers::AudioWriteProcessor>();
183 m_buffer_manager->add_processor(writer, op.m_target_audio_buffer,
184 Buffers::ProcessingToken::AUDIO_BACKEND);
185 op.m_attached_processor = writer;
186 }
187
188 std::static_pointer_cast<Buffers::AudioWriteProcessor>(op.m_attached_processor)
189 ->set_data(data_to_route);
190
191 } else if (op.m_target_graphics_buffer) {
192 queue_graphics_write(op, op.m_target_graphics_buffer, data_to_route);
193
194 } else if (op.m_target_audio_stream) {
195 detail::write_to_container(op.m_target_audio_stream, data_to_route, op.m_target_audio_channel);
196
197 } else if (op.m_target_graphics_stream) {
198 detail::write_to_container(op.m_target_graphics_stream, data_to_route);
199 }

217void BufferPipeline::process_fuse(BufferOperation& op, uint64_t cycle)
218{
219 std::vector<Kakshya::DataVariant> fusion_inputs;
220
221 for (auto& source_buffer : op.m_source_audio_buffers) {
222 bool should_process = op.m_capture.get_processing_control() == BufferCapture::ProcessingControl::ON_CAPTURE;
223 auto buffer_data = detail::extract_buffer_data(source_buffer, should_process);
224 fusion_inputs.push_back(buffer_data);
225 }
226
227 for (auto& source_container : op.m_source_audio_streams) {
228 auto container_data = detail::read_from_container(source_container, 0, 0);
229 fusion_inputs.push_back(container_data);
230 }
231
232 bool graphics_ready = true;
233 for (auto& source_buffer : op.m_source_graphics_buffers) {
234 if (auto buffer_data = read_graphics_buffer(op, source_buffer)) {
235 fusion_inputs.push_back(std::move(*buffer_data));
236 } else {
237 graphics_ready = false;
238 }
239 }
240
241 if (!graphics_ready) {
242 return;
243 }
244
245 for (auto& source_container : op.m_source_graphics_streams) {
246 fusion_inputs.push_back(detail::read_from_container(source_container, 0, 0));
247 }
248
249 if (op.m_fusion_function && !fusion_inputs.empty()) {
250 auto fused_data = op.m_fusion_function(fusion_inputs, cycle);
251
252 if (op.m_target_audio_buffer) {
253 detail::write_to_buffer(op.m_target_audio_buffer, fused_data);
254 } else if (op.m_target_graphics_buffer) {
255 queue_graphics_write(op, op.m_target_graphics_buffer, fused_data);
256 } else if (op.m_target_audio_stream) {
257 detail::write_to_container(op.m_target_audio_stream, fused_data, op.m_target_audio_channel);
258 } else if (op.m_target_graphics_stream) {
259 detail::write_to_container(op.m_target_graphics_stream, fused_data);
260 }
261
262 m_operation_data[&op] = fused_data;
263 }

278void BufferPipeline::process_modify(BufferOperation& op, uint64_t cycle)
279{
280 if (!m_buffer_manager) {
281 error<std::invalid_argument>(Journal::Component::Kriya,
282 Journal::Context::CoroutineScheduling,

289 op.m_attached_processor = m_buffer_manager->attach_quick_process(

292 } else {
293 op.m_attached_processor = m_buffer_manager->attach_quick_process(
294 op.m_audio_buffer_modifier,
295 op.m_target_audio_buffer, Buffers::ProcessingToken::AUDIO_BACKEND);
296 }
297 if (m_max_cycles != 0 && op.is_streaming()) {
298 op.m_modify_cycle_count = m_max_cycles - cycle;
299 }
300 }
301
302 if (op.m_modify_cycle_count > 0 && cycle >= op.m_modify_cycle_count - 1) {
303 if (op.m_attached_processor) {
304 if (op.m_target_graphics_buffer) {
305 m_buffer_manager->remove_processor(
306 op.m_attached_processor,
307 op.m_target_graphics_buffer);
308 } else {
309 m_buffer_manager->remove_processor(
310 op.m_attached_processor,
311 op.m_target_audio_buffer);

345void BufferPipeline::queue_graphics_write(BufferOperation& op, const std::shared_ptr<Buffers::VKBuffer>& target, const Kakshya::DataVariant& data)
346{

350 }
351

356 "BufferPipeline has no BufferManager for graphics buffer write");
357 }

360 error<std::invalid_argument>(Journal::Component::Kriya,
361 Journal::Context::CoroutineScheduling,
362 std::source_location::current(),

364 }
365

367 auto writer = std::make_shared<Buffers::DataWriteProcessor>();
368 m_buffer_manager->add_processor(writer, target, Buffers::ProcessingToken::GRAPHICS_BACKEND);
369 op.m_attached_processor = writer;
370 }
371
372 std::static_pointer_cast<Buffers::DataWriteProcessor>(op.m_attached_processor)->set_data(data);
