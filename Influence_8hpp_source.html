74/**
75 * @class Influence
76 * @brief A value of any type, built from the influence context and handed to its targets.
77 *
78 * T is whatever the targets need: a struct for a shader block, a vector, a
79 * matrix, a Tendency, a DataVariant. A target is any callable taking const T&
80 * and decides what the value affects. On each invoke the producer runs once and
81 * every target gets the value, in the order they were added.
82 *
83 * How many targets an Influence may have is decided by its owner: an Emitter
84 * keeps one, an Agent keeps as many as are added.
85 *
86 * Context is what the producer reads. Emitter and Agent supply an
87 * InfluenceContext, an Expanse supplies a CrossingContext. Targets take only
88 * const T&, so one target works with either.
89 */
90template <typename T, typename Context = InfluenceContext>
91class Influence {
92public:
93 using Producer = std::function<T(const Context&)>;
94 using Target = std::function<void(const T&)>;
95
96 explicit Influence(Producer producer)
97 : m_producer(std::move(producer))
98 {
99 }
100
101 /** @brief Add a target. Returns this influence for chaining. */
102 Influence& add_target(Target target)
103 {
104 m_targets.push_back(std::move(target));
105 return *this;
106 }
107
108 /** @brief Remove every target. */
109 void clear_targets() { m_targets.clear(); }
110
111 /** @brief Number of targets currently added. */
112 [[nodiscard]] size_t target_count() const { return m_targets.size(); }
113
114 /** @brief Build the value from @p ctx and hand it to every target. */
115 void invoke(const Context& ctx) const
116 {
117 if (!m_producer) {
118 return;
119 }
120 const T value = m_producer(ctx);
121 for (const auto& target : m_targets) {
122 target(value);
123 }
124 }
125
126private:
127 Producer m_producer;
128 std::vector<Target> m_targets;
129};
