46bool CompositeContainer::materialize_next()
47{
48 if (at_end())
49 return false;
50
51 auto traversal = traverse(m_batch_size);
52 traversal.seek(m_next_element);
53 const auto window = traversal.current();
54 if (!window || window->size() == 0)
55 return false;
56
57 auto next = copy_slice(*window);
58 if (!next)
59 return false;
60
61 m_materialized_data = std::move(*next);
62 m_next_element += window->size();

99std::vector<CompositeArray> CompositeContainer::get_segments_data(const std::vector<RegionSegment>& segments) const
100{
101 std::vector<CompositeArray> result;
102 result.reserve(segments.size());
103
104 for (const auto& segment : segments) {
105 const auto& region = segment.source_region;
106
107 if (region.start_coordinates.size() != 1 || region.end_coordinates.size() != 1
108 || segment.offset_in_region.size() != 1 || segment.segment_size.size() != 1)
109 continue;
110
111 const uint64_t source_start = region.start_coordinates[0];
112 const uint64_t source_end = region.end_coordinates[0];
113 const uint64_t offset = segment.offset_in_region[0];
114 const uint64_t count = segment.segment_size[0];
115
116 if (source_start > source_end || offset > source_end - source_start || count == 0)
117 continue;
118
119 const uint64_t start = source_start + offset;
120 if (count - 1 > source_end - start)
121 continue;
122
123 auto copied = get_region_data(Region(
124 std::vector<uint64_t> { start }, std::vector<uint64_t> { start + count - 1 }));
125
126 if (!copied.empty())
127 result.push_back(std::move(copied.front()));
128 }

132void CompositeContainer::set_region_data(const Region& region, const std::vector<CompositeArray>& data)
133{
134 const auto target = m_data.slice(region);
135 if (!target)
136 return;
137
138 size_t supplied = 0;
139 for (const auto& array : data) {
140 if (array.size() > target->size() - supplied)
141 return;
142 supplied += array.size();
143 }
144 if (supplied != target->size())
145 return;
146
147 CompositeArray replacement(m_data.layout());
148 CompositeInsertion insertion(replacement);
149
150 const auto before = m_data.slice(0, target->start());
151 const auto after_start = target->start() + target->size();
152 const auto after = m_data.slice(after_start, m_data.size() - after_start);
153
154 if (!before || !after || !append_slice(insertion, *before))
155 return;
156
157 for (const auto& array : data) {
158 const auto selected = array.slice(0, array.size());
159 if (!selected || !append_slice(insertion, *selected))
160 return;
161 }
162 if (!append_slice(insertion, *after))
163 return;
164
165 m_data = std::move(replacement);
166}
