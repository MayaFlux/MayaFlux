109void GpuDispatchCore::prepare_gpu_inputs(
110 const std::vector<std::vector<double>>& channels,
111 const DataStructureInfo& structure_info)
112{
113 const auto& key = dispatch_key();
114
115 flatten_channels_to_staging(channels, structure_info);
116 const size_t float_byte_size = m_staging_floats.size() * sizeof(float);
117
118 const size_t fallback_bytes = float_byte_size > 0
119 ? float_byte_size
120 : Kakshya::ContainerDataStructure::get_total_elements(structure_info.dimensions) * sizeof(float);
121
122 std::vector<GpuResourceManager::ImageBind> image_binds;
123
124 for (auto b : m_bindings) {
125 const size_t idx = b.binding;
126
127 if (idx < m_binding_data.size() && !m_binding_data[idx].empty()) {
128 m_resources.ensure_buffer(key, idx, m_binding_data[idx].size());
129 m_resources.upload_raw(key, idx, m_binding_data[idx].data(), m_binding_data[idx].size());
130 continue;
131 }
132
133 if (b.direction == GpuBufferBinding::Direction::OUTPUT) {
134 const auto et = b.element_type;
135 if (et == GpuBufferBinding::ElementType::IMAGE_STORAGE
136 || et == GpuBufferBinding::ElementType::IMAGE_SAMPLED) {
137 } else {
138 const size_t sz = (idx < m_output_size_overrides.size() && m_output_size_overrides[idx] > 0)
139 ? m_output_size_overrides[idx]
140 : fallback_bytes;
141 m_resources.ensure_buffer(key, idx, sz);
142 if (idx < m_output_size_overrides.size() && m_output_size_overrides[idx] > 0) {
143 std::vector<uint8_t> zeros(sz, 0);
144 m_resources.upload_raw(key, idx, zeros.data(), sz);
145 }
146 continue;
147 }
148 }
149
150 switch (b.element_type) {
151 case GpuBufferBinding::ElementType::PASSTHROUGH:
152 if (idx < m_passthrough_bytes.size() && !m_passthrough_bytes[idx].empty()) {
153 m_resources.ensure_buffer(key, idx, m_passthrough_bytes[idx].size());
154 m_resources.upload_raw(key, idx, m_passthrough_bytes[idx].data(),
155 m_passthrough_bytes[idx].size());
156 }
157 break;
158
159 case GpuBufferBinding::ElementType::IMAGE_STORAGE: {
160 if (idx >= m_image_bindings.size() || !m_image_bindings[idx].image)
161 continue;
162 auto& img = m_image_bindings[idx].image;
163 if (img->get_current_layout() != vk::ImageLayout::eGeneral) {
164 m_resources.transition_image(img, img->get_current_layout(),
165 vk::ImageLayout::eGeneral);
166 }
167 image_binds.push_back({ .index = idx, .image = img, .sampler = nullptr, .spec = b });
168 } break;
169
170 case GpuBufferBinding::ElementType::IMAGE_SAMPLED: {
171 if (idx >= m_image_bindings.size() || !m_image_bindings[idx].image)
172 continue;
173 auto& img = m_image_bindings[idx].image;
174 auto sampler = m_image_bindings[idx].sampler;
175 if (img->get_current_layout() != vk::ImageLayout::eShaderReadOnlyOptimal) {
176 m_resources.transition_image(img, img->get_current_layout(),
177 vk::ImageLayout::eShaderReadOnlyOptimal);
178 }
179 image_binds.push_back({ .index = idx, .image = img, .sampler = sampler, .spec = b });
180 } break;
181
182 case GpuBufferBinding::ElementType::UINT32:
183 case GpuBufferBinding::ElementType::INT32:
184 if (!channels.empty()) {
185 const size_t raw_bytes = channels[0].size()
186 * (b.element_type == GpuBufferBinding::ElementType::UINT32
187 ? sizeof(uint32_t)
188 : sizeof(int32_t));
189 m_resources.ensure_buffer(key, idx, raw_bytes);
190 m_resources.upload_raw(key, idx,
191 reinterpret_cast<const uint8_t*>(channels[0].data()),
192 raw_bytes);
193 }
194 break;
195
196 case GpuBufferBinding::ElementType::FLOAT32:
197 default:
198 if (!m_native_staging_bytes.empty()) {
199 m_resources.ensure_buffer(key, idx, m_native_staging_bytes.size());
200 m_resources.upload_raw(key, idx, m_native_staging_bytes.data(), m_native_staging_bytes.size());
201 } else {
202 m_resources.ensure_buffer(key, idx, float_byte_size);
203 m_resources.upload(key, idx, m_staging_floats.data(), float_byte_size);
204 }
205 break;
206 }
207 }
208
209 if (!image_binds.empty())
210 m_resources.bind_images_batch(key, image_binds);
211}

213std::array<uint32_t, 3> GpuDispatchCore::calculate_dispatch_size(
214 size_t total_elements, const DataStructureInfo& structure_info) const
215{
216 uint64_t sz_x = 0, sz_y = 0, sz_z = 0;
217 for (const auto& dim : structure_info.dimensions) {
218 switch (dim.role) {
219 case Kakshya::DataDimension::Role::SPATIAL_X:
220 sz_x = dim.size;
221 break;
222 case Kakshya::DataDimension::Role::SPATIAL_Y:
223 sz_y = dim.size;
224 break;
225 case Kakshya::DataDimension::Role::SPATIAL_Z:
226 sz_z = dim.size;
227 break;
228 default:
229 break;
230 }
231 }
232
233 const auto& ws = m_gpu_config.workgroup_size;
234 if (sz_x > 0) {
235 return {
236 static_cast<uint32_t>((sz_x + ws[0] - 1) / ws[0]),
237 sz_y > 0 ? static_cast<uint32_t>((sz_y + ws[1] - 1) / ws[1]) : 1U,
238 sz_z > 0 ? static_cast<uint32_t>((sz_z + ws[2] - 1) / ws[2]) : 1U,
239 };
240 }
241
242 return { static_cast<uint32_t>((total_elements + ws[0] - 1) / ws[0]), 1U, 1U };
243}

375GpuDispatchCore::DependencyPlan GpuDispatchCore::prepare_dependency(const std::vector<DependencyStage>& stages)
376{
377 const GpuComputeConfig original_config = m_gpu_config;
378 const auto original_bindings = m_bindings;
379 const auto original_image_bindings = m_image_bindings;
380 const auto original_binding_data = m_binding_data;
381 const auto original_passthrough_bytes = m_passthrough_bytes;
382 const auto original_push_constants = m_push_constants;
383
384 DependencyPlan plan;
385 auto& keys = plan.keys;
386 auto& groups_per_key = plan.groups;
387 auto& pc_per_key = plan.push_constants;
388 auto& hazards_per_key = plan.hazards;
389 auto& indirect_per_key = plan.indirect;
390
391 keys.reserve(stages.size());
392 groups_per_key.reserve(stages.size());
393 pc_per_key.reserve(stages.size());
394 hazards_per_key.reserve(stages.size());
395 indirect_per_key.reserve(stages.size());
396
397 for (const auto& stage : stages) {
398 m_gpu_config = stage.config;
399 update_dispatch_key_cache();
400 m_bindings = declare_buffer_bindings();
401 m_image_bindings.clear();
402 m_binding_data.clear();
403 m_passthrough_bytes.clear();
404 m_push_constants.clear();
405
406 stage.stage_fn(*this);
407
408 if (!ensure_gpu_ready()) {
409 error<std::runtime_error>(Journal::Component::Yantra,
410 Journal::Context::BufferProcessing,
411 std::source_location::current(),
412 "GpuDispatchCore: dispatch_core_dependency failed to initialise key '{}'",
413 stage.config.shader_path);
414 }
415
416 prepare_gpu_inputs({}, {});
417
418 bind_all_descriptors();
419
420 keys.push_back(dispatch_key());
421 groups_per_key.push_back(stage.explicit_groups ? *stage.explicit_groups : calculate_dispatch_size(largest_binding_data_element_count(), {}));
422 pc_per_key.push_back(m_push_constants);
423 hazards_per_key.push_back(stage.hazard_fn ? stage.hazard_fn(*this) : std::vector<Portal::Graphics::HazardResource> {});
424
425 Portal::Graphics::IndirectDispatch indirect;
426 if (stage.indirect_groups) {
427 indirect.buffer = m_resources.shared_buffer_handle(stage.indirect_groups->set, stage.indirect_groups->binding).buffer;
428 indirect.offset = stage.indirect_groups->offset_bytes;
429 }
430 indirect_per_key.push_back(indirect);
431 }
432
433 m_gpu_config = original_config;
434 update_dispatch_key_cache();
435 m_bindings = original_bindings;
436 m_image_bindings = original_image_bindings;
437 m_binding_data = original_binding_data;
438 m_passthrough_bytes = original_passthrough_bytes;
439 m_push_constants = original_push_constants;
440
441 return plan;
442}

448std::vector<float> GpuDispatchCore::readback_primary(size_t float_count)
449{
450 const size_t idx = find_first_output_index();

452 for (const auto& b : m_bindings) {

464 m_resources.download(dispatch_key(), idx, out.data(), byte_size);

500void GpuDispatchCore::flatten_channels_to_staging(
501 const std::vector<std::vector<double>>& channels,
502 const DataStructureInfo& structure_info)
503{
504 m_staging_floats.clear();
505 m_native_staging_bytes.clear();
506
507 if (Kakshya::is_structured_modality(structure_info.modality))
508 return;
509
510 bool all_inputs_staged = !m_bindings.empty();
511 for (size_t i = 0; i < m_bindings.size(); ++i) {
512 if (m_bindings[i].direction == GpuBufferBinding::Direction::OUTPUT)
513 continue;
514 if (i >= m_binding_data.size() || m_binding_data[i].empty()) {
515 all_inputs_staged = false;
516 break;
517 }
518 }
519 if (all_inputs_staged)
520 return;
521
522 size_t total = 0;
523
524 for (const auto& ch : channels)
525 total += ch.size();
526 m_staging_floats.reserve(total);
527
528 for (const auto& ch : channels) {
529 for (double v : ch)
530 m_staging_floats.push_back(static_cast<float>(v));
531 }
532}

534void GpuDispatchCore::flatten_native_variants_to_staging(
535 const std::vector<Kakshya::DataVariant>& variants,
536 const DataStructureInfo& structure_info)
537{
538 m_native_staging_bytes.clear();
539
540 if (variants.empty() || Kakshya::is_structured_modality(structure_info.modality))
541 return;
542
543 size_t total_bytes = 0;
544 for (const auto& v : variants) {
545 std::visit([&](const auto& vec) {
546 total_bytes += vec.size() * sizeof(typename std::decay_t<decltype(vec)>::value_type);
547 },
548 v);
549 }
550
551 m_native_staging_bytes.reserve(total_bytes);
552
553 for (const auto& v : variants) {
554 std::visit([&](const auto& vec) {
555 using T = typename std::decay_t<decltype(vec)>::value_type;
556 const auto* bytes = reinterpret_cast<const uint8_t*>(vec.data());
557 m_native_staging_bytes.insert(
558 m_native_staging_bytes.end(),
559 bytes,
560 bytes + vec.size() * sizeof(T));
561 },
562 v);
563 }
