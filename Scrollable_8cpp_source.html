12 /// @brief (position_frac, extent_frac) pair an indicator's geometry function reacts to.
13 glm::vec2 scroll_extent(const Kinesis::ScrollState& scroll) noexcept
14 {
15 const float content_h = scroll.content_bounds.height();
16 const float viewport_h = scroll.viewport_bounds.height();
17 const float max_offset_y = std::max(0.F, content_h - viewport_h);
18
19 const float extent_frac = content_h > 0.F ? std::clamp(viewport_h / content_h, 0.F, 1.F) : 1.F;
20 const float position_frac = max_offset_y > 0.F ? std::clamp(scroll.offset.y / max_offset_y, 0.F, 1.F) : 0.F;
21
22 return { position_frac, extent_frac };
23 }

25 void reflow(
26 Kinesis::ScrollState& scroll,
27 MappedState<glm::vec2>& offset,
28 std::vector<Tracked>& tracked,
29 Layer& layer,
30 Mapped<glm::vec2>* indicator)
31 {
32 offset.write(scroll.offset);
33 for (auto& t : tracked) {
34 const Kinesis::AABB2D shifted = t.base_bounds.translated(scroll.offset);
35 layer.set_bounds(t.id, shifted);
36 t.reposition(shifted);
37 }
38
39 if (indicator && indicator->state) {
40 indicator->state->write(scroll_extent(scroll));
41 indicator->sync(&layer);
42 }
43 }

149 scroll_state->content_bounds.height() - scroll_state->viewport_bounds.height());

156 Kinesis::apply_scroll(*scroll_state, { 0.F, target - scroll_state->offset.y });
157 reflow(*scroll_state, *scroll_offset, *tracked, surface.layer(), ind.get());
