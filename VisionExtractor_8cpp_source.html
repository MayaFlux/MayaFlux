138 /** @brief Run a reduce-context dispatch of x invocations and return output buffer 6 as floats. */
139 [[nodiscard]] std::vector<float> dispatch_reduce(TextureExecutionContext& ctx, uint32_t invocations, size_t output_bytes)
140 {
141 ctx.set_output_size(6, output_bytes);
142 ctx.set_output_dimensions(invocations, 1);
143 dispatch_and_wait(ctx);
144
145 const auto result = ctx.collect_result();
146 const auto it = result.aux.find(6);
147 if (it == result.aux.end())
148 return {};
149
150 std::vector<float> out(it->second.size() / sizeof(float));
151 std::memcpy(out.data(), it->second.data(), out.size() * sizeof(float));
152 return out;
153 }

299std::optional<size_t> VisionExtractor::select_element(
300 const Kinesis::Vision::VisionAnalysis& analysis,
301 const ElementFocus& focus)
302{
303 if (!analysis.find_elements || analysis.find_elements->boxes.empty())
304 return std::nullopt;
305
306 const auto& elements = *analysis.find_elements;
307 const auto count = elements.boxes.size();
308
309 std::vector<float> metrics;
310 metrics.reserve(count * 6);
311 for (size_t i = 0; i < count; ++i) {
312 const auto& box = elements.boxes[i];
313 const uint32_t label = box.label_id != 0 ? box.label_id : static_cast<uint32_t>(i + 1U);
314 const auto contour = std::ranges::find_if(elements.contours,
315 [label](const auto& c) { return c.label_id == label; });
316
317 metrics.push_back(contour != elements.contours.end() ? contour->area : box.w * box.h);
318 metrics.push_back(i < elements.shapes.size() ? elements.shapes[i].compactness : 0.0F);
319 metrics.push_back(i < elements.shapes.size() ? elements.shapes[i].aspect_ratio : 0.0F);
320 metrics.push_back(i < elements.nearest_neighbor_distance_px.size() ? elements.nearest_neighbor_distance_px[i] : 0.0F);
321 metrics.push_back(box.x + box.w * 0.5F);
322 metrics.push_back(box.y + box.h * 0.5F);
323 }
324
325 auto& ctx = reduce_context();
326 use_shader(ctx, "element_select.comp.spv", k_wg_reduce, sizeof(SelectPC));
327 ctx.set_binding_data(3, std::span<const float>(metrics));
328 ctx.set_push_constants(SelectPC {
329 .count = static_cast<uint32_t>(count),
330 .criterion = static_cast<uint32_t>(focus.criterion),
331 .largest = focus.largest ? 1U : 0U,
332 .px = focus.point.x,
333 .py = focus.point.y,
334 });
335
336 const auto raw = dispatch_reduce(ctx, 256, 2 * sizeof(float));
337 if (raw.empty())
338 return std::nullopt;
339
340 const auto index = static_cast<size_t>(std::lround(raw[0]));
341 return index < count ? std::optional<size_t> { index } : std::nullopt;
342}

399std::shared_ptr<Core::VKImage> VisionExtractor::fill_elements(
400 const std::shared_ptr<Core::VKImage>& source,
401 const Kinesis::Vision::FindElementsAnalysis& elements,
402 const std::vector<size_t>& indices,
403 uint32_t tile_w, uint32_t tile_h)
404{
405 if (indices.empty() || tile_w == 0 || tile_h == 0)
406 return nullptr;
407
408 std::vector<float> points;
409 std::vector<uint32_t> polygons;
410 std::vector<uint32_t> table;
411 table.reserve(indices.size() * 6);
412
413 for (const auto index : indices) {
414 const auto& box = elements.boxes[index];
415 const uint32_t label = box.label_id != 0 ? box.label_id : static_cast<uint32_t>(index + 1U);
416
417 const auto first_polygon = static_cast<uint32_t>(polygons.size() / 2);
418 uint32_t polygon_count = 0;
419 for (const auto& contour : elements.contours) {
420 const bool owned = contour.label_id == label || (contour.label_id == 0 && contour.parent_label == label);
421 if (!owned || contour.points.size() < 3)
422 continue;
423
424 polygons.push_back(static_cast<uint32_t>(points.size() / 2));
425 polygons.push_back(static_cast<uint32_t>(contour.points.size()));
426 for (const auto& p : contour.points)
427 points.insert(points.end(), { p.x, p.y });
428 ++polygon_count;
429 }
430
431 table.insert(table.end(), {
432 std::bit_cast<uint32_t>(box.x),
433 std::bit_cast<uint32_t>(box.y),
434 std::bit_cast<uint32_t>(box.w),
435 std::bit_cast<uint32_t>(box.h),
436 first_polygon,
437 polygon_count,
438 });
439 }
440
441 if (points.empty())
442 points.assign(2, 0.0F);
443 if (polygons.empty())
444 polygons.assign(2, 0U);
445
446 const auto count = static_cast<uint32_t>(indices.size());
447
448 auto& ctx = image_context();
449 use_shader(ctx, "contour_fill.comp.spv", k_wg_image, sizeof(AtlasPC));
450 ctx.stage_image(source);
451 ctx.set_binding_data(3, std::span<const float>(points));
452 ctx.set_binding_data(4, std::span<const uint32_t>(polygons));
453 ctx.set_binding_data(5, std::span<const uint32_t>(table));
454 ctx.set_push_constants(AtlasPC { .tile_w = tile_w, .tile_h = tile_h, .count = count });
455 return dispatch_image(ctx, tile_w * count, tile_h);
456}

504std::vector<Kinesis::Vision::FieldSample> VisionExtractor::sample_regions(
505 const std::shared_ptr<Core::VKImage>& source,
506 const Kinesis::Vision::VisionAnalysis& analysis,
507 uint32_t source_w, uint32_t source_h)
508{
509 if (!analysis.find_elements || analysis.find_elements->boxes.empty())
510 return {};
511
512 const auto& boxes = analysis.find_elements->boxes;
513 const auto count = static_cast<uint32_t>(boxes.size());
514
515 std::vector<uint32_t> rects;
516 rects.reserve(boxes.size() * 4);
517 for (const auto& b : boxes) {
518 const auto rect = pixel_rect(b, source_w, source_h);
519 rects.insert(rects.end(), rect.begin(), rect.end());
520 }
521
522 auto& ctx = reduce_context();
523 use_shader(ctx, "region_sample_batch.comp.spv", k_wg_reduce, sizeof(BatchCountPC));
524 ctx.stage_image(source);
525 ctx.set_binding_data(3, std::span<const uint32_t>(rects));
526 ctx.set_push_constants(BatchCountPC { .count = count });
527
528 const auto raw = dispatch_reduce(ctx, 256U * count, static_cast<size_t>(count) * 3 * sizeof(glm::vec4));
529 std::vector<Kinesis::Vision::FieldSample> out;
530 if (raw.size() < static_cast<size_t>(count) * 12)
531 return out;
532
533 out.reserve(count);
534 for (uint32_t i = 0; i < count; ++i)
535 out.push_back(field_sample(raw.data() + static_cast<size_t>(i) * 12));
536 return out;
537}

667std::shared_ptr<Core::VKImage> VisionExtractor::annotate(
668 const std::shared_ptr<Core::VKImage>& source,
669 const Kinesis::Vision::VisionAnalysis& analysis,
670 uint32_t source_w, uint32_t source_h)
671{
672 const glm::vec4 yellow { 1.0F, 1.0F, 0.0F, 1.0F };
673 const glm::vec4 cyan { 0.0F, 1.0F, 1.0F, 1.0F };
674 const glm::vec4 green { 0.0F, 1.0F, 0.0F, 1.0F };
675 const glm::vec4 red { 1.0F, 0.0F, 0.0F, 1.0F };
676 const glm::vec4 white { 1.0F, 1.0F, 1.0F, 1.0F };
677 const glm::vec4 magenta { 1.0F, 0.0F, 1.0F, 1.0F };
678
679 const glm::vec2 half { 3.0F / static_cast<float>(source_w), 3.0F / static_cast<float>(source_h) };
680
681 std::vector<float> segments;
682
683 if (analysis.find_elements) {
684 for (const auto& box : analysis.find_elements->boxes)
685 add_rect(segments, box, yellow);
686 for (const auto& contour : analysis.find_elements->contours) {
687 const auto n = contour.points.size();
688 for (size_t i = 0; n > 1 && i < n; ++i)
689 add_line(segments, contour.points[i], contour.points[(i + 1) % n], cyan);
690 }
691 }
692
693 if (analysis.track_objects && !analysis.track_objects->tracks.empty()) {
694 for (const auto& t : analysis.track_objects->tracks) {
695 add_line(segments, t.previous, t.position, t.tracked ? green : red);
696 add_cross(segments, t.position, half, t.tracked ? green : red);
697 }
698 add_rect(segments, analysis.track_objects->bounds, white);
699 add_cross(segments, analysis.track_objects->centroid, half, white);
700 }
701
702 if (analysis.detect_features && !analysis.detect_features->keypoints.empty()) {
703 for (const auto& k : analysis.detect_features->keypoints)
704 add_cross(segments, k.position, half, magenta);
705 add_rect(segments, analysis.detect_features->bounds, white);
706 }
707
708 const auto count = static_cast<uint32_t>(segments.size() / 8);
709 if (segments.empty())
710 segments.assign(8, 0.0F);
711
712 auto& ctx = image_context();
713 use_shader(ctx, "vision_crop.comp.spv", k_wg_image, sizeof(CropPC));
714 ctx.stage_image(source);
715 ctx.set_binding_data(3, std::span<const float>(segments));
716 ctx.set_push_constants(CropPC {
717 .src_x = 0.0F, .src_y = 0.0F, .src_w = 1.0F, .src_h = 1.0F,
718 .out_w = source_w, .out_h = source_h });
719 auto image = dispatch_image(ctx, source_w, source_h);
720
721 if (count == 0)
722 return image;
723
724 ctx.swap_shader({
725 .shader_path = "overlay_segments.comp.spv",
726 .workgroup_size = k_wg_image,
727 .push_constant_size = sizeof(SegmentsPC),
728 });
729 ctx.set_push_constants(SegmentsPC { .count = count, .width = source_w, .height = source_h });
730 ctx.set_output_dimensions(source_w, source_h);
731 dispatch_and_wait(ctx);
732
733 return ctx.get_output_image(0);
734}

736std::shared_ptr<Core::VKImage> VisionExtractor::resolve_image(
737 const std::shared_ptr<Kakshya::SignalSourceContainer>& source,
738 size_t index)
739{
740 if (!source)
741 return nullptr;
742
743 if (auto tc = std::dynamic_pointer_cast<Kakshya::TextureContainer>(source))
744 return tc->to_image(static_cast<uint32_t>(index));
745
746 if (auto wc = std::dynamic_pointer_cast<Kakshya::WindowContainer>(source))
747 return wc->to_image();
748
749 auto vc = std::dynamic_pointer_cast<Kakshya::VideoStreamContainer>(source);
750 if (!vc)
751 return nullptr;
752
753 const uint32_t w = vc->get_width();
754 const uint32_t h = vc->get_height();
755
756 const void* raw = source->get_raw_data();
757 if (!raw)
758 return nullptr;
759
760 if (!m_upload_image || m_upload_w != w || m_upload_h != h) {
761 m_upload_image = Portal::Graphics::TextureLoom::instance().create_2d(
762 w, h, Portal::Graphics::ImageFormat::RGBA8, nullptr);
763 m_upload_w = w;
764 m_upload_h = h;
765 }
766 if (!m_upload_image)
767 return nullptr;
768
769 Portal::Graphics::TextureLoom::instance().upload_data(m_upload_image, raw, m_upload_image->get_size_bytes());
770 return m_upload_image;
771}
772
773VisionExtractor::output_type VisionExtractor::run_operation(const input_type& input)
774{
775 output_type output;
776 output.metadata = input.metadata;
777
778 const auto supplied = Kakshya::get_metadata_value<std::shared_ptr<Core::VKImage>>(input.metadata, "vision_image");
779 const bool has_image = supplied && *supplied;
780
781 if (!has_image && !input.data) {
782 output.metadata["error"] = std::string("VisionExtractor: missing container");
783 return output;
784 }
785
786 const auto analysis = Kakshya::get_metadata_value<Kinesis::Vision::VisionAnalysis>(
787 input.metadata, "vision_analysis");
788 if (!analysis) {
789 output.metadata["error"] = std::string("VisionExtractor: missing vision_analysis metadata");
790 return output;
791 }
792
793 const auto index = Kakshya::get_metadata_value<size_t>(input.metadata, "container_index").value_or(0);
794 const auto image = has_image ? *supplied : resolve_image(input.data, index);
795 if (!image) {
796 output.metadata["error"] = std::string("VisionExtractor: could not resolve image from container");
797 return output;
798 }
799
800 const uint32_t w = image->get_width();
801 const uint32_t h = image->get_height();
802
803 switch (m_mode) {
804 case VisionExtractMode::Crop:
805 output.metadata["vision_extraction"] = crop(image, *analysis, m_out_w, m_out_h);
806 break;
807 case VisionExtractMode::Sample:
808 output.metadata["vision_extraction"] = sample(image, *analysis, w, h);
809 break;
810 case VisionExtractMode::Patches:
811 output.metadata["vision_extraction"] = patches(image, *analysis, m_patch_w, m_patch_h, w, h);
812 break;
813 case VisionExtractMode::Crops:
814 output.metadata["vision_extraction"] = crops(image, *analysis, m_tile_w, m_tile_h);
815 break;
816 case VisionExtractMode::SampleRegions:
817 output.metadata["vision_extraction"] = sample_regions(image, *analysis, w, h);
818 break;
819 case VisionExtractMode::SamplePoints:
820 output.metadata["vision_extraction"] = sample_points(image, *analysis, m_sample_radius, w, h);
821 break;
822 case VisionExtractMode::Silhouettes:
823 output.metadata["vision_extraction"] = silhouettes(image, *analysis, m_tile_w, m_tile_h);
824 break;
825 case VisionExtractMode::Equalize:
826 output.metadata["vision_extraction"] = equalize(image, *analysis, w, h);
827 break;
828 case VisionExtractMode::Expose:
829 output.metadata["vision_extraction"] = expose(image, *analysis, m_exposure_target, w, h);
830 break;
831 case VisionExtractMode::Edges:
832 output.metadata["vision_extraction"] = edges(image, *analysis, w, h);
833 break;
834 case VisionExtractMode::MotionMask:
835 output.metadata["vision_extraction"] = motion_mask(image, *analysis, w, h);
836 break;
837 case VisionExtractMode::Annotate:
838 output.metadata["vision_extraction"] = annotate(image, *analysis, w, h);
839 break;
840 case VisionExtractMode::SelectElement:
841 if (const auto selected = select_element(*analysis, m_focus.value_or(ElementFocus {})))
842 output.metadata["vision_extraction"] = *selected;
843 break;
844 }
845
846 return output;
847}
