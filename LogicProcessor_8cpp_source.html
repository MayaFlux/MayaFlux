23bool LogicProcessor::generate(size_t num_samples, const std::vector<double>& input_data)
24{
25 if (!m_logic || input_data.empty()) {
26 return false;
27 }
28
29 if (m_pending_logic) {
30 m_logic = m_pending_logic;
31 m_pending_logic.reset();
32 m_use_internal = true;
33 }
34
35 m_logic_data.resize(num_samples, 0);
36
37 if (m_reset_between_buffers) {
38 m_logic->reset();
39 for (const auto& node : m_plane_nodes) {
40 node->reset();
41 }
42 }
43
44 const bool guarded = m_logic->m_state.load() != Nodes::NodeState::INACTIVE;
45 if (guarded) {
46 m_logic->save_state();
47 }
48
49 const bool parallel = m_logic->get_mode() == Nodes::Generator::LogicMode::MULTI_INPUT;
50
51 if (m_generations > 0 && parallel) {
52 generate_spatial(num_samples, input_data);
53 } else if (m_plane_bits > 0) {
54 generate_planes(num_samples, input_data);
55 } else if (!m_keys.empty() && parallel) {
56 generate_keyed(num_samples, input_data);
57 } else {
58 generate_temporal(num_samples, input_data);
59 }
60
61 if (guarded) {
62 m_logic->restore_state();
63 }
64
65 m_has_generated_data = true;
66 return true;
67}

102void LogicProcessor::generate_planes(size_t num_samples, const std::vector<double>& input_data)
103{
104 const uint32_t bits = std::clamp(m_plane_bits, uint32_t { 1 }, uint32_t { 24 });
105
106 if (m_plane_nodes.size() != bits || m_plane_template != m_logic) {
107 m_plane_nodes.clear();
108 for (uint32_t plane = 0; plane < bits; ++plane) {
109 m_plane_nodes.push_back(m_logic->clone());
110 }
111 m_plane_template = m_logic;
112 }
113
114 const bool parallel = m_logic->get_mode() == Nodes::Generator::LogicMode::MULTI_INPUT;
115
116 lock_keys();
117 const size_t key_count = parallel ? m_live_keys.size() : 0;
118 m_key_words.assign(key_count, 0);
119 m_slots.assign(key_count + 1, 0.0);
120
121 const size_t available = input_data.size();
122 const uint32_t silence = Kinesis::Discrete::to_word(0.0, bits);
123
124 for (size_t i = 0; i < num_samples; ++i) {
125 const uint32_t own = Kinesis::Discrete::to_word(i < available ? input_data[i] : 0.0, bits);
126 for (size_t k = 0; k < key_count; ++k) {
127 const auto& key = m_live_keys[k];
128 m_key_words[k] = (key && i < key->get_data().size())
129 ? Kinesis::Discrete::to_word(key->get_data()[i], bits)
130 : silence;
131 }
132
133 uint32_t word = 0;
134 for (uint32_t plane = 0; plane < bits; ++plane) {
135 const auto bit_of = [plane](uint32_t w) { return static_cast<double>((w >> plane) & 1U); };
136
137 double out = 0.0;
138 if (parallel) {
139 for (size_t k = 0; k < key_count; ++k) {
140 m_slots[k] = bit_of(m_key_words[k]);
141 }
142 m_slots.back() = bit_of(own);
143 out = m_plane_nodes[plane]->process_multi_input(m_slots);
144 } else {
145 out = m_plane_nodes[plane]->process_sample(bit_of(own));
146 }
147 word |= (out > 0.5 ? 1U : 0U) << plane;
148 }
149
150 m_logic_data[i] = Kinesis::Discrete::from_word(word, bits);
151 }

154void LogicProcessor::generate_spatial(size_t num_samples, const std::vector<double>& input_data)
155{
156 if (num_samples == 0) {
157 return;
158 }
159
160 const size_t available = input_data.size();
161 const size_t width = 2 * m_radius + 1;
162 const auto count = static_cast<std::ptrdiff_t>(num_samples);
163
164 m_cells.assign(num_samples, 0);
165 m_next_cells.assign(num_samples, 0);
166 m_slots.assign(width, 0.0);
167
168 for (size_t i = 0; i < num_samples; ++i) {
169 uint8_t cell = (i < available && input_data[i] > 0.0) ? 1 : 0;
170 if (m_memory && i < m_previous_cells.size()) {
171 cell ^= m_previous_cells[i];
172 }
173 m_cells[i] = cell;
174 }
175
176 for (size_t generation = 0; generation < m_generations; ++generation) {
177 for (std::ptrdiff_t i = 0; i < count; ++i) {
178 for (size_t k = 0; k < width; ++k) {
179 std::ptrdiff_t at = i + static_cast<std::ptrdiff_t>(m_radius) - static_cast<std::ptrdiff_t>(k);
180 double cell = 0.0;
181 if (at >= 0 && at < count) {
182 cell = m_cells[static_cast<size_t>(at)];
183 } else if (m_wrap) {
184 at = ((at % count) + count) % count;
185 cell = m_cells[static_cast<size_t>(at)];
186 }
187 m_slots[k] = cell;
188 }
189 m_next_cells[static_cast<size_t>(i)] = m_logic->process_multi_input(m_slots) > 0.5 ? 1 : 0;
190 }
191 m_cells.swap(m_next_cells);
192 }
193
194 for (size_t i = 0; i < num_samples; ++i) {
195 m_logic_data[i] = m_cells[i];
196 }
197 if (m_memory) {
198 m_previous_cells = m_cells;
199 }
200}

236bool LogicProcessor::apply(const std::shared_ptr<Buffer>& buffer, ModulationFunction modulation_func)
237{
238 if (!buffer || !m_has_generated_data) {
239 return false;
240 }
241
242 auto& buffer_data = std::dynamic_pointer_cast<AudioBuffer>(buffer)->get_data();
243 size_t min_size = std::min(m_logic_data.size(), buffer_data.size());
244
245 if (!modulation_func) {
246 switch (m_modulation_type) {
247 case ModulationType::REPLACE:
248 modulation_func = [](double logic_val, double /*buffer_val*/) {
249 return logic_val;
250 };
251 break;
252
253 case ModulationType::MULTIPLY:
254 modulation_func = [](double logic_val, double buffer_val) {
255 return logic_val * buffer_val;
256 };
257 break;
258
259 case ModulationType::ADD:
260 modulation_func = [](double logic_val, double buffer_val) {
261 return logic_val + buffer_val;
262 };
263 break;
264
265 case ModulationType::INVERT_ON_TRUE:
266 modulation_func = [](double logic_val, double buffer_val) {
267 return logic_val > 0.5 ? -buffer_val : buffer_val;
268 };
269 break;
270
271 case ModulationType::HOLD_ON_FALSE:
272 if (min_size > 0) {
273 m_last_held_value = buffer_data[0];
274 }
275
276 modulation_func = [this](double logic_val, double buffer_val) mutable {
277 if (logic_val > 0.5) {
278 m_last_held_value = buffer_val;
279 return buffer_val;
280 }
281 return m_last_held_value;
282 };
283 break;
284
285 case ModulationType::ZERO_ON_FALSE:
286 modulation_func = [](double logic_val, double buffer_val) {
287 return logic_val > 0.5 ? buffer_val : 0.0;
288 };
289 break;
290
291 case ModulationType::CROSSFADE:
292 modulation_func = [](double logic_val, double buffer_val) {
293 return buffer_val * logic_val;
294 };
295 break;
296
297 case ModulationType::THRESHOLD_REMAP:
298 modulation_func = [this](double logic_val, double /*buffer_val*/) {
299 return logic_val > 0.5 ? m_high_value : m_low_value;
300 };
301 break;
302
303 case ModulationType::SAMPLE_AND_HOLD:
304 if (min_size > 0) {
305 m_last_held_value = buffer_data[0];
306 m_last_logic_value = m_logic_data[0];
307 }
308
309 modulation_func = [this, first_sample = true](double logic_val, double buffer_val) mutable {
310 if (first_sample) {
311 first_sample = false;
312 return buffer_val;
313 }
314
315 bool logic_changed = std::abs(logic_val - m_last_logic_value) > 0.01;
316 m_last_logic_value = logic_val;
317
318 if (logic_changed) {
319 m_last_held_value = buffer_val;
320 }
321 return m_last_held_value;
322 };
323 break;
324
325 case ModulationType::CUSTOM:
326 modulation_func = m_modulation_function;
327 break;
328
329 default:
330 modulation_func = [](double logic_val, double /*buffer_val*/) {
331 return logic_val;
332 };
333 }
334 }
335
336 for (size_t i = 0; i < min_size; ++i) {
337 buffer_data[i] = modulation_func(m_logic_data[i], buffer_data[i]);
338 }
339
340 return true;
341}
