34/**
35 * @class DynamicVideoStream
36 * @brief Video container that frames are appended to at runtime, the counterpart
37 * of DynamicSoundStream.
38 *
39 * Frames arrive through append_frame instead of being decoded from a file. In
40 * linear mode the stream grows with each frame. In circular mode
41 * (enable_circular_buffer) it keeps only the most recent frames.
42 *
43 * Frame indices are absolute. Frame i is stored in slot i % capacity and stays
44 * readable until a later frame overwrites it, so get_write_head keeps counting
45 * past the capacity and is_frame_available says whether an older frame is still
46 * there. Reading is unchanged: get_frame_pixels and FrameAccessProcessor work as
47 * for any VideoStreamContainer. A FrameAccessProcessor playhead that falls more
48 * than the capacity behind the head finds its frames overwritten.
49 *
50 * Pixels are stored as bytes whatever the format.
51 */
52class MAYAFLUX_API DynamicVideoStream : public VideoStreamContainer {
53public:
54 /**
55 * @brief Build a stream from a spec, enabling the ring when it asks for one.
56 * @param spec Frame size, format, rate and optional ring size. The rate is
57 * used by readers that auto advance; zero leaves it unset.
58 */
59 explicit DynamicVideoStream(const VideoStreamSpec& spec);
60
61 /**
62 * @brief Write one frame at the write head and advance it.
63 * @param pixels Exactly get_frame_byte_size() bytes.
64 * @return True if the frame was stored.
65 */
66 bool append_frame(std::span<const uint8_t> pixels);
67
68 /**
69 * @brief Write one image, for example one read back from a GPU image.
70 *
71 * Width, height and format must match the stream.
72 *
73 * @return True if the frame was stored.
74 */
75 bool append_frame(const ImageData& image);
76
77 /**
78 * @brief Download a GPU image straight into the next frame slot.
79 *
80 * Blocking, like TextureContainer::from_image. The image must match the
81 * stream's width, height and format. Pass a persistent staging buffer

83 * per-call Vulkan allocation.

85 * @param image Initialised source image.
