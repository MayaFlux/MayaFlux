79bool DynamicVideoStream::append_frame(std::span<const uint8_t> pixels)
80{
81 const size_t frame_bytes = get_frame_byte_size();
82
83 if (frame_bytes == 0 || pixels.size() != frame_bytes) {
84 MF_WARN(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
85 "DynamicVideoStream::append_frame expected {} bytes, got {}", frame_bytes, pixels.size());
86 return false;
87 }
88
89 uint8_t* slot = claim_frame_slot();
90 if (!slot) {
91 return false;
92 }
93

157std::shared_ptr<DynamicVideoStream> DynamicVideoStream::snapshot(uint64_t frames) const
158{
159 const uint64_t available = is_ring_mode()
160 ? std::min<uint64_t>(m_write_head, m_ring_capacity)
161 : m_write_head;
162 const uint64_t count = std::min(frames, available);
163
164 if (count == 0) {
165 return nullptr;
166 }
167
168 auto copy = std::make_shared<DynamicVideoStream>(VideoStreamSpec {
169 .width = m_width,
170 .height = m_height,
171 .format = m_format,
172 .frame_rate = m_frame_rate });
173 copy->ensure_capacity(count);
174
175 const std::vector<uint8_t> blank(get_frame_byte_size(), 0);
176
177 for (uint64_t index = m_write_head - count; index < m_write_head; ++index) {
178 const auto pixels = get_frame_pixels(index);
179 copy->append_frame(pixels.empty() ? std::span<const uint8_t>(blank) : pixels);
180 }
181
182 return copy;
183}
