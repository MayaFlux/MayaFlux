10/**
11 * @file VisionGpuDispatch.hpp
12 * @brief GPU execution layer for Kinesis::Vision::VisionSequence.
13 *
14 * VisionGpuExecutor::run() mirrors VisionExecutor::run() in contract: same
15 * input types, same VisionResult output. Internally it drives a
16 * VisionGpuContexts through the sequence via swap_shader() + stage_image()
17 * + dispatch_core() per step, keeping the working image GPU-resident
18 * between steps via OutputMode::IMAGE.
19 *
20 * VisionGpuExecutor::config() is the inspectable shader config table. It
21 * returns the GpuComputeConfig for any given VisionOp. Ops that are
22 * expressible via ShaderSpec are assembled at call time (no .comp file);
23 * ops requiring neighbourhood access or structured output name a .comp
24 * file. Ops with no GPU implementation return a config with INVALID_SHADER.
25 *
26 * The caller is responsible for only passing sequences composed of ops
27 * whose config is valid. run() logs an error and returns a default
28 * VisionResult on encountering INVALID_SHADER mid-sequence.
29 */
30
31/**
32 * @brief Cross-run state of the flow context.
33 *
34 * Two atlases are bound for their whole life. Which one holds the current
35 * frame is the curr parity, passed to shaders as a push constant, so ping-pong
36 * is a bit flip and never rewrites a descriptor. Detections, retained previous
37 * points and tracks live in the flow context's shared buffers and never leave
38 * the GPU except for the final tracks readback.
39 */
40struct FlowState {
41 std::shared_ptr<Core::VKImage> atlas[2];
42 Kinesis::Vision::PyramidLayout layout;
43
44 /**
45 * @brief Dense flow images, allocated on first use of OpticalFlowDense.
46 *
47 * flow_out holds the finest-level result at frame resolution, alternated
48 * by the same parity as the atlases so a result stays valid for one more
49 * run. The rest share the atlas layout: flow_lvl holds the coarser levels,
50 * dense_a and dense_b are the box filter scratch, and dense_tensor holds
51 * the previous frame's summed structure tensor and confidence.
52 */
53 std::shared_ptr<Core::VKImage> flow_out[2];
54 std::shared_ptr<Core::VKImage> flow_lvl;
55 std::shared_ptr<Core::VKImage> dense_a;
56 std::shared_ptr<Core::VKImage> dense_b;
57 std::shared_ptr<Core::VKImage> dense_tensor;
58 std::shared_ptr<Core::VKImage> flow_vis;
59 Kinesis::Vision::PyramidLayout dense_layout;
60
61 /**
62 * @brief Most recent flow image and track list produced from a frame that
63 * differed from the one before it.
64 *
65 * A repeated camera frame carries no motion, so a run on one republishes
66 * these instead of a field of zeros. Before any result exists, the fresh
67 * result is published as is.
68 */
69 std::shared_ptr<Core::VKImage> last_flow;
70 std::vector<Kinesis::Vision::TrackResult> last_tracks;
71
72 /**
73 * @brief Track count of the last distinct frame, independent of whether a
74 * host list was produced for it, and whether the submitted track
75 * sequence should read its tracks back to the host.
76 */
77 uint32_t last_track_count { 0 };
78 bool host_pending { true };
79
80 /**
81 * @brief Bookkeeping for the device resident track export.
82 *
83 * The two export buffers are shared buffers of the flow context and
84 * alternate per published frame, so a delivered buffer stays valid for one
85 * more publish. export_slot is the buffer the next publish writes,
86 * last_export the view of the most recent distinct frame, and
87 * export_pending marks a submitted track sequence whose export has not
88 * been delivered yet. export_view caches one handle per buffer, created on
89 * first delivery, so publishing allocates nothing per frame.
90 */
91 std::shared_ptr<Portal::Graphics::GpuBufferHandle> export_view[2];
92 std::shared_ptr<Portal::Graphics::GpuBufferHandle> last_export;
93 uint32_t export_slot { 0 };
94 bool export_pending { false };
95
96 uint32_t curr { 0 };
97 bool have_prev { false };
98 bool curr_ready { false };
99 bool buffers_ready { false };
100 Portal::Graphics::FenceID build_fence { Portal::Graphics::INVALID_FENCE };

242/**
243 * @class VisionGpuExecutor
244 * @brief Stateful GPU dispatch engine for VisionSequence execution.
245 *
246 * Owns a lazily-constructed VisionGpuContexts (m_contexts) plus any
247 * op-specific persistent GPU state that doesn't belong on a shared
248 * context (e.g. ConnectedComponents' ping-pong output image). Mirrors
249 * VisionExecutor's ownership model on the CPU side: construct one
250 * instance per independent pipeline, hold it, call run() every frame.
251 *
252 * Immovable: TextureExecutionContext owns GPU resource handles with no
253 * copy or move constructor, so VisionGpuContexts and therefore
254 * VisionGpuExecutor cannot be copied or moved either. Construct once,
255 * hold by reference, pointer, or shared_ptr, reuse indefinitely.
256 *
257 * Not safe to call run() concurrently from multiple threads on the same
258 * instance. Independent pipelines running concurrently should each own
259 * a separate VisionGpuExecutor instance.
