15size_t ImageKeyHash::operator()(const ImageKey& key) const noexcept
16{
17 uint64_t hash = 14695981039346656037ULL;
18 const auto mix = [&hash](uint64_t value) {
19 hash = (hash ^ value) * 1099511628211ULL;
20 };
21
22 mix(key.width);
23 mix(key.height);
24 mix(key.layers);
25 mix(key.mip_levels);
26 mix(static_cast<uint64_t>(key.format));
27 mix(static_cast<uint64_t>(key.kind));
28

434std::shared_ptr<Core::VKImage> TextureLoom::acquire_cached_image(
435 ImageCacheEntry& entry, const ImageKey& key)
436{
437 if (key.width == 0 || key.height == 0 || key.layers == 0
438 || key.mip_levels == 0)
439 return nullptr;
440
441 const auto usage = key.kind == ImageKey::Kind::STORAGE_2D
442 ? Core::VKImage::Usage::STORAGE
443 : Core::VKImage::Usage::TEXTURE_2D;
444 if (entry.image && entry.key == key && entry.image->is_initialized()
445 && entry.image->get_width() == key.width
446 && entry.image->get_height() == key.height
447 && entry.image->get_array_layers() == key.layers
448 && entry.image->get_mip_levels() == key.mip_levels
449 && entry.image->get_format() == to_vulkan_format(key.format)
450 && entry.image->get_usage() == usage) {
451 return entry.image;
452 }
453
454 std::shared_ptr<Core::VKImage> image;
455 switch (key.kind) {
456 case ImageKey::Kind::SAMPLED_2D:
457 if (key.layers != 1)
458 return nullptr;
459 image = create_2d(key.width, key.height, key.format, nullptr, key.mip_levels);
460 break;
461 case ImageKey::Kind::SAMPLED_ARRAY:
462 if (key.mip_levels != 1)
463 return nullptr;
464 image = create_2d_array(key.width, key.height, key.layers, key.format, nullptr);
465 break;
466 case ImageKey::Kind::STORAGE_2D:
467 if (key.layers != 1 || key.mip_levels != 1)
468 return nullptr;
469 image = create_storage_image(key.width, key.height, key.format);
470 break;
471 }
472
473 if (image) {
474 entry.key = key;
475 entry.image = image;
476 }
477 return image;
478}

480std::shared_ptr<Core::VKImage> TextureLoom::refresh_cached_image(
481 ImageCacheEntry& entry,
482 const ImageKey& key,
483 std::span<const uint8_t> pixels,
484 const std::shared_ptr<Buffers::VKBuffer>& staging)
485{
486 if (key.kind == ImageKey::Kind::STORAGE_2D || key.width == 0
487 || key.height == 0 || key.layers == 0)
488 return nullptr;
489
490 const size_t required = calculate_image_size(key.width, key.height, 1, key.format)
491 * key.layers;
492 if (pixels.size() != required) {
493 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
494 "Cached image upload has {} bytes, expected {}", pixels.size(), required);
495 return nullptr;
496 }
497
498 if (staging && (staging->get_size() < required || !staging->get_mapped_ptr())) {
499 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
500 "Cached image staging buffer is unavailable or too small");
501 return nullptr;
502 }
503
504 auto image = acquire_cached_image(entry, key);
505 if (!image)
506 return nullptr;
507
508 if (staging) {
509 upload_data(image, pixels.data(), pixels.size(), staging);
510 } else {
511 upload_data(image, pixels.data(), pixels.size());
512 }
513
514 return image;
515}

517std::shared_ptr<Core::VKImage> ImageCacheSet::acquire(
518 TextureLoom& loom, const ImageKey& key)
519{
520 if (m_last_entry && m_last_key == key)
521 return m_last_entry->cache.image;
522
523 auto found = m_entries.find(key);
524 if (found != m_entries.end()) {
525 m_last_key = key;
526 m_last_entry = &found->second;
527 return found->second.cache.image;
528 }
529
530 ImageCacheEntry cache;
531 auto image = loom.acquire_cached_image(cache, key);
532 if (!image)
533 return nullptr;
534
535 m_last_key.reset();
536 m_last_entry = nullptr;
537
538 auto [added, inserted] = m_entries.emplace(key, Entry { .cache = std::move(cache) });
539 if (!inserted)
540 return added->second.cache.image;
541
542 m_cached_bytes += image->get_size_bytes();
543 if (!m_budget_warned && m_cached_bytes > m_byte_budget) {
544 m_budget_warned = true;
545 MF_WARN(Journal::Component::Portal, Journal::Context::ImageProcessing,
546 "Image cache holds {} bytes, exceeding its {}-byte warning threshold",
547 m_cached_bytes, m_byte_budget);
548 }
549
550 m_last_key = key;
551 m_last_entry = &added->second;
552 return image;
553}

637std::optional<Kakshya::ImageData> TextureLoom::download_image(
638 const std::shared_ptr<Core::VKImage>& image,
639 const std::shared_ptr<Buffers::VKBuffer>& staging,
640 vk::ImageLayout restore_layout,
641 vk::PipelineStageFlags restore_stage)
642{
643 if (!is_initialized()) {
644 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
645 "download_image: TextureLoom is not initialised");
646 return std::nullopt;
647 }
648
649 const auto format = readable_format(image);
650 if (!format) {
651 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
652 "download_image: image is not readable as a single-layer 2D base level");
653 return std::nullopt;

657
658 auto result = Kakshya::ImageData::allocate(
659 image->get_width(), image->get_height(), get_channel_count(*format), *format);
660 if (!result || result->byte_size() != byte_count) {
661 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
662 "download_image: format {} has no typed pixel storage matching its size", static_cast<int>(*format));
663 return std::nullopt;
664 }
665
666 download_data(image, result->data(), byte_count, staging, false, restore_layout, restore_stage);
667
668 if (!result->is_consistent()) {
669 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
670 "download_image: downloaded pixels do not match format {}", static_cast<int>(*format));
671 return std::nullopt;
672 }
673

707 std::optional<std::pair<vk::Offset3D, vk::Offset3D>> region_offsets(
708 const vk::Rect2D& region, const Core::VKImage& image)
709 {
710 const int32_t left = region.offset.x;
711 const int32_t top = region.offset.y;
712
713 if (left < 0 || top < 0
714 || static_cast<uint32_t>(left) >= image.get_width()
715 || static_cast<uint32_t>(top) >= image.get_height()) {
716 return std::nullopt;
717 }
718
719 const auto x = static_cast<uint32_t>(left);
720 const auto y = static_cast<uint32_t>(top);
721 const uint32_t width = region.extent.width != 0 ? region.extent.width : image.get_width() - x;
722 const uint32_t height = region.extent.height != 0 ? region.extent.height : image.get_height() - y;
723
724 if (x + width > image.get_width() || y + height > image.get_height()) {
725 return std::nullopt;
726 }
727
728 return std::pair {
729 vk::Offset3D { left, top, 0 },
730 vk::Offset3D { static_cast<int32_t>(x + width), static_cast<int32_t>(y + height), 1 }
731 };
732 }

766bool TextureLoom::blit_layer(
767 const std::shared_ptr<Core::VKImage>& src,
768 const std::shared_ptr<Core::VKImage>& dst,
769 const LayerBlit& blit,
770 bool deferred)
771{
772 if (!is_initialized() || !src || !dst) {
773 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
774 "Invalid parameters for blit_layer");
775 return false;
776 }
777
778 if (blit.src_layer >= src->get_array_layers() || blit.dst_layer >= dst->get_array_layers()) {
779 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
780 "blit_layer: layer {} of {} or layer {} of {} does not exist",
781 blit.src_layer, src->get_array_layers(), blit.dst_layer, dst->get_array_layers());
782 return false;
783 }
784
785 constexpr auto color = vk::ImageAspectFlagBits::eColor;
786 if (src->get_aspect_flags() != color || dst->get_aspect_flags() != color) {
787 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
788 "blit_layer: only color images can be blitted");
789 return false;
790 }
791
792 const auto src_offsets = region_offsets(blit.src, *src);
793 auto dst_offsets = region_offsets(blit.dst, *dst);
794 if (!src_offsets || !dst_offsets) {
795 MF_ERROR(Journal::Component::Portal, Journal::Context::ImageProcessing,
796 "blit_layer: a rectangle lies outside its image");
797 return false;
798 }
799
800 if (blit.filter == FilterMode::CUBIC) {
801 MF_WARN(Journal::Component::Portal, Journal::Context::ImageProcessing,
802 "blit_layer: cubic filtering is not available, using linear");
803 }
804
805 if (blit.flip_x) {
806 std::swap(dst_offsets->first.x, dst_offsets->second.x);
807 }
808 if (blit.flip_y) {
809 std::swap(dst_offsets->first.y, dst_offsets->second.y);
810 }
811
812 const vk::ImageBlit region {
813 vk::ImageSubresourceLayers { color, 0, blit.src_layer, 1 },
814 { src_offsets->first, src_offsets->second },
815 vk::ImageSubresourceLayers { color, 0, blit.dst_layer, 1 },
816 { dst_offsets->first, dst_offsets->second }
817 };
818
819 return m_resource_manager->blit_image(src, dst, region, to_blit_filter(blit.filter), deferred);
820}

869size_t TextureLoom::hash_sampler_config(const SamplerConfig& config)
870{
871 size_t hash = 0;
872 hash ^= std::hash<int> {}(static_cast<int>(config.mag_filter)) << 0;
873 hash ^= std::hash<int> {}(static_cast<int>(config.min_filter)) << 4;
874 hash ^= std::hash<int> {}(static_cast<int>(config.address_mode_u)) << 8;
875 hash ^= std::hash<int> {}(static_cast<int>(config.address_mode_v)) << 12;
876 hash ^= std::hash<int> {}(static_cast<int>(config.address_mode_w)) << 16;
877 hash ^= std::hash<float> {}(config.max_anisotropy) << 20;
878 hash ^= std::hash<bool> {}(config.enable_mipmaps) << 24;

1045 if (static_cast<bool>(aspects & vk::ImageAspectFlagBits::eDepth)

1051 if (!format) {
1052 return std::nullopt;

1055 const size_t byte_count = calculate_image_size(image->get_width(), image->get_height(), 1, *format);
1056 if (byte_count == 0 || byte_count > image->get_size_bytes()) {
