84/**
85 * @brief Pixel rectangle of a normalised region over a w x h image, clamped
86 * to the image. No region is the whole image. A region that clamps to
87 * nothing gives an empty rectangle, which masks everything.
88 */
89inline PixelRect region_rect(
90 const std::optional<Kinesis::Vision::BoundingBox>& region, uint32_t w, uint32_t h)
91{
92 if (!region)
93 return { .x0 = 0U, .y0 = 0U, .x1 = w, .y1 = h };
94
95 const auto fw = static_cast<float>(w);
96 const auto fh = static_cast<float>(h);
97 const auto clamp_to = [](float v, uint32_t hi) {
98 return static_cast<uint32_t>(std::clamp(v, 0.0F, static_cast<float>(hi)));
99 };
100 return {
101 .x0 = clamp_to(std::floor(region->x * fw), w),
102 .y0 = clamp_to(std::floor(region->y * fh), h),
103 .x1 = clamp_to(std::ceil((region->x + region->w) * fw), w),
104 .y1 = clamp_to(std::ceil((region->y + region->h) * fh), h),
105 };
106}

330/**
331 * @brief Clamped record count from an exported buffer's vec4 header, or
332 * nullopt when the buffer isn't readable.
333 *
334 * Shared by every read_exported_* decode: header[0].x holds the count as
335 * uint bits, clamped to the buffer's actual capacity at stride_vec4 per
336 * record (one header vec4 assumed).
337 */
338inline std::optional<uint32_t> exported_record_count(
339 const std::shared_ptr<Portal::Graphics::GpuBufferHandle>& handle,
340 size_t stride_vec4)
341{
342 if (!handle || !handle->mapped_ptr || handle->size_bytes < sizeof(glm::vec4))
343 return std::nullopt;
344
345 const auto* header = static_cast<const glm::vec4*>(handle->mapped_ptr);
346 const size_t capacity = (handle->size_bytes / sizeof(glm::vec4) - 1U) / stride_vec4;
347 return static_cast<uint32_t>(std::min<size_t>(std::bit_cast<uint32_t>(header[0].x), capacity));
348}
