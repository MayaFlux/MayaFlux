59bool VolumeGridBuffer::allocate_field(
60 const std::string& name,
61 size_t stride_bytes,
62 bool double_buffered,
63 Kinesis::LatticeSemantics semantics)
64{
65 if (name.empty()) {
66 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
67 "VolumeGridBuffer: field declared with empty name, skipped");
68 return false;
69 }
70
71 if (stride_bytes == 0) {
72 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
73 "VolumeGridBuffer: field '{}' declares zero stride, skipped", name);
74 return false;
75 }
76
77 if (m_fields.contains(name)) {
78 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
79 "VolumeGridBuffer: duplicate field '{}', later declaration discarded", name);
80 return false;
81 }
82
83 auto buffer_service = Registry::BackendRegistry::instance()
84 .get_service<Registry::Service::BufferService>();
85
86 if (!buffer_service || !buffer_service->allocate_raw_buffer) {
87 error<std::runtime_error>(
88 Journal::Component::Buffers,
89 Journal::Context::Init,
90 std::source_location::current(),
91 "VolumeGridBuffer requires a valid buffer service");
92 }
93
94 const auto usage_flags = static_cast<uint32_t>(
95 static_cast<VkBufferUsageFlags>(
96 vk::BufferUsageFlagBits::eStorageBuffer
97 | vk::BufferUsageFlagBits::eTransferSrc
98 | vk::BufferUsageFlagBits::eTransferDst));
99
100 const auto memory_flags = static_cast<uint32_t>(
101 static_cast<VkMemoryPropertyFlags>(vk::MemoryPropertyFlagBits::eDeviceLocal));
102
103 auto& resources = get_buffer_resources();
104 const size_t field_bytes = static_cast<size_t>(get_cell_count()) * stride_bytes;
105 const uint32_t slot_count = double_buffered ? 2 : 1;
106
107 Field field {
108 .stride_bytes = stride_bytes,
109 .slot_a = static_cast<uint32_t>(resources.back_buffers.size()),
110 .slot_b = 0,
111 .read_is_a = true,
112 .semantics = semantics,
113 };
114
115 for (uint32_t i = 0; i < slot_count; ++i) {
116 void* out_buffer = nullptr;
117 void* out_memory = nullptr;
118 void* out_mapped = nullptr;
119
120 buffer_service->allocate_raw_buffer(
121 field_bytes, usage_flags, memory_flags, false,
122 out_buffer, out_memory, out_mapped);
123
124 VKBufferResources::GenerationSlot slot;
125 slot.buffer = static_cast<vk::Buffer>(static_cast<VkBuffer>(out_buffer));
126 slot.memory = static_cast<vk::DeviceMemory>(static_cast<VkDeviceMemory>(out_memory));
127 slot.mapped_ptr = out_mapped;
128
129 resources.back_buffers.push_back(slot);
130 }
131
132 field.slot_b = double_buffered ? field.slot_a + 1 : field.slot_a;
133
134 m_fields.emplace(name, field);
135 m_field_order.push_back(name);
136
137 MF_DEBUG(Journal::Component::Buffers, Journal::Context::Init,
138 "VolumeGridBuffer: field '{}', stride {}, {} slot(s), {} bytes, class {}",
139 name, stride_bytes, slot_count, field_bytes * slot_count,
140 Reflect::enum_to_string(semantics.value_class));
141
142 return true;
143}

299VolumeGridBuffer::FlowStages VolumeGridBuffer::setup_flow(const FlowConfig& config)
300{
301 FlowStages stages;
302
303 auto self = std::dynamic_pointer_cast<VolumeGridBuffer>(shared_from_this());
304 auto chain = get_processing_chain();
305
306 if (!chain) {
307 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
308 "setup_flow: no processing chain, call after registration");
309 return stages;
310 }
311
312 const auto require = [this](const std::string& name, const char* role) {
313 if (name.empty()) {
314 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
315 "setup_flow: no field supplied for '{}'", role);
316 return false;
317 }
318 if (!has_field(name)) {
319 MF_ERROR(Journal::Component::Buffers, Journal::Context::Init,
320 "setup_flow: '{}' names no field on this volume, supplied as '{}'",
321 name, role);
322 return false;
323 }
324 return true;
325 };
326
327 if (!require(config.velocity, "velocity")
328 || !require(config.divergence, "divergence")
329 || !require(config.pressure, "pressure")) {
330 return stages;
331 }
332
333 if (config.viscosity > 0.0F && !require(config.scratch, "scratch")) {
334 return stages;
335 }
336
337 stages.self_advect = std::make_shared<AdvectProcessor>(
338 config.velocity, config.velocity, "volume_advect_vector.comp.spv");
339 stages.self_advect->set_time_step(config.time_step);
340 chain->add_processor(stages.self_advect, self);
341
342 if (config.buoyancy) {
343 const auto& b = *config.buoyancy;
344
345 if (!require(b.temperature, "buoyancy.temperature")
346 || !require(b.density, "buoyancy.density")) {
347 return stages;
348 }
349
350 stages.buoyancy = std::make_shared<BuoyancyProcessor>(
351 b.temperature, b.density, config.velocity,
352 b.direction, "volume_buoyancy.comp.spv");
353 stages.buoyancy->set_time_step(config.time_step);
354 stages.buoyancy->set_temperature_gain(b.temperature_gain);
355 stages.buoyancy->set_density_gain(b.density_gain);
356 stages.buoyancy->set_ambient(b.ambient);
357 chain->add_processor(stages.buoyancy, self);
358 }
359
360 if (config.viscosity > 0.0F) {
361 stages.diffuse = std::make_shared<DiffuseProcessor>(
362 config.velocity, config.scratch, "volume_diffuse_vector.comp.spv");
363 stages.diffuse->set_rate(config.viscosity);
364 stages.diffuse->set_time_step(config.time_step);
365 chain->add_processor(stages.diffuse, self);
366 }
367
368 if (config.walls) {
369 stages.wall_advected = std::make_shared<WallProcessor>(
370 config.velocity, "volume_wall.comp.spv");
371 chain->add_processor(stages.wall_advected, self);
372 }
373
374 stages.divergence = std::make_shared<DivergenceProcessor>(
375 config.velocity, config.divergence, "volume_divergence.comp.spv");
376 chain->add_processor(stages.divergence, self);
377
378 stages.pressure = std::make_shared<PressureProcessor>(
379 config.divergence, config.pressure, "volume_pressure_jacobi.comp.spv");
380 stages.pressure->set_iteration_count(config.jacobi_iterations);
381 chain->add_processor(stages.pressure, self);
382
383 stages.solenoidal = std::make_shared<SolenoidalProcessor>(
384 config.pressure, config.velocity, "volume_solenoidal.comp.spv");
385 chain->add_processor(stages.solenoidal, self);
386
387 if (config.walls) {
388 stages.wall_projected = std::make_shared<WallProcessor>(
389 config.velocity, "volume_wall.comp.spv");
390 chain->add_processor(stages.wall_projected, self);
391 }
392
393 for (const auto& carried : config.carried) {
394 if (!require(carried.field, "carried")) {
395 continue;
396 }
397
398 auto advect = std::make_shared<AdvectProcessor>(
399 config.velocity, carried.field, "volume_advect_scalar.comp.spv");
400 advect->set_time_step(config.time_step);
401 advect->set_dissipation(carried.dissipation);
402 chain->add_processor(advect, self);
403
404 stages.carriers.push_back(std::move(advect));
405 }
406
407 MF_INFO(Journal::Component::Buffers, Journal::Context::Init,
408 "VolumeGridBuffer::setup_flow: {} carried, buoyancy {}, viscosity {}, walls {}",
409 stages.carriers.size(), config.buoyancy ? "on" : "off",
410 config.viscosity, config.walls ? "on" : "off");
411
412 return stages;
413}

502void VolumeGridBuffer::seed(const std::string& name, const Kinesis::SpatialField& field)
503{
504 const auto* decl = find_field(name, "seed");
505 if (!decl) {
506 return;
507 }
508
509 if (decl->stride_bytes != sizeof(float)) {
510 MF_ERROR(Journal::Component::Buffers, Journal::Context::BufferManagement,
511 "VolumeGridBuffer::seed: field '{}' has stride {}, SpatialField requires {}",
512 name, decl->stride_bytes, sizeof(float));
513 return;
514 }
515
516 const glm::uvec3 res = m_lattice.resolution;
517 std::vector<float> values(m_lattice.cell_count());
518
519 size_t i = 0;
520 for (uint32_t z = 0; z < res.z; ++z) {
521 for (uint32_t y = 0; y < res.y; ++y) {
522 for (uint32_t x = 0; x < res.x; ++x) {
523 values[i++] = field(m_lattice.cell_center({ x, y, z }));
524 }
525 }
526 }
527
528 seed_raw(name, values.data(), values.size() * sizeof(float));
529}

679 constexpr size_t k_alignment = 16;
680
681 auto& resources = get_buffer_resources();
682 std::vector<size_t> indices;
683 std::vector<size_t> offsets;
684 size_t total = 0;
685
686 for (size_t i = 0; i < names.size(); ++i) {
687 const auto* field = find_field(names[i], "read_fields");
688 if (!field) {
689 continue;
690 }
691
692 const size_t expected = static_cast<size_t>(get_cell_count()) * field->stride_bytes;
693 if (dsts[i].second != expected) {
694 MF_ERROR(Journal::Component::Buffers, Journal::Context::BufferManagement,
695 "VolumeGridBuffer::read_fields: size {} does not match expected {} for '{}'",
696 dsts[i].second, expected, names[i]);
697 continue;
698 }
699
700 indices.push_back(i);
701 offsets.push_back(total);
702 total += (dsts[i].second + k_alignment - 1) & ~(k_alignment - 1);
703 }
704
705 if (indices.empty()) {
706 return;
707 }
708
709 resolve_transfer(m_pending_transfer);
710
711 if (!m_transfer_staging || m_transfer_staging->get_size_bytes() < total) {
712 m_transfer_staging = create_staging_buffer(total);
713 }
714
715 std::vector<TransferHandle> handles;
716 handles.reserve(indices.size());
717
718 for (size_t k = 0; k < indices.size(); ++k) {
719 const auto* field = find_field(names[indices[k]], "read_fields");
720 const uint32_t slot = field->read_is_a ? field->slot_a : field->slot_b;
721 handles.push_back(download_back_buffer_async(
722 resources.back_buffers[slot], dsts[indices[k]].second,
723 m_transfer_staging, offsets[k]));
724 }
725
726 for (size_t k = 0; k < indices.size(); ++k) {
727 const auto* field = find_field(names[indices[k]], "read_fields");
728 const uint32_t slot = field->read_is_a ? field->slot_a : field->slot_b;
729 resolve_back_buffer_read(
730 handles[k], resources.back_buffers[slot],
731 dsts[indices[k]].first, dsts[indices[k]].second,
732 m_transfer_staging, offsets[k]);
733 }
734}

736size_t VolumeGridBuffer::surface_storage_bytes(const std::optional<SurfaceConfig>& surface)
737{
738 if (!surface) {
739 return 1;
740 }
741
742 const glm::uvec3 res = glm::max(surface->resolution, glm::uvec3(1U));
743 const uint64_t voxels = static_cast<uint64_t>(res.x) * res.y * res.z;
744 return static_cast<size_t>(voxels * 15U * sizeof(Kakshya::MeshVertex));
745}
