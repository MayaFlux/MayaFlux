171 const ICompoundProperty& arb, const std::string& stream_name)
172 {

198 /**
199 * @brief Identify an archive by its leading bytes without involving
200 * Alembic.
201 *
202 * Ogawa files begin with the ASCII "Ogawa", HDF5 files with the HDF5
203 * signature. Checking first means a missing, unreadable or foreign file
204 * is rejected here with one clear message, instead of Alembic probing
205 * each core in turn and printing its own diagnostics for the ones that
206 * fail.
207 *
208 * @return The format, or nullopt if the file cannot be read or is neither.
209 */
210 std::optional<ArchiveFormat> sniff_format(const std::string& path)
211 {
212 constexpr std::string_view k_ogawa { "Ogawa" };
213 constexpr std::string_view k_hdf5 { "\x89HDF\r\n\x1a\n", 8 };
214
215 std::ifstream file(path, std::ios::binary);
216 if (!file) {
217 return std::nullopt;
218 }
219
220 std::array<char, 8> head {};
221 file.read(head.data(), static_cast<std::streamsize>(head.size()));
222 const auto count = static_cast<size_t>(file.gcount());
223
224 if (count >= k_ogawa.size() && std::string_view(head.data(), k_ogawa.size()) == k_ogawa) {
225 return ArchiveFormat::Ogawa;
226 }
227 if (count >= k_hdf5.size() && std::string_view(head.data(), k_hdf5.size()) == k_hdf5) {
228 return ArchiveFormat::HDF5;
229 }
230 return std::nullopt;
231 }

255void SpatialReader::Impl::index_streams()
256{
257 names.clear();
258 records.clear();
259
260 const IObject top = archive.getTop();
261
262 for (size_t i = 0; i < top.getNumChildren(); ++i) {
263 const auto& header = top.getChildHeader(i);
264 const std::string& name = header.getName();
265
266 StreamRecord record;
267 record.metadata = user_metadata(header.getMetaData());
268
269 ICompoundProperty arb;
270
271 if (IPoints::matches(header)) {
272 IPoints points(top, name);
273 auto& schema = points.getSchema();
274 record.topology = PrimitiveTopology::POINT_LIST;
275 record.sample_count = schema.getNumSamples();
276 record.time_sampling = schema.getTimeSampling();
277 arb = schema.getArbGeomParams();
278 record.object = std::move(points);
279 } else if (ICurves::matches(header)) {
280 ICurves curves(top, name);
281 auto& schema = curves.getSchema();
282 record.topology = PrimitiveTopology::LINE_STRIP;
283 record.sample_count = schema.getNumSamples();
284 record.time_sampling = schema.getTimeSampling();
285 arb = schema.getArbGeomParams();
286 record.object = std::move(curves);
287 } else {
288 MF_WARN(Journal::Component::IO, Journal::Context::FileIO,
289 "SpatialReader: object '{}' is neither points nor curves, skipped", name);
290 continue;
291 }
292
293 record.attributes = declare_attributes(arb, name);
294
295 names.push_back(name);
296 records.emplace(name, std::move(record));
297 }
298}

309bool SpatialReader::open(const std::string& filepath)
310{
311 close();
312
313 const auto resolved = FileReader::resolve_path(filepath);
314
315 std::error_code ec;
316 if (!std::filesystem::is_regular_file(resolved, ec)) {
317 set_error("open: no such file: " + resolved);
318 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
319 "SpatialReader::open: '{}' does not exist", resolved);
320 return false;
321 }
322
323 const auto format = sniff_format(resolved);
324 if (!format) {
325 set_error("open: not an Alembic archive: " + resolved);
326 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
327 "SpatialReader::open: '{}' is not an Alembic archive", resolved);
328 return false;
329 }
330
331 try {
332 IArchive archive = *format == ArchiveFormat::Ogawa
333 ? IArchive(Alembic::AbcCoreOgawa::ReadArchive(), resolved)
334 : Alembic::AbcCoreFactory::IFactory().getArchive(resolved);
335 if (!archive.valid()) {
336 set_error("open: archive could not be read: " + resolved);
337 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
338 "SpatialReader::open: '{}' could not be read as an archive", resolved);
339 return false;
340 }
341
342 m_impl->archive = archive;
343 m_impl->index_streams();
344 } catch (const std::exception& e) {
345 close();
346 set_error(std::string("open: ") + e.what());
347 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
348 "SpatialReader::open: failed for '{}': {}", resolved, e.what());
349 return false;
350 }
351
352 MF_INFO(Journal::Component::IO, Journal::Context::FileIO,
353 "SpatialReader: opened '{}', {} stream(s)", resolved, m_impl->names.size());
354 return true;
355}

399std::optional<SpatialSample> SpatialReader::read_sample(
400 const std::string& stream_name, size_t index) const
401{
402 if (!m_impl->archive.valid()) {
403 set_error("read_sample: no archive open");
404 return std::nullopt;
405 }
406
407 const auto it = m_impl->records.find(stream_name);
408 if (it == m_impl->records.end()) {
409 set_error("read_sample: no stream named '" + stream_name + "'");
410 return std::nullopt;
411 }
412
413 auto& record = it->second;
414 if (index >= record.sample_count) {
415 set_error("read_sample: index " + std::to_string(index) + " out of range for '"
416 + stream_name + "' (" + std::to_string(record.sample_count) + " samples)");
417 return std::nullopt;
418 }
419
420 try {
421 const ISampleSelector selector(static_cast<index_t>(index));
422 SampleStorage fresh;
423 ICompoundProperty arb;
424
425 if (auto* points = std::get_if<IPoints>(&record.object)) {
426 auto& schema = points->getSchema();
427 IPointsSchema::Sample sample;
428 schema.get(sample, selector);
429
430 fresh.positions = copy_array<glm::vec3>(sample.getPositions());
431 fresh.ids = copy_array<uint64_t>(sample.getIds());
432 fresh.velocities = copy_array<glm::vec3>(sample.getVelocities());
433 arb = schema.getArbGeomParams();
434 } else if (auto* curves = std::get_if<ICurves>(&record.object)) {
435 auto& schema = curves->getSchema();
436 ICurvesSchema::Sample sample;
437 schema.get(sample, selector);
438
439 fresh.positions = copy_array<glm::vec3>(sample.getPositions());
440 fresh.vertex_counts_per_curve = copy_array<int32_t>(sample.getCurvesNumVertices());
441 arb = schema.getArbGeomParams();
442
443 if (!record.warned_curve_type
444 && (sample.getType() != kLinear || sample.getWrap() != kNonPeriodic)) {
445 record.warned_curve_type = true;
446 MF_WARN(Journal::Component::IO, Journal::Context::FileIO,
447 "SpatialReader: '{}' is not linear and non periodic, its control points are "
448 "returned as plain polylines",
449 stream_name);
450 }
451 }
452
453 if (arb.valid()) {
454 for (const auto& decl : record.attributes) {
455 auto attribute = read_attribute(arb, decl, selector);
456 if (!attribute) {
457 MF_WARN(Journal::Component::IO, Journal::Context::FileIO,
458 "SpatialReader: '{}' sample {}: attribute '{}' has an unsupported scope or "
459 "no values, skipped",
460 stream_name, index, decl.name);
461 continue;
462 }
463 fresh.attributes.push_back(std::move(*attribute));
464 }
465 }
466
467 record.storage = std::move(fresh);
468 return SpatialSample {
469 .topology = record.topology,
470 .positions = record.storage.positions,
471 .vertex_counts_per_curve = record.storage.vertex_counts_per_curve,
472 .ids = record.storage.ids,
473 .velocities = record.storage.velocities,
474 .attributes = record.storage.attributes,
475 };
476 } catch (const std::exception& e) {
477 set_error(std::string("read_sample: ") + e.what());
478 MF_ERROR(Journal::Component::IO, Journal::Context::FileIO,
479 "SpatialReader::read_sample: '{}' sample {}: {}", stream_name, index, e.what());
480 return std::nullopt;
481 }
482}
