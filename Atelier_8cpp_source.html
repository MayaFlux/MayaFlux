116 m_buffer_manager->add_buffer(buf, Buffers::ProcessingToken::GRAPHICS_BACKEND);

118 if (!additional_textures.empty()) {
119 buf->setup_rendering({

130 }

179 }
180

187 k_rect_capacity,

189 (void)Plot::place_rect(surface, std::move(buf), swatch, relate_to);
