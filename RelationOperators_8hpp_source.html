13/**
14 * @class Source
15 * @brief Anything that can supply a number when evaluated
16 *
17 * The right-hand side of a mapping wherever an operator parameter should be
18 * drivable by whatever is at hand:
19 * - A constant returns itself.
20 * - A node returns its get_last_output().
21 * - A NodeNetwork plus an index returns that network's
22 * get_node_output(index), or 0.0 if it has nothing there. Works the same
23 * whether the network is this RelationNetwork or any other NodeNetwork.
24 * - A callable is invoked on whichever thread evaluates the source, so it
25 * follows the same rules as a node hook on the audio thread.
26 */
27class Source {
28public:
29 using Function = std::function<double()>;
30
31 Source(double constant)
32 : m_value(constant)
33 {
34 }
35
36 template <std::derived_from<Node> NodeType>
37 Source(std::shared_ptr<NodeType> node)
38 : m_value(std::shared_ptr<Node>(std::move(node)))
39 {
40 }
41
42 /**
43 * @brief One indexed output of a NodeNetwork
44 * @param network Read through get_node_output(index) on every evaluate()
45 * @param index Position within network, the same index
46 * NodeNetwork::map_parameter's ONE_TO_ONE mode and
47 * apply_one_to_one() read
48 */
49 template <std::derived_from<NodeNetwork> NetworkType>
50 Source(std::shared_ptr<NetworkType> network, size_t index)
51 : m_value(NetworkIndex { std::shared_ptr<NodeNetwork>(std::move(network)), index })
52 {
53 }
54
55 template <typename Callable>
56 requires(!std::same_as<std::remove_cvref_t<Callable>, Source>
57 && std::is_invocable_r_v<double, Callable&>)
58 Source(Callable&& callable)
59 : m_value(Function(std::forward<Callable>(callable)))
60 {
61 }
62
63 /**
64 * @brief Current value
65 * @return The constant, the node's last output, the network index's
66 * output, or the callable's result. Zero for an empty node, a
67 * network index the network has no value for, or an empty
68 * callable.
69 */
70 [[nodiscard]] double evaluate() const
71 {
72 if (const auto* constant = std::get_if<double>(&m_value)) {
73 return *constant;
74 }
75 if (const auto* node = std::get_if<std::shared_ptr<Node>>(&m_value)) {
76 return *node ? (*node)->get_last_output() : 0.0;
77 }
78 if (const auto* index = std::get_if<NetworkIndex>(&m_value)) {
79 return index->network ? index->network->get_node_output(index->index).value_or(0.0) : 0.0;
80 }
81 const auto& function = std::get<Function>(m_value);
82 return function ? function() : 0.0;
83 }
84
85private:
86 struct NetworkIndex {
87 std::shared_ptr<NodeNetwork> network;
88 size_t index;
89 };
90
91 std::variant<double, std::shared_ptr<Node>, NetworkIndex, Function> m_value;
92};

94/**
95 * @class RelationOperator
96 * @brief Operator that acts on a network's slots over one block at a time
97 *
98 * The network supplies the current slot storage through set_slots(), called
99 * once when the operator is added and again at the start of every
100 * process_batch(). An operator never caches a slot address past the current
101 * cycle, since the network's slot storage can grow (and reallocate) between
102 * cycles.
103 *
104 * A concrete operator has no generic initialize()/reset() lifecycle: it
105 * sizes any internal state from its own constructor arguments.
106 *
107 * Once per block the host calls begin_block(), which applies mapped
108 * parameters, then process(frames).
109 *
110 * An operator has no clock of its own; process_batch() is its only cadence.
111 * Anything finer (per-sample or event-driven behavior) is composed from
112 * outside: node hooks, tasks, or direct calls to the operator's own public
113 * methods. An operator that acts only when called leaves process() empty.
114 * Operators communicate through slots: level and activity writes, and
115 * rendered blocks.
116 *
117 * An operator that publishes signals overrides publishes_output(),
118 * get_output_count(), and get_output_block().
119 */
120class MAYAFLUX_API RelationOperator : public NetworkOperator {
121public:
122 /**
123 * @brief Supply the current slot storage
124 * @param slots RelationNetwork's slot storage, valid only for the caller's
125 * current cycle
126 * @note Called by RelationNetwork::add() and at the start of every
127 * process_batch(). An operator is not usable before the first call.
128 */
129 void set_slots(std::vector<RelationSlot>& slots) noexcept { m_slots = &slots; }
130
131 /**
132 * @brief Apply mapped parameters at the start of a block
133 */
134 void begin_block()
135 {
136 for (const auto& binding : m_bindings) {
137 set_parameter(binding.parameter, binding.source.evaluate());
138 }
139 }
140
141 /**
142 * @brief Drive an operator parameter from a source
143 * @param parameter Name passed to set_parameter
144 * @param source Evaluated once per block, at begin_block()
145 *
146 * A parameter has at most one mapping. Mapping it again replaces the
147 * source. Must not run concurrently with begin_block().
148 */
149 RelationOperator& map(std::string parameter, Source source)
150 {
151 auto it = std::ranges::find(m_bindings, parameter, &Binding::parameter);
152 if (it != m_bindings.end()) {
153 it->source = std::move(source);
154 } else {
155 m_bindings.push_back({ std::move(parameter), std::move(source) });
156 }
157 return *this;
158 }
159
160 /**
161 * @brief Remove the mapping of a parameter. No-op when absent.
162 * @note Must not run concurrently with begin_block().
163 */
164 void unmap(std::string_view parameter)
165 {
166 std::erase_if(m_bindings, [&](const Binding& b) { return b.parameter == parameter; });
167 }
168
169 [[nodiscard]] virtual bool publishes_output() const { return false; }
170 [[nodiscard]] virtual size_t get_output_count() const { return 0; }
171 [[nodiscard]] virtual std::span<const double> get_output_block(size_t /*index*/) const { return {}; }
172
173protected:
174 [[nodiscard]] std::span<RelationSlot> slots() const noexcept
175 {
176 return m_slots ? std::span<RelationSlot>(*m_slots) : std::span<RelationSlot> {};
177 }
178
179private:
180 struct Binding {
181 std::string parameter;
182 Source source;
183 };
184
185 std::vector<Binding> m_bindings;
186 std::vector<RelationSlot>* m_slots { nullptr };
187};
