151bool galois_lfsr(double input, std::span<double> coefs) noexcept
152{
153 if (coefs.size() < 4)
154 return false;
155
156 uint32_t reg = word_of(coefs[3]);
157 if (rises(input > coefs[0], coefs[1])) {
158 if (reg == 0)
159 reg = 1;
160 const bool fell_out = (reg & 1U) != 0U;
161 reg >>= 1U;
162 if (fell_out)

182bool state_machine(double input, std::span<double> coefs) noexcept
183{
184 if (coefs.size() < 3)
185 return false;
186
187 const size_t states = index_of(coefs[1]);
188 if (states == 0 || coefs.size() < 3 + 4 * states)
189 return false;
190
191 size_t current = index_of(coefs[2]);
192 if (current >= states)
193 current = 0;
194
195 const size_t row = 3 + 4 * current + (input > coefs[0] ? 2 : 0);
196 const size_t next = index_of(coefs[row]);
197 coefs[2] = static_cast<double>(next < states ? next : 0);
198 return coefs[row + 1] > 0.5;
199}

350{
351 if (inputs.size() < 2 || coefs.size() < 4)
352 return false;
353
354 const size_t length = index_of(coefs[2]);
355 if (length == 0 || coefs.size() < 4 + length)
356 return false;
357
358 const auto stages = coefs.subspan(4, length);
359 if (rises(inputs[0] > coefs[0], coefs[1])) {
360 for (size_t i = length - 1; i > 0; --i)
361 stages[i] = stages[i - 1];
362 stages[0] = inputs[1] > coefs[0] ? 1.0 : 0.0;
363 }
