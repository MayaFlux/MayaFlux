77/**
78 * @class IOManager
79 * @brief Optional orchestration layer for IO reader lifetime and IOService dispatch.
80 *
81 * MayaFlux IO readers are fully self-sufficient — VideoFileReader, SoundFileReader,
82 * and ImageReader can all be used standalone without this class. IOManager exists
83 * to solve problems that only arise at scale:
84 *
85 * - Globally unique reader_ids across concurrent VideoFileReader instances, so
86 * IOService::request_decode routes to the correct decode thread.
87 * - A single IOService registration backed by a keyed dispatch map, rather than
88 * each reader overwriting the previous registration.
89 * - Shared ownership of VideoFileReader instances whose background decode thread
90 * must outlive the scope in which they were created.
91 * - High-level load/hook functions that collapse the full open→load→processor
92 * setup→buffer wiring sequence into single calls, mirroring what Depot does
93 * for audio and eliminating the boilerplate currently required for video.
94 *
95 * VideoFileReader detects the IOService registration generically via BackendRegistry
96 * and skips its own registration — no IOManager-specific knowledge in the reader.
97 *
98 * Levels of integration (all valid):
99 * 1. VideoFileReader standalone, single stream — self-registers, self-ids.
100 * 2. IOManager constructed directly — multi-stream safe, no Engine required.
101 * 3. Engine constructs IOManager in Init() — fully orchestrated, zero user wiring.
102 */
103class MAYAFLUX_API IOManager {
104public:
105 /**
106 * @brief Construct IOManager and register the IOService into BackendRegistry.
107 *
108 * Must be constructed before any VideoFileReader::load_into_container() call
109 * that should participate in managed dispatch.
110 */
111 IOManager(Core::GlobalStreamInfo& stream_info, uint32_t frame_rate, const std::shared_ptr<Buffers::BufferManager>& buffer_manager, const std::shared_ptr<Vruta::TaskScheduler>& scheduler);
112
113 /**
114 * @brief Unregisters IOService, releases all owned readers, clears stored buffers.
115 *
116 * Also drains any still-running spatial captures through
117 * stop_spatial_capture() rather than leaving them to plain unique_ptr
118 * destruction, the same way active audio captures are drained above:
119 * an Ogawa archive left open is not a valid file, only a growing one,
120 * so a capture the caller never explicitly stopped must still be
121 * finalized here.
122 */
123 ~IOManager();
124
125 IOManager(const IOManager&) = delete;
126 IOManager& operator=(const IOManager&) = delete;
127 IOManager(IOManager&&) = delete;
