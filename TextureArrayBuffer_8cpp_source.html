67bool TextureArrayBuffer::submit_layer(uint32_t layer, std::span<const uint8_t> pixels)
68{
69 if (layer >= m_layers || pixels.size() != m_layer_bytes) {
70 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
71 "TextureArrayBuffer::submit_layer needs a layer below {} and {} bytes, got layer {} and {} bytes",
72 m_layers, m_layer_bytes, layer, pixels.size());
73 return false;
74 }
75
76 if (m_layer_mode) {
77 if (!m_layer_staging) {
78 m_layer_staging = create_image_staging_buffer(m_layer_bytes);
79 }
80 return Portal::Graphics::get_texture_manager().upload_layer(
81 get_texture(), layer, pixels.data(), pixels.size(), m_layer_staging);
82 }
83
84 auto& storage = get_pixel_data_mutable();
85 if (storage.size() != m_layer_bytes * m_layers) {
86 storage.assign(m_layer_bytes * m_layers, 0);
87 }
88
89 std::memcpy(storage.data() + static_cast<size_t>(layer) * m_layer_bytes, pixels.data(), m_layer_bytes);
90 mark_pixels_dirty();
91 return true;
92}

94bool TextureArrayBuffer::submit_layer(uint32_t layer, const Kakshya::ImageData& image)
95{
96 if (layer >= m_layers || image.byte_size() == 0 || !image.is_consistent()) {
97 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
98 "TextureArrayBuffer::submit_layer needs a layer below {} and a consistent image, got layer {}",
99 m_layers, layer);
100 return false;
101 }
102
103 const bool same_extent = image.width == get_width() && image.height == get_height();
104 if (same_extent && image.format == get_format()) {
105 return submit_layer(layer,
106 std::span<const uint8_t>(static_cast<const uint8_t*>(image.data()), image.byte_size()));
107 }
108
109 if (!same_extent && !fit_for(layer)) {
110 return false;
111 }
112
113 if (!m_source_staging || m_source_staging->get_size() < image.byte_size()) {
114 m_source_staging = create_image_staging_buffer(image.byte_size());
115 }
116
117 const Portal::Graphics::ImageKey key {
118 .width = image.width,
119 .height = image.height,
120 .format = image.format
121 };
122
123 const auto source = Portal::Graphics::get_texture_manager().refresh_cached_image(
124 m_source_cache, key,
125 std::span<const uint8_t>(static_cast<const uint8_t*>(image.data()), image.byte_size()),
126 m_source_staging);
127 if (!source) {
128 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
129 "TextureArrayBuffer::submit_layer failed to upload a {}x{} image", image.width, image.height);
130 return false;

136bool TextureArrayBuffer::submit_layer(uint32_t layer, const std::shared_ptr<Core::VKImage>& image)
137{
138 if (layer >= m_layers || !image) {
139 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
140 "TextureArrayBuffer::submit_layer needs a layer below {} and an image, got layer {}",
141 m_layers, layer);
142 return false;
143 }
144
145 const bool same_extent = image->get_width() == get_width() && image->get_height() == get_height();
146 const auto fit = fit_for(layer);
147 if (!same_extent && !fit) {
148 return false;
149 }
150
151 enter_layer_mode();
152 if (!m_layer_mode) {
153 return false;
154 }
155
156 auto plan = Portal::Graphics::SamplerForge::plan_fit(
157 image->get_width(), image->get_height(), get_width(), get_height(),
158 same_extent ? Portal::Graphics::FitMode::STRETCH : *fit, m_filter);
159
160 auto& loom = Portal::Graphics::get_texture_manager();
161 bool stored = !plan.blits.empty();
162
163 if (stored) {
164 const vk::Rect2D& first = plan.blits.front().dst;
165 const bool covers = plan.blits.size() == 1
166 && first.offset.x == 0 && first.offset.y == 0
167 && first.extent.width == get_width() && first.extent.height == get_height();
168 if (!covers) {
169 clear_layer(layer);
170 }
171 }
172
173 for (auto& blit : plan.blits) {
174 blit.dst_layer = layer;
175 stored = loom.blit_layer(image, get_texture(), blit) && stored;
176 }
177
178 return stored;
179}

314void TextureArrayBuffer::enter_layer_mode()
315{
316 if (m_layer_mode) {
317 return;
318 }
319
320 ensure_array();
321 if (!has_texture()) {
322 return;
323 }
324
325 auto& storage = get_pixel_data_mutable();
326 const size_t total = m_layer_bytes * m_layers;
327 const auto staging = create_image_staging_buffer(total);
328 if (!staging) {
329 MF_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
330 "TextureArrayBuffer: failed to allocate staging for {} bytes", total);
331 return;
332 }
333
334 auto& loom = Portal::Graphics::get_texture_manager();
335 if (storage.size() == total) {
336 loom.upload_data(get_texture(), storage.data(), total, staging);
337 } else {
338 const std::vector<uint8_t> blank(total, 0);
339 loom.upload_data(get_texture(), blank.data(), total, staging);
340 }
341
342 std::vector<uint8_t>().swap(storage);
343 m_layer_mode = true;
