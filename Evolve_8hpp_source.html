16/**
17 * @concept GridCellData
18 * @brief Cell types a relaxation grid can be seeded with from a host callable.
19 *
20 * Arithmetic scalars, two component vectors and four component vectors whose
21 * size is a multiple of four bytes, so sizeof matches the std430 array stride
22 * the rule and emit shaders read. Three component vectors are excluded:
23 * glm::vec3 is 12 bytes but strides 16 in an SSBO array, so a rule with such a
24 * cell declares a four component format instead.
25 */
26template <typename State>
27concept GridCellData = (ArithmeticData<State> || GlmVec2Type<State> || GlmVec4Type<State>)
28 && (sizeof(State) % sizeof(uint32_t) == 0);

30/**
31 * @class Evolve
32 * @brief Sets a system going from a config: state that persists, a rule that
33 * advances it every cycle, and rendering attached.
34 *
35 * Owned by Creator as a member. Each call performs the whole sequence its
36 * arguments imply and returns the real object it built or was handed, so
37 * nothing is left to attach afterwards. The returned object carries its own
38 * processors and advances on its own each cycle. The call sets it going and
39 * does not step it.
40 *
41 * An evolving system is not a type here. It is a config handed to a class,
42 * and the sequence that class needs is what this performs. The name says
43 * nothing about a medium: it covers whatever the config's class advances,
44 * which today is three families, selected by the config they consume:
45 * - RelaxationGridBuffer::GridConfig builds a grid, seeds it, registers it in
46 * the graphics domain and attaches rendering.
47 * - VolumeGridBuffer::FlowConfig runs the incompressible flow stages over a
48 * volume. Required refs left unnamed are declared under default names, and
49 * any named ref the volume does not yet hold is declared.
50 * - Nodes::Network::SpatialFieldConfig drives a network's vertices through a
51 * GpuFieldOperator, reading its vertex layout from the network's own
52 * primary operator.
53 *
54 * All three route to the graphics domain and every call requires
55 * RenderConfig::target_window. There is no default window. A null target logs
56 * an error and returns nullptr, and nothing is registered. Exceptions thrown
57 * by the classes' constructors propagate.
58 *
59 * Nothing here is realtime safe. Call from the thread that constructs the
60 * objects.
61 *
62 * @code
63 * auto life = vega.evolve(
64 * Buffers::RelaxationGridBuffer::GridConfig {
65 * .width = 256,
66 * .height = 256,
67 * .rule = { .state = Kakshya::GpuDataFormat::UINT32, .shader = "relax_conway_rule.comp" },
68 * .emit = { .state = Kakshya::GpuDataFormat::UINT32, .shader = "relax_vertex_emit.comp" },
69 * },
70 * [](uint32_t x, uint32_t y) { return (x ^ y) % 7U == 0U ? 1U : 0U; },
71 * { .target_window = window });
72 *
73 * auto volume = std::make_shared<Buffers::VolumeGridBuffer>(
74 * lattice,
75 * Buffers::VolumeGridBuffer::SurfaceConfig { "density", { 48, 48, 48 }, 0.5F });
76 * auto density = volume->declare_scalar("density");
77 * auto heat = volume->declare_scalar("heat");
78 * vega.evolve(
79 * volume,
80 * { .carried = { { density, 0.99F } },
81 * .buoyancy = Buffers::VolumeGridBuffer::FlowConfig::Buoyancy { .temperature = heat, .density = density } },
82 * { .target_window = window });
83 *
84 * auto particles = vega.ParticleNetwork(50000, glm::vec3(-1.0F), glm::vec3(1.0F), distribution);
85 * vega.evolve(
86 * particles,
87 * { .absorb_radius = 0.05F },
88 * [](const auto& field) { field->bind(Nodes::Network::FieldTarget::POSITION, swirl); },
89 * { .target_window = window });
90 * @endcode
91 */

138 /**
139 * @brief Build a relaxation grid seeded from a host callable.
140 * @param config Grid arrangement.
141 * @param seed Callable taking cell coordinates (x, y) and returning the
142 * cell's state. x runs over [0, width) and y over [0, height).
143 * @param render Render target. target_window is required.
144 * @return The registered grid, or nullptr if the target window is null or
145 * the seed's return type does not match the grid's cell stride.
146 *
147 * The seed's return type is the cell type, constrained by GridCellData and
148 * checked against the stride the config resolved. The state is written
149 * before registration, so the first generation steps from it.
150 */
151 template <typename Seed>
152 requires std::invocable<Seed&, uint32_t, uint32_t>
153 && GridCellData<std::remove_cvref_t<std::invoke_result_t<Seed&, uint32_t, uint32_t>>>
154 auto operator()(
155 const Buffers::RelaxationGridBuffer::GridConfig& config,
156 Seed&& seed,
157 const RenderConfig& render)
158 -> std::shared_ptr<Buffers::RelaxationGridBuffer>
159 {
160 using State = std::remove_cvref_t<std::invoke_result_t<Seed&, uint32_t, uint32_t>>;
161
162 std::vector<std::byte> cells(static_cast<size_t>(config.width) * config.height * sizeof(State));
163 for (uint32_t y = 0; y < config.height; ++y) {
164 for (uint32_t x = 0; x < config.width; ++x) {
165 const State value = seed(x, y);
166 std::memcpy(
167 cells.data() + (static_cast<size_t>(y) * config.width + x) * sizeof(State),
168 &value,
169 sizeof(State));
170 }
171 }
172
173 auto grid = build_grid(config, render, cells, sizeof(State));
174 if (grid) {
175 MF_LIVE_EXPOSE_AUTO(grid);
176 }
177 return grid;
178 }
