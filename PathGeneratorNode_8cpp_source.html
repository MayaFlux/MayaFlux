16 SegmentRange calculate_affected_segment_range(
17 size_t control_idx,
18 size_t total_controls,
19 Kinesis::InterpolationMode mode,
20 Eigen::Index samples_per_segment)
21 {
22 SegmentRange range {};
23
24 switch (mode) {
25 case Kinesis::InterpolationMode::CATMULL_ROM:
26 case Kinesis::InterpolationMode::BSPLINE:
27 range.start_control_idx = (control_idx > 0) ? control_idx - 1 : 0;
28 range.end_control_idx = std::min(control_idx + 2, total_controls - 1);
29 break;
30
31 case Kinesis::InterpolationMode::CUBIC_BEZIER:
32 case Kinesis::InterpolationMode::CUBIC_HERMITE:
33 range.start_control_idx = (control_idx / 4) * 4;
34 range.end_control_idx = std::min(range.start_control_idx + 3, total_controls - 1);
35 break;
36
37 case Kinesis::InterpolationMode::QUADRATIC_BEZIER:
38 range.start_control_idx = (control_idx / 3) * 3;
39 range.end_control_idx = std::min(range.start_control_idx + 2, total_controls - 1);
40 break;
41
42 default:
43 range.start_control_idx = control_idx;
44 range.end_control_idx = control_idx;
45 break;
46 }
47
48 range.start_vertex_idx = range.start_control_idx * samples_per_segment;
49 range.end_vertex_idx = (range.end_control_idx + 1) * samples_per_segment;
50
51 return range;
52 }

125void PathGeneratorNode::write_segment_attributes(
126 const std::vector<LineVertex>& curve_verts,
127 size_t start_idx,
128 LineVertex* dst) const
129{
130 if (m_samples_per_segment < 2) {
131 return;
132 }
133
134 const auto span = static_cast<size_t>(m_samples_per_segment - 1);
135
136 for (size_t i = 0; i < span; ++i) {
137 const size_t c0 = start_idx + std::min((i * 3) / span, size_t(3));
138 const size_t c1 = start_idx + std::min(((i + 1) * 3) / span, size_t(3));
139
140 LineVertex& v0 = dst[i * 2];
141 LineVertex& v1 = dst[i * 2 + 1];
142
143 v0.color = m_force_uniform_color ? m_current_color : curve_verts[c0].color;
144 v1.color = m_force_uniform_color ? m_current_color : curve_verts[c1].color;
145
146 v0.thickness = m_force_uniform_thickness ? m_current_thickness : curve_verts[c0].thickness;
147 v1.thickness = m_force_uniform_thickness ? m_current_thickness : curve_verts[c1].thickness;
148 }

151void PathGeneratorNode::write_curve_segment(
152 const std::vector<LineVertex>& curve_verts,
153 size_t start_idx,
154 LineVertex* dst)
155{
156 if (start_idx + 3 >= curve_verts.size()) {
157 return;
158 }
159
160 m_evaluator.configure(m_mode, m_tension);
161
162 for (size_t i = 0; i < 4; ++i) {
163 const auto& pt = curve_verts[start_idx + i].position;
164 m_segment_controls[i * 3 + 0] = pt.x;
165 m_segment_controls[i * 3 + 1] = pt.y;
166 m_segment_controls[i * 3 + 2] = pt.z;
167 }
168
169 m_evaluator.evaluate_planar(m_segment_controls, 3, m_samples_per_segment, m_curve_primary);
170
171 const std::vector<double>* curve = &m_curve_primary;
172
173 if (m_arc_length_parameterization) {
174 m_evaluator.reparameterize_planar(m_curve_primary, 3,
175 m_samples_per_segment, m_samples_per_segment, m_curve_secondary);
176 curve = &m_curve_secondary;
177 }
178
179 const auto count = static_cast<size_t>(m_samples_per_segment);
180 if (count < 2) {
181 return;
182 }
183
184 const double* x = curve->data();
185 const double* y = x + count;
186 const double* z = y + count;
187
188 for (size_t i = 0; i < count - 1; ++i) {
189 dst[i * 2].position = {
190 static_cast<float>(x[i]), static_cast<float>(y[i]), static_cast<float>(z[i])
191 };
192 dst[i * 2 + 1].position = {
193 static_cast<float>(x[i + 1]), static_cast<float>(y[i + 1]), static_cast<float>(z[i + 1])
194 };
195 }
196
197 write_segment_attributes(curve_verts, start_idx, dst);
198}

201{

516{
517 const size_t num_points = m_control_points.size();
518
519 if (start_ctrl_idx >= num_points || end_ctrl_idx >= num_points) {
520 MF_ERROR(Journal::Component::Nodes, Journal::Context::NodeProcessing,
521 "Invalid segment range [{}, {}] for {} control points",
522 start_ctrl_idx, end_ctrl_idx, num_points);
523 return;

525
526 m_range_cache.clear();
527 m_range_cache.reserve(end_ctrl_idx - start_ctrl_idx + 1);
528 for (size_t i = start_ctrl_idx; i <= end_ctrl_idx; ++i) {
529 m_range_cache.push_back(m_control_points[i]);
530 }
531

550void PathGeneratorNode::compute_frame()
551{
552 if (m_control_points.empty() && m_draw_vertices.empty() && m_completed_draws.empty()) {
553 resize_vertex_buffer(0);
554 return;
555 }
556
557 if (m_geometry_dirty) {
558 regenerate_geometry();
559 m_attributes_dirty = false;
560 } else if (m_attributes_dirty) {
561 refresh_attributes();

564 }
565
566 if (!m_vertex_data_dirty) {
567 return;
568 }

573 m_combined_cache.insert(m_combined_cache.end(), m_completed_draws.begin(), m_completed_draws.end());
574 m_combined_cache.insert(m_combined_cache.end(), m_draw_vertices.begin(), m_draw_vertices.end());

577 resize_vertex_buffer(0);

599void PathGeneratorNode::complete()
600{
601 if (m_draw_window.size() < 4) {
602 MF_WARN(Journal::Component::Nodes, Journal::Context::NodeProcessing,
603 "Not enough points in draw window to generate curve segment ({} points)",
604 m_draw_window.size());
605 m_completed_draws.insert(
606 m_completed_draws.end(),
607 m_draw_vertices.begin(),
608 m_draw_vertices.end());
609 m_draw_vertices.clear();
610 m_draw_window.clear();
611 m_vertex_data_dirty = true;
612 return;
613 }
614
615 const size_t per_window = vertices_per_window();
616 const size_t windows = m_draw_window.size() - 3;
617 const size_t base = m_completed_draws.size();
618
619 m_completed_draws.resize(base + windows * per_window);
620
621 for (size_t w = 0; w < windows; ++w) {
622 write_curve_segment(m_draw_window, w, m_completed_draws.data() + base + w * per_window);
623 }
624
625 for (size_t i = windows + 1; i < m_draw_window.size(); ++i) {
626 append_line_segment(m_draw_window[i - 1], m_draw_window[i], m_completed_draws);
627 }
628
629 m_draw_vertices.clear();
630 m_draw_window.clear();
631 m_vertex_data_dirty = true;
632}
