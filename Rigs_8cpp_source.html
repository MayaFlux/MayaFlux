70std::vector<std::shared_ptr<Kriya::SamplingPipeline>> create_samplers(
71 const std::string& filepath, uint32_t num_samples, bool truncate,
72 uint64_t max_dur_ms, uint32_t max_channels)

75
76 if (!stream) {
77 MF_ERROR(Journal::Component::API, Journal::Context::FileIO,
78 "create_samplers: failed to load '{}'", filepath);
79 return {};
80 }
81
82 const uint32_t ch_count = (max_channels == 0)
83 ? stream->get_num_channels()
84 : std::min(max_channels, stream->get_num_channels());
85
86 std::vector<std::shared_ptr<Kriya::SamplingPipeline>> result;
87 result.reserve(ch_count);
88
89 for (uint32_t i = 0; i < ch_count; ++i)
90 result.push_back(create_sampler_from_stream(stream, i, max_dur_ms));
91
92 return result;

147std::shared_ptr<Kakshya::DynamicSoundStream> create_ring(double seconds, uint32_t channels)
148{
149 const uint32_t rate = Config::get_sample_rate();
150 const auto capacity = static_cast<uint64_t>(seconds * static_cast<double>(rate));
151
152 if (capacity == 0 || channels == 0) {
153 MF_ERROR(Journal::Component::API, Journal::Context::Configuration,
154 "create_ring: {} seconds on {} channels is empty", seconds, channels);
155 return nullptr;
156 }
157
158 auto ring = std::make_shared<Kakshya::DynamicSoundStream>(rate, channels);
159 ring->enable_circular_buffer(capacity);
160 return ring;
