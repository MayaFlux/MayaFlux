19 /**
20 * @brief Format a stage declares, or the one its ShaderSpec binding carries.
21 * @param stage Stage to resolve.
22 * @param binding Binding whose format a spec stage is read from.
23 * @return The format, or nullopt when a path stage declares none.
24 */
25 std::optional<Kakshya::GpuDataFormat> resolve_state(
26 const RelaxationGridBuffer::GridConfig::Stage& stage,
27 std::string_view binding)
28 {
29 if (stage.state) {
30 return stage.state;
31 }
32
33 const auto* spec = std::get_if<Portal::Graphics::ShaderSpec>(&stage.shader);
34 if (!spec) {
35 return std::nullopt;
36 }
37
38 const auto it = std::ranges::find_if(
39 spec->bindings,
40 [binding](const auto& slot) { return slot.name == binding; });
41
42 if (it == spec->bindings.end()) {
43 return std::nullopt;
44 }
45
46 return it->format;
47 }

49 /**
50 * @brief Cell stride in bytes for a GridConfig, after checking its stages.
51 * @param config Grid arrangement.
52 * @return gpu_data_format_bytes of the shared cell format.
53 */
54 size_t resolve_stride(const RelaxationGridBuffer::GridConfig& config)
55 {
56 const auto rule = resolve_state(config.rule, "state_in");
57 const auto emit = resolve_state(config.emit, "cell_state");
58
59 if (!rule || !emit) {
60 error<std::invalid_argument>(
61 Journal::Component::Buffers, Journal::Context::Init,
62 std::source_location::current(),
63 "GridConfig: a stage whose shader is a path must declare its state format");
64 }
65
66 if (*rule != *emit) {
67 error<std::invalid_argument>(
68 Journal::Component::Buffers, Journal::Context::Init,
69 std::source_location::current(),
70 "GridConfig: rule and emit disagree on the cell format");
71 }
72
73 if (*rule == Kakshya::GpuDataFormat::VEC3_F32 || *rule == Kakshya::GpuDataFormat::VEC3_F64) {
74 error<std::invalid_argument>(
75 Journal::Component::Buffers, Journal::Context::Init,
76 std::source_location::current(),
77 "GridConfig: a three component cell strides 16 bytes in std430, declare a four component format");
78 }
79
80 return Kakshya::gpu_data_format_bytes(*rule);
81 }

83 /**
84 * @brief Write constants after a stage's fixed prefix in its push constant block.
85 * @param processor Stage processor, attached and not yet dispatched.
86 * @param offset Byte offset of the first constant.
87 * @param constants Words to write, in shader declaration order.
88 *
89 * Grows the block through set_push_constant_size so the pipeline range is
90 * created at the final width on the first cycle.
91 */
92 template <typename Processor>
93 void apply_constants(
94 const std::shared_ptr<Processor>& processor,
95 size_t offset,
96 const std::vector<RelaxationGridBuffer::GridConfig::Constant>& constants)
97 {
98 if (constants.empty()) {
99 return;
100 }
101
102 const size_t end = offset + constants.size() * sizeof(uint32_t);
103 if (processor->get_push_constant_data().size() < end) {
104 processor->set_push_constant_size(end);
105 }
106
107 auto& data = processor->get_push_constant_data();
108 if (data.size() < end) {
109 data.resize(end);
110 }
111
112 size_t at = offset;
113 for (const auto& constant : constants) {
114 std::visit([&](auto value) { std::memcpy(data.data() + at, &value, sizeof(value)); }, constant);
115 at += sizeof(uint32_t);
116 }
117 }

187void RelaxationGridBuffer::setup_processors(ProcessingToken token)
188{
189 auto layout = Kakshya::VertexLayout::for_raw(sizeof(Kakshya::Vertex));
190 layout.vertex_count = get_cell_count();
191 set_vertex_layout(layout);
192
193 m_step_processor = std::visit(
194 [](const auto& src) { return std::make_shared<RelaxationStepProcessor>(src); },
195 m_rule_source);
196
197 m_step_processor->set_processing_token(token);
198 set_default_processor(m_step_processor);
199 apply_constants(m_step_processor, sizeof(RelaxationStepProcessor::GridExtent), m_rule_constants);
200
201 auto chain = get_processing_chain();
202 if (!chain) {
203 chain = std::make_shared<BufferProcessingChain>();
204 set_processing_chain(chain);
205 }
206 chain->set_preferred_token(token);
207
208 m_emit_processor = std::visit(
209 [](const auto& src) { return std::make_shared<RelaxationEmitProcessor>(src); },
210 m_emit_source);
211 m_emit_processor->set_processing_token(token);
212 chain->add_processor(m_emit_processor, shared_from_this());
213 apply_constants(m_emit_processor, sizeof(RelaxationEmitProcessor::EmitParams), m_emit_constants);
214 m_emit_processor->set_extent(m_extent);
215 m_emit_processor->set_point_size(m_point_size);
216
217 MF_DEBUG(Journal::Component::Buffers, Journal::Context::Init,
218 "RelaxationGridBuffer setup_processors: step + emit attached");
219}
