13/**
14 * @class Emitter
15 * @brief Object that acts on existing MayaFlux objects when committed.
16 *
17 * Constructed with an optional influence function. Position is optional: call
18 * @c set_position before registering with @c Fabric if spatial indexing is
19 * required. Entities without a position are committed normally but are not
20 * inserted into the spatial index.
21 *
22 * The id is assigned by @c Fabric::wire and is stable for the object's
23 * lifetime. It is zero until the object has been registered.
24 *
25 * RenderConfig takes a designated initializer.
26 *
27 * @code
28 * emitter->set_influence<InfluenceBlock>(InfluenceBlock::from, bind_influence_block(proc));
29 * emitter->render(mgr, { .target_window = window });
30 * @endcode

231 /**
232 * @brief Set the one target of an influence of any type.
233 * @tparam T Type the producer returns; whatever the target takes.
234 * @param producer Builds the value from the influence context.
235 * @param target Takes the value and decides what it affects.
236 *
237 * An Emitter has a single target: setting another replaces it. Runs on
238 * every invoke after the influence function, independent of the render
239 * processor target. Not part of encoded state.
240 */
241 template <typename T>
242 void set_influence(typename Influence<T>::Producer producer, typename Influence<T>::Target target)
243 {
244 auto influence = std::make_shared<Influence<T>>(std::move(producer));
245 influence->add_target(std::move(target));
246 m_influence = [influence](const InfluenceContext& ctx) { influence->invoke(ctx); };
247 }
248
249 /** @brief Remove the target set with set_influence(). */
250 void clear_influence() { m_influence = nullptr; }
251
252 /**
253 * @brief Invoke the influence function with the supplied context.
254 * @param ctx Populated context for this commit.
255 */
256 void invoke(const InfluenceContext& ctx) const
257 {
258 if (m_fn) {
259 m_fn(ctx);
260 }
261 dispatch_audio_sinks(m_audio_sinks, ctx);
262 dispatch_render_sinks(m_render_sinks, ctx);
263 if (m_attachment)
264 apply_attachment(*m_attachment, ctx);
265 if (m_influence) {
266 m_influence(ctx);
267 }
268 }
