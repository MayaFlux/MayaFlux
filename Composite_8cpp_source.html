63void CompositeLayout::finalize()
64{
65 if (m_finalized)
66 return;
67
68 if (m_fields.empty()) {
69 Journal::error<std::invalid_argument>(
70 Journal::Component::Kakshya,
71 Journal::Context::Runtime,
72 std::source_location::current(),
73 "CompositeLayout requires at least one field");
74 }
75
76 m_presence_bytes = m_fields.size() / 8 + (m_fields.size() % 8 != 0);
77 size_t offset = m_presence_bytes;
78
79 for (auto& field : m_fields) {
80 if (field.size_bytes > std::numeric_limits<size_t>::max() - offset) {
81 Journal::error<std::length_error>(
82 Journal::Component::Kakshya,
83 Journal::Context::Runtime,
84 std::source_location::current(),
85 "CompositeLayout exceeds addressable size at field '{}'", field.name);
86 }
87 field.offset_bytes = offset;
88 offset += field.size_bytes;
89 }
90
91 m_stride_bytes = offset;
92 m_finalized = true;
93}

215 std::memcpy(destination, &cell, sizeof(cell));

332std::optional<std::string_view> Composite::text(std::string_view field_name) const noexcept
333{
334 const auto field_index = m_layout->find_field(field_name);
335 if (!field_index || !has(field_name))
336 return std::nullopt;
337
338 const auto& field = m_layout->fields()[*field_index];
339 if (field.type != typeid(std::string) && field.type != typeid(detail::CompositeDynamic))
340 return std::nullopt;
341
342 const auto& rows = std::get<std::vector<uint8_t>>(*m_rows);
343 const auto* source = rows.data() + m_index * m_layout->stride_bytes() + field.offset_bytes;
344 TextSlice slice;
345 if (field.type == typeid(detail::CompositeDynamic)) {
346 DynamicCell cell;
347 std::memcpy(&cell, source, sizeof(cell));
348 if (cell.kind != text_kind)
349 return std::nullopt;
350 slice = { cell.payload, cell.length };
351 } else {
352 std::memcpy(&slice, source, sizeof(slice));
353 }
354
355 const auto& text = std::get<std::vector<uint8_t>>(*m_text);
356 if (slice.offset > text.size() || slice.length > text.size() - slice.offset)
357 return std::nullopt;
358
359 if (slice.length == 0)
360 return std::string_view {};
361
362 return std::string_view(
363 reinterpret_cast<const char*>(text.data() + slice.offset),
364 static_cast<size_t>(slice.length));
365}

396std::optional<CompositeValue> Composite::value(std::string_view field_name) const noexcept
397{
398 const auto field_index = m_layout->find_field(field_name);
399 if (!field_index || !has(field_name)
400 || m_layout->fields()[*field_index].type != typeid(detail::CompositeDynamic))
401 return std::nullopt;
402
403 const auto& rows = std::get<std::vector<uint8_t>>(*m_rows);
404 DynamicCell cell;
405 std::memcpy(&cell, rows.data() + m_index * m_layout->stride_bytes()
406 + m_layout->fields()[*field_index].offset_bytes, sizeof(cell));
407 if (cell.kind == null_kind)
408 return CompositeValue { std::monostate {} };
409 if (cell.kind == integer_kind) {
410 int64_t result;
411 std::memcpy(&result, &cell.payload, sizeof(result));
412 return CompositeValue { result };
413 }
414 if (cell.kind == real_kind) {
415 double result;
416 std::memcpy(&result, &cell.payload, sizeof(result));
417 return CompositeValue { result };
418 }
419 if (cell.kind == text_kind) {
420 if (auto result = text(field_name))
421 return CompositeValue { *result };
422 }
423 if (cell.kind == blob_kind) {
424 if (auto result = blob(field_name))
425 return CompositeValue { *result };
426 }
427 return std::nullopt;
