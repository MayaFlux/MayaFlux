49void CompositeLayout::finalize()
50{
51 if (m_finalized)
52 return;
53
54 if (m_fields.empty()) {
55 Journal::error<std::invalid_argument>(
56 Journal::Component::Kakshya,
57 Journal::Context::Runtime,
58 std::source_location::current(),
59 "CompositeLayout requires at least one field");
60 }
61
62 m_presence_bytes = m_fields.size() / 8 + (m_fields.size() % 8 != 0);
63 size_t offset = m_presence_bytes;
64
65 for (auto& field : m_fields) {
66 if (field.size_bytes > std::numeric_limits<size_t>::max() - offset) {
67 Journal::error<std::length_error>(
68 Journal::Component::Kakshya,
69 Journal::Context::Runtime,
70 std::source_location::current(),
71 "CompositeLayout exceeds addressable size at field '{}'", field.name);
72 }
73 field.offset_bytes = offset;
74 offset += field.size_bytes;
75 }
76
77 m_stride_bytes = offset;
78 m_finalized = true;
79}

232std::optional<std::string_view> Composite::text(std::string_view field_name) const noexcept
233{
234 const auto field_index = m_layout->find_field(field_name);
235 if (!field_index || !has(field_name))
236 return std::nullopt;
237
238 const auto& field = m_layout->fields()[*field_index];
239 if (field.type != typeid(std::string))
240 return std::nullopt;
241
242 const auto& rows = std::get<std::vector<uint8_t>>(*m_rows);
243 TextSlice slice;
244 std::memcpy(&slice,
245 rows.data() + m_index * m_layout->stride_bytes() + field.offset_bytes,
246 sizeof(slice));
247
248 const auto& text = std::get<std::vector<uint8_t>>(*m_text);
249 if (slice.offset > text.size() || slice.length > text.size() - slice.offset)
250 return std::nullopt;
251
252 if (slice.length == 0)
253 return std::string_view {};
254
255 return std::string_view(
256 reinterpret_cast<const char*>(text.data() + slice.offset),
257 static_cast<size_t>(slice.length));
