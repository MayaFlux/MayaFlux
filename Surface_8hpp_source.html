12/**
13 * @struct SurfaceConfig
14 * @brief The single argument every Surface-creating function in this module
15 * takes, at every level of abstraction.
16 *
17 * Surface, Atelier::create_surface, and Portal::Forma::create_surface all
18 * take this instead of a loose (window, name) pair: when @c layer/@c ctx are
19 * unset, a fresh Layer and Context are built from @c window/@c name against
20 * the global EventManager and filled in; when both are already set (the
21 * power-tinkerer case) they are used as-is and @c name is ignored. This type
22 * must not itself reach into internal::atelier(). Only Forma.hpp/.cpp touch
23 * that singleton. Building stays the job of whichever function is given
24 * the config.
25 */
26struct SurfaceConfig {
27 /// @brief Target window. Must outlive the Surface.
28 std::shared_ptr<Core::Window> window;
29
30 /**
31 * @brief Unique name scoping the Context's event coroutines.
32 *
33 * Only consulted when @c layer/@c ctx are unset and need to be built.
34 * Ignored when both are already supplied.
35 */
36 std::string name;
37
38 /// @brief Pre-built Layer, constructed against @c window. Optional,
39 /// built automatically from @c name when unset.
40 std::shared_ptr<Layer> layer;
41
42 /// @brief Pre-built Context, already wired to @c window. Optional,
43 /// built automatically from @c name when unset.
44 std::shared_ptr<Context> ctx;
45
46 /**
47 * @brief Whether the built-in close handling (Bridge::stop_sync plus a
48 * deferred window release, see Surface::Surface) runs. Forced
49 * true whenever on_close is set. See should_detach_on_close().
50 */
51 bool detach_on_close { true };
52
53 /**
54 * @brief Runs in addition to, never instead of, the built-in close
55 * handling. Setting this forces should_detach_on_close() true
56 * regardless of detach_on_close, so supplying a hook can never
57 * silently lose the window-release cleanup.
58 *
59 * Context::on_close is a single slot per id, and the last write
60 * wins, so a bare replacement field here would silently clobber
61 * Surface's own close handler on first use. This field exists so
62 * that cannot happen.
63 */
64 std::function<void()> on_close;
65
66 /// @brief Whether the built-in detach behavior actually runs.
67 [[nodiscard]] bool should_detach_on_close() const noexcept
68 {
69 return detach_on_close || static_cast<bool>(on_close);
70 }
71};

73/**
74 * @class Surface
75 * @brief Named owner of a (Window, Layer, Context) triple: the Forma canvas.
76 *
77 * In Forma, three things always travel together: a Window (the rendering
78 * target and coordinate space), a Layer (the spatial registry of elements),
79 * and a Context (the event router that hit-tests against the layer). They
80 * are the three faces of one concept: the canvas. Surface names that
81 * concept and owns the triple, so that downstream construction functions
82 * accept one argument instead of three.
83 *
84 * Surface is not a widget toolkit, not a layout engine, not an event
85 * dispatcher. It does not know what an element looks like, how a buffer
86 * is built, or where bindings are routed. Construction of FormaBuffers,
87 * Mapped<T>s, and Bridge registrations is the job of the free functions
88 * in Forma.hpp, which take a Surface& as their canvas argument and read
89 * the BufferManager, Bridge, and other module-level state from the
90 * singletons established by Portal::Forma::initialize().
91 *
92 * The Layer and Context owned by Surface remain fully accessible through
93 * the layer() and ctx() accessors. Surface is a named owner, not a wall.
94 * Anything that worked against Layer or Context before continues to work
95 * against surface.layer() and surface.ctx().
96 *
97 * Surface(SurfaceConfig) never creates the Layer or Context itself. It only
98 * ever wraps ones already built. Portal::Forma::create_surface (in
99 * Forma.hpp) also takes a SurfaceConfig: the default path leaves layer/ctx
100 * unset, so a fresh Layer and Context are built against the global
101 * EventManager from window/name and the Surface is constructed from those.
102 * Set layer/ctx yourself in the same SurfaceConfig for the power-tinkerer
103 * path, when you need a custom Context subclass, want to share one Layer
104 * across multiple Contexts (split-pane editing), or are constructing in a
105 * test against a non-global EventManager.
106 *
107 * @code
108 * // Default path
109 * auto surface = Portal::Forma::create_surface(
110 * SurfaceConfig { .window = window, .name = "plot_live" });
111 *
112 * auto el = Portal::Forma::create_element<float>(surface, geom, 0.5F);
113 * surface.ctx().on_press(el.element.id, IO::MouseButtons::Left, ...);
114 *
115 * // Power-tinkerer path: pre-built layer and context
116 * auto layer = std::make_shared<Layer>();
117 * auto ctx = std::make_shared<MyCustomContext>(layer, window, em, "custom");
118 * Surface surface(SurfaceConfig { .window = window, .layer = layer, .ctx = ctx });
119 * @endcode
120 */
