7/**
8 * @class FunctionalOperation
9 * @brief Wraps a single callable as a ComputeOperation, for a wild or
10 * one-off idea that does not warrant a hand-written class.
11 *
12 * Satisfies ComputeOperation's four pure virtuals with the smallest
13 * possible surface: operation_function delegates to the supplied
14 * function; set_parameter/get_parameter store into a plain map (no
15 * type-specific defaults, since a generic wrapper has none to offer);
16 * get_operation_type always returns OperationType::CUSTOM, since a
17 * callable supplied at construction time carries no inherent verb
18 * category. Fully interchangeable with any hand-written
19 * ComputeOperation<InputType, OutputType> subclass everywhere
20 * ComputeMatrix, GPU backend attachment, and dependency graphs are
21 * concerned: none of that machinery keys off get_operation_type().
22 *
23 * @tparam InputType ComputeData type accepted.
24 * @tparam OutputType ComputeData type produced. Defaults to InputType.
25 */
26template <ComputeData InputType = std::vector<Kakshya::DataVariant>,
27 ComputeData OutputType = InputType>
28class FunctionalOperation : public ComputeOperation<InputType, OutputType> {
29public:
30 using input_type = Datum<InputType>;
31 using output_type = Datum<OutputType>;
32 using Fn = std::function<output_type(const input_type&)>;
33
34 explicit FunctionalOperation(Fn fn)
35 : m_fn(std::move(fn))
36 {
37 }
38
39 /**
40 * @brief Construct GPU-only: dispatches entirely via gpu_backend, no
41 * callable, no CPU fallback. Matches GpuAnalyzer's contract.
42 * @param gpu_backend Configured executor. Must not be null.
43 */
44 explicit FunctionalOperation(std::shared_ptr<GpuExecutionContext<InputType, OutputType>> gpu_backend)
