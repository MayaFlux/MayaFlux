91Kinesis::Vision::TrackObjectsAnalysis VisionAnalyzer::reduce_tracks(
92 const std::vector<Kinesis::Vision::TrackResult>& tracks, uint32_t w, uint32_t h)
93{
94 Kinesis::Vision::TrackObjectsAnalysis result;
95 result.tracks = tracks;
96
97 if (tracks.empty())
98 return result;
99
100 std::vector<float> track_data;
101 track_data.reserve(tracks.size() * 4);
102 std::vector<float> extra_data;
103 extra_data.reserve(tracks.size() * 2);
104 for (const auto& t : tracks) {
105 track_data.push_back(t.position.x);
106 track_data.push_back(t.position.y);
107 track_data.push_back(t.previous.x);
108 track_data.push_back(t.previous.y);
109 extra_data.push_back(static_cast<float>(t.age));
110 extra_data.push_back(t.tracked ? 1.0F : 0.0F);
111 }
112
113 if (!m_track_reducer) {
114 m_track_reducer = std::make_shared<ShaderExecutionContext<>>(
115 GpuComputeConfig {
116 .shader_path = "track_reduce.comp.spv",
117 .workgroup_size = { 256, 1, 1 },
118 .push_constant_size = sizeof(TrackReducePC) });
119 }
120
121 m_track_reducer->input(0, track_data)
122 .output(1, 3 * sizeof(glm::vec4))
123 .input(2, extra_data)
124 .push(TrackReducePC { .count = static_cast<uint32_t>(tracks.size()) });
125
126 const auto output = m_track_reducer->execute(Datum<std::vector<Kakshya::DataVariant>> {}, ExecutionContext {});
127 const auto raw = ShaderExecutionContext<>::read_output<float>(output, 1);
128
129 if (raw.size() >= 12) {
130 const auto n = static_cast<float>(tracks.size());
131 result.centroid = { raw[0] / n, raw[1] / n };
132 result.velocity = { raw[2] / n, raw[3] / n };
133 result.bounds = {
134 .x = raw[4],
135 .y = raw[5],
136 .w = raw[6] - raw[4],
137 .h = raw[7] - raw[5],
138 };
139 result.mean_speed_px = glm::length(glm::vec2 {
140 result.velocity.x * static_cast<float>(w),
141 result.velocity.y * static_cast<float>(h) });
142 result.mean_age = raw[8] / n;
143 result.max_age = static_cast<uint32_t>(std::lround(raw[9]));
144 result.tracked_count = static_cast<uint32_t>(std::lround(raw[10]));
145 result.lost_count = static_cast<uint32_t>(tracks.size()) - result.tracked_count;
146 }
147
148 return result;
149}

204void VisionAnalyzer::compute_shapes(Kinesis::Vision::FindElementsAnalysis& fe, uint32_t w, uint32_t h)
205{
206 const auto n = fe.boxes.size();
207 fe.shapes.assign(n, {});
208 fe.nearest_neighbor_distance_px.assign(n, 0.0F);
209 if (n == 0)
210 return;
211
212 const auto fw = static_cast<float>(w);
213 const auto fh = static_cast<float>(h);
214
215 std::vector<float> element_data;
216 element_data.reserve(n * 4);
217 std::vector<float> centroid_data;
218 centroid_data.reserve(n * 2);
219
220 for (const auto& b : fe.boxes) {
221 const auto it = std::ranges::find_if(fe.contours,
222 [&](const auto& c) { return c.label_id == b.label_id; });
223 const float area_px = (it != fe.contours.end()) ? it->area * fw * fh : 0.0F;
224 const float perimeter_px = (it != fe.contours.end()) ? it->perimeter : 0.0F;
225 element_data.push_back(area_px);
226 element_data.push_back(perimeter_px);
227 element_data.push_back(b.w * fw);
228 element_data.push_back(b.h * fh);
229
230 centroid_data.push_back((b.x + b.w * 0.5F) * fw);
231 centroid_data.push_back((b.y + b.h * 0.5F) * fh);
232 }
233
234 if (!m_shape_ctx) {
235 m_shape_ctx = std::make_shared<ShaderExecutionContext<>>(
236 GpuComputeConfig {
237 .shader_path = "element_shape.comp.spv",
238 .workgroup_size = { 256, 1, 1 },
239 .push_constant_size = sizeof(TrackReducePC) });
240 }
241 m_shape_ctx->input(0, element_data)
242 .output(1, n * sizeof(glm::vec2))
243 .push(TrackReducePC { .count = static_cast<uint32_t>(n) });
244
245 const auto shape_output = m_shape_ctx->execute(Datum<std::vector<Kakshya::DataVariant>> {}, ExecutionContext {});
246 const auto shape_raw = ShaderExecutionContext<>::read_output<float>(shape_output, 1);
247 for (size_t i = 0; i < n && (i + 1) * 2 <= shape_raw.size(); ++i) {
248 fe.shapes[i].compactness = shape_raw[i * 2];
249 fe.shapes[i].aspect_ratio = shape_raw[i * 2 + 1];
250 }
251
252 if (n < 2)
253 return;
254
255 if (!m_neighbor_ctx) {
256 m_neighbor_ctx = std::make_shared<ShaderExecutionContext<>>(
257 GpuComputeConfig {
258 .shader_path = "nearest_neighbor.comp.spv",
259 .workgroup_size = { 256, 1, 1 },
260 .push_constant_size = sizeof(TrackReducePC) });
261 }
262 m_neighbor_ctx->input(0, centroid_data)
263 .output(1, n * sizeof(float))
264 .push(TrackReducePC { .count = static_cast<uint32_t>(n) });
265
266 const auto dist_output = m_neighbor_ctx->execute(Datum<std::vector<Kakshya::DataVariant>> {}, ExecutionContext {});
267 const auto dist_raw = ShaderExecutionContext<>::read_output<float>(dist_output, 1);
268 for (size_t i = 0; i < n && i < dist_raw.size(); ++i)
269 fe.nearest_neighbor_distance_px[i] = dist_raw[i];
270}

290Kinesis::Vision::FieldSample VisionAnalyzer::sample_full_frame(
291 const std::shared_ptr<Core::VKImage>& image, uint32_t w, uint32_t h)
292{
293 if (!m_gradient_sample_ctx) {
294 m_gradient_sample_ctx = std::make_shared<TextureExecutionContext>(
295 GpuComputeConfig {
296 .shader_path = "region_sample.comp.spv",
297 .workgroup_size = { 256, 1, 1 },
298 .push_constant_size = sizeof(RegionSamplePC) },
299 Portal::Graphics::ImageFormat::RGBA8,
300 TextureExecutionContext::OutputMode::SCALAR,
301 1,
302 std::vector<GpuBufferBinding> {
303 { .set = 0, .binding = 2, .direction = GpuBufferBinding::Direction::OUTPUT, .element_type = GpuBufferBinding::ElementType::FLOAT32 } });
304 }
305
306 auto& foundry = Portal::Graphics::get_shader_foundry();
307
308 m_gradient_sample_ctx->stage_image(image);
309 m_gradient_sample_ctx->set_push_constants(RegionSamplePC { .px = 0, .py = 0, .pw = w, .ph = h });
310 m_gradient_sample_ctx->set_output_size(2, 3 * sizeof(glm::vec4));
311 m_gradient_sample_ctx->set_output_dimensions(1, 1);
312
313 const auto fence = m_gradient_sample_ctx->dispatch_async({});
314 m_gradient_sample_ctx->clear_output_dimensions();
315 foundry.wait_for_fence(fence);
316 foundry.release_fence(fence);
317
318 const auto gpu_result = m_gradient_sample_ctx->collect_result();
319
320 Kinesis::Vision::FieldSample out;
321 if (auto it = gpu_result.aux.find(2); it != gpu_result.aux.end() && it->second.size() >= 3 * sizeof(glm::vec4)) {
322 std::array<glm::vec4, 3> raw {};
323 std::memcpy(raw.data(), it->second.data(), sizeof(raw));
324 out.mean = raw[0];
325 out.min_val = raw[1];
326 out.max_val = raw[2];
327 }
328 return out;
329}

356 m_histogram_ctx->set_output_dimensions(w, h);
357
358 auto& foundry = Portal::Graphics::get_shader_foundry();
359 const auto fence = m_histogram_ctx->dispatch_async({});
360 m_histogram_ctx->clear_output_dimensions();
361 foundry.wait_for_fence(fence);
362 foundry.release_fence(fence);
363
364 std::array<uint32_t, 256> histogram {};
365 m_histogram_ctx->download_shared(0, 3, histogram.data(), sizeof(histogram));
366 return histogram;

442std::optional<Kinesis::Vision::VisionAnalysis> VisionAnalyzer::advance(
443 const std::shared_ptr<Core::VKImage>& image)
444{
445 if (!m_run.active) {
446 if (!image)
447 return std::nullopt;
448
449 m_frame = hold_frame(image);
450 if (!m_frame)
451 return std::nullopt;
452
453 m_run.resolved = Kinesis::Vision::resolve(m_query);
454 for (auto& entry : m_run.resolved) {
455 for (auto& step : entry.sequence.steps) {
456 if (step.op == Kinesis::Vision::VisionOp::TrackKeypoints
457 || step.op == Kinesis::Vision::VisionOp::OpticalFlowDense)
458 step.deferred = true;
459 }
460 }
461
462 m_run.results.clear();
463 m_run.results.reserve(m_run.resolved.size());
464 m_run.w = m_frame->get_width();
465 m_run.h = m_frame->get_height();
466 m_run.active = true;
467 m_label_lane = 0;
468 }
469
470 const size_t lane = m_run.results.size();
471 if (lane < m_run.resolved.size() && !run_lane(lane, m_run.resolved, m_run.results, m_frame))
472 return std::nullopt;
473
474 if (m_run.results.size() < m_run.resolved.size())
475 return std::nullopt;
476
477 m_run.active = false;
478 auto analysis = assemble(m_run.resolved, m_run.results, m_run.w, m_run.h);
479 m_results = std::make_shared<const std::vector<Kinesis::Vision::VisionResult>>(std::move(m_run.results));
480 return analysis;
481}

484{

495 return nullptr;

518bool VisionAnalyzer::run_lane(
519 size_t lane,
520 const std::vector<Kinesis::Vision::ResolvedSequence>& resolved,
521 std::vector<Kinesis::Vision::VisionResult>& results,
522 const std::shared_ptr<Core::VKImage>& image)
523{
524 if (m_executors.size() < resolved.size())
525 m_executors.resize(resolved.size());
526
527 const auto& entry = resolved[lane];
528 auto& executor = m_executors[lane];
529 if (!executor)
530 executor = std::make_unique<VisionGpuExecutor>();
531
532 if (std::ranges::any_of(entry.sequence.steps,
533 [](const auto& step) { return step.op == Kinesis::Vision::VisionOp::ConnectedComponents; }))
534 m_label_lane = lane;
535
536 const auto seeded = entry.seed_field ? entry.seed_field(results[entry.seed_lane]) : nullptr;
537 const auto& source = seeded ? seeded : image;
538 auto result = executor->run(entry.sequence, source, source->get_width(), source->get_height());
539
540 if (result.status != Kinesis::Vision::VisionStatus::COMPLETE)
541 return false;
542
543 results.push_back(std::move(result));
544 return true;
545}

547Kinesis::Vision::VisionAnalysis VisionAnalyzer::assemble(
548 const std::vector<Kinesis::Vision::ResolvedSequence>& resolved,
549 const std::vector<Kinesis::Vision::VisionResult>& results,
550 uint32_t w, uint32_t h)
551{
552 Kinesis::Vision::VisionAnalysis analysis;
553 std::optional<std::vector<Kinesis::Vision::TrackResult>> raw_tracks;
554 std::optional<std::vector<Kinesis::Vision::Keypoint>> raw_keypoints;
555
556 for (size_t lane = 0; lane < results.size(); ++lane) {
557 const auto& result = results[lane];
558 const auto& entry = resolved[lane];
559
560 collect_into(raw_tracks, raw_keypoints, analysis, result);
561
562 if (result.images.sobel) {
563 const auto& gray = result.gray || !entry.seed_field ? result.gray : results[entry.seed_lane].gray;
564 analysis.measure_appearance = measure_appearance(result, gray, w, h);
565 }
566 }
567
568 if (raw_tracks && m_context.track_objects) {
569 const auto& ctx = *m_context.track_objects;
570 if (ctx.subject_track_id) {
571 const auto id = *ctx.subject_track_id;
572 std::erase_if(*raw_tracks, [&](const auto& t) { return t.id != id; });
573 } else if (ctx.subject_bounds) {
574 const auto& bounds = *ctx.subject_bounds;
575 std::erase_if(*raw_tracks, [&](const auto& t) { return !inside(bounds, t.position); });
576 }
577 }
578
579 if (raw_tracks)
580 analysis.track_objects = reduce_tracks(*raw_tracks, w, h);
581
582 if (raw_keypoints && m_context.detect_features_bounds) {
583 const auto& bounds = *m_context.detect_features_bounds;
584 std::erase_if(*raw_keypoints, [&](const auto& k) { return !inside(bounds, k.position); });
585 }
586
587 if (raw_keypoints)
588 analysis.detect_features = reduce_keypoints(*raw_keypoints);
589
590 if (analysis.find_elements)
591 compute_shapes(*analysis.find_elements, w, h);
592
593 return analysis;
594}

596VisionAnalyzer::output_type VisionAnalyzer::run_operation(const input_type& input)
597{
598 output_type output;
