166std::string encode_utf8(uint32_t codepoint)
167{
168 if (!utf8proc_codepoint_valid(static_cast<utf8proc_int32_t>(codepoint))) {
169 return {};
170 }
171
172 std::array<utf8proc_uint8_t, 4> buf {};
173 const utf8proc_ssize_t n = utf8proc_encode_char(
174 static_cast<utf8proc_int32_t>(codepoint), buf.data());
175
176 if (n <= 0) {
177 return {};
178 }
179
180 return { reinterpret_cast<const char*>(buf.data()), static_cast<size_t>(n) };

183size_t next_codepoint_offset(std::string_view text, size_t byte_offset)
184{
185 if (byte_offset >= text.size()) {
186 return text.size();
187 }
188
189 const auto* bytes = reinterpret_cast<const utf8proc_uint8_t*>(text.data());
190 const auto remaining = static_cast<utf8proc_ssize_t>(text.size() - byte_offset);
191
192 utf8proc_int32_t codepoint = 0;
193 const utf8proc_ssize_t n = utf8proc_iterate(bytes + byte_offset, remaining, &codepoint);
194
195 return byte_offset + static_cast<size_t>(n > 0 ? n : 1);

198size_t previous_codepoint_offset(std::string_view text, size_t byte_offset)
199{
200 if (byte_offset == 0) {
201 return 0;
202 }
203
204 size_t pos = byte_offset - 1;

206 size_t steps = 0;
207
208 while (pos > 0 && steps < max_continuation_bytes
209 && (static_cast<uint8_t>(text[pos]) & 0xC0) == 0x80) {
210 --pos;
211 ++steps;

263 if (dy < best_line_dy) {
264 best_line_dy = dy;
265 line_y = q.y0;
266 }
267 }
268
269 const GlyphQuad* left = nullptr;
270 const GlyphQuad* right = nullptr;
271
272 for (const auto& q : layout.quads) {

274 continue;
275 }
276
277 if (x >= q.x0 && x <= q.x1) {

279 ? next_codepoint_offset(text, q.byte_offset)
280 : q.byte_offset;

286 right = &q;
287 }
288 }

290 const GlyphQuad* anchor = left ? left : right;
291 const float pen_line_y = x_at(text, atlas, anchor->byte_offset, pen_x, pen_y, wrap_w, tab_width).y;

295 };

297 size_t bracket_start = 0;

300 } else {

302 while (bracket_start > 0) {

304 if (!same_line(prev)) {
305 break;

307 bracket_start = prev;
308 }
309 }
310
311 size_t bracket_end = text.size();
312 if (right) {
313 bracket_end = right->byte_offset;
314 } else {
315 bracket_end = next_codepoint_offset(text, left->byte_offset);
316 while (bracket_end < text.size()) {
317 const size_t next = next_codepoint_offset(text, bracket_end);
318 if (!same_line(next)) {
319 break;
320 }
321 bracket_end = next;
322 }
323 }
324
325 if (bracket_start >= bracket_end) {
326 return bracket_start;
327 }
328
329 size_t best_offset = bracket_start;
330 float best_dx = abs_f(x_at(text, atlas, bracket_start, pen_x, pen_y, wrap_w, tab_width).x - x);
331
332 for (size_t off = bracket_start; off < bracket_end;) {
333 const size_t next = next_codepoint_offset(text, off);
334 const float dx = abs_f(x_at(text, atlas, next, pen_x, pen_y, wrap_w, tab_width).x - x);
335 if (dx < best_dx) {
336 best_dx = dx;
337 best_offset = next;
338 }
339 off = next;
340 }
341
342 return best_offset;
