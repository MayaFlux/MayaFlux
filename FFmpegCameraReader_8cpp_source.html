150bool FFmpegCameraReader::pull_frame(
151 const std::shared_ptr<Kakshya::CameraContainer>& container)
152{
153 if (!container)
154 return false;
155
156 std::shared_lock lock(m_ctx_mutex);
157 if (!m_video->is_codec_valid())
158 return false;
159
160 uint8_t* dest = container->mutable_frame_ptr();
161 if (!dest)
162 return false;
163
164 AVPacket* pkt = av_packet_alloc();
165 if (!pkt)
166 return false;
167
168 AVFrame* frame = av_frame_alloc();
169 if (!frame) {
170 av_packet_free(&pkt);
171 return false;
172 }
173
174 bool got_frame = false;
175
176 while (!got_frame) {
177 int ret = av_read_frame(m_demux->format_context, pkt);
178 if (ret < 0)
179 break;
180
181 if (pkt->stream_index != m_video->stream_index) {
182 av_packet_unref(pkt);
183 continue;
184 }
185
186 ret = avcodec_send_packet(m_video->codec_context, pkt);
187 av_packet_unref(pkt);
188
189 if (ret < 0 && ret != AVERROR(EAGAIN))
190 break;
191
192 ret = avcodec_receive_frame(m_video->codec_context, frame);
193 if (ret == AVERROR(EAGAIN))
194 continue;
195 if (ret < 0)
196 break;
197
198 if (!m_scaler_ready) {
199 if (!m_video->rebuild_scaler_from_frame(
200 frame,
201 static_cast<uint32_t>(frame->width),
202 static_cast<uint32_t>(frame->height))) {
203 MF_ERROR(Journal::Component::IO, Journal::Context::Runtime,
204 "FFmpegCameraReader: scaler init failed: {}",
205 m_video->last_error());
206 break;
207 }
208 const size_t buf_bytes = static_cast<size_t>(m_video->out_linesize) * m_video->out_height;
209 m_sws_buf.assign(buf_bytes, 0);
210 m_scaler_ready = true;
211 }
212
213 uint8_t* sws_dst[1] = { m_sws_buf.data() };
214 int sws_stride[1] = { m_video->out_linesize };
215
216 sws_scale(m_video->sws_context,
217 frame->data, frame->linesize,
218 0, static_cast<int>(m_video->height),
219 sws_dst, sws_stride);
220
221 const int packed_stride = static_cast<int>(m_video->out_width * m_video->out_bytes_per_pixel);
222
223 if (m_video->out_linesize == packed_stride) {
224 std::memcpy(dest, m_sws_buf.data(),
225 static_cast<size_t>(packed_stride) * m_video->out_height);
226 } else {
227 for (uint32_t row = 0; row < m_video->out_height; ++row) {
228 std::memcpy(
229 dest + static_cast<size_t>(row) * packed_stride,
230 m_sws_buf.data() + static_cast<size_t>(row) * m_video->out_linesize,
231 static_cast<size_t>(packed_stride));
232 }
233 }
234
235 container->mark_ready_for_processing(true);
236 got_frame = true;
237 }
238
239 av_frame_free(&frame);
240 av_packet_free(&pkt);
241
242 return got_frame;
243}

307void FFmpegCameraReader::decode_thread_func()
308{
309 while (!m_decode_stop.load()) {
310 {
311 std::unique_lock lock(m_decode_mutex);
312 m_decode_cv.wait(lock, [this] {
313 return m_decode_stop.load()
314 || m_frame_requested.load(std::memory_order_relaxed);
315 });
316 }
317
318 if (m_decode_stop.load())
319 break;
320
321 m_frame_requested.store(false, std::memory_order_relaxed);
322
323 auto container = m_container_ref.lock();
324 if (container)
325 pull_frame(container);
326 }
327
328 m_decode_active.store(false);
329}
