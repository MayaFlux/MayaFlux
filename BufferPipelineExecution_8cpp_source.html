45void BufferPipeline::flow_through(size_t index, uint64_t cycle)
46{
47 if (index + 1 >= m_operations.size()) {
48 return;
49 }
50
51 auto& dependent_op = m_operations[index + 1];
52
53 if (m_data_states[index + 1] != DataState::CONSUMED
54 && BufferOperation::is_process_phase_operation(dependent_op)) {
55 process_operation(dependent_op, cycle);
56 m_data_states[index + 1] = DataState::READY;

61{

68
