118 std::shared_ptr<Core::VKImage> hold_copy(
119 Portal::Graphics::ImageCacheEntry& cache,

121 {

123 return nullptr;

139 /**
140 * @brief Analyses of @p frames into @p results, paced by the work:
141 * advanced one sequence per graphics frame, signalled as each
142 * completes.
143 *
144 * With @p apply_to, its frame is copied in the tick the analyzed frame
145 * is taken, and the analysis is applied to that copy.
146 */
147 void fit_results(
148 Vruta::TaskScheduler& scheduler,
149 const FrameResults& results,
150 const std::shared_ptr<VisionAnalyzer>& analyzer,
151 const std::shared_ptr<Buffers::TextureBuffer>& frames,
152 const std::shared_ptr<Buffers::TextureBuffer>& apply_to,
153 const std::shared_ptr<Kakshya::SignalSourceContainer>& source,
154 const std::function<bool()>& ended,
155 const std::string& name)
156 {
157 auto held = std::make_shared<Portal::Graphics::ImageCacheEntry>();
158 auto companion = std::make_shared<std::shared_ptr<Core::VKImage>>();
159 auto mid_run = std::make_shared<bool>(false);
160
161 Kriya::frame_results<VisionIO>(scheduler, results,
162 [analyzer, frames, apply_to, source, ended, held, companion, mid_run]() -> std::optional<VisionIO> {
163 if (ended())
164 return std::nullopt;
165
166 const auto image = frames->get_gpu_texture();
167 if (!*mid_run) {
168 if (!image)
169 return std::nullopt;
170 if (apply_to)
171 *companion = hold_copy(*held, apply_to->get_gpu_texture());
172 *mid_run = true;
173 }
174
175 const auto analysis = analyzer->advance(image);
176 if (!analysis)
177 return std::nullopt;
178 *mid_run = false;
179
180 const auto target = apply_to ? *companion : analyzer->get_frame();
181 if (!target)
182 return std::nullopt;
183 return analyzed_frame(source, *analysis, target);
184 },
185 name);
186 }

188 /**
189 * @brief Analyses of @p frames into @p results every @p interval_seconds
190 * on the graphics clock, each run to completion when it starts.
191 *
192 * With @p apply_to, the analysis is applied to its frame of the same
193 * tick.
194 */
195 void interval_results(
196 Vruta::TaskScheduler& scheduler,
197 const FrameResults& results,
198 const std::shared_ptr<VisionAnalyzer>& analyzer,
199 const std::shared_ptr<Buffers::TextureBuffer>& frames,
200 const std::shared_ptr<Buffers::TextureBuffer>& apply_to,
201 const std::shared_ptr<Kakshya::SignalSourceContainer>& source,
202 double interval_seconds,
203 const std::function<bool()>& ended,
204 const std::string& name)
205 {
206 scheduler.add_task(Kriya::metro(interval_seconds,
207 [weak_results = std::weak_ptr(results), analyzer, frames, apply_to, source, ended]() {
208 if (ended())
209 return;
210 const auto target = weak_results.lock();
211 const auto image = frames->get_gpu_texture();
212 if (!target || !image)
213 return;
214
215 const auto applied = apply_to ? apply_to->get_gpu_texture() : image;
216 if (!applied)
217 return;
218 target->signal(analyzed_frame(source, analyzer->analyze_vision(image), applied));
219 },
220 Vruta::ProcessingToken::FRAME_ACCURATE),
221 name);
222 }

225 * @brief Producer of analyses of @p source's frames into @p results,
226 * paced by @p interval_seconds or, at zero, by the work, as a task
227 * named @p name. Results carry the image the analysis is applied
228 * to: the analyzed frame, or @p apply_to's frame of the same

231 void start_producer(

307 /**
308 * @brief Analyze and extract @p count frames in order on one matrix,
309 * appending each extracted image to a collection of exactly that
310 * many layers at most.
311 */
312 std::shared_ptr<Kakshya::TextureCollection> record_frames(
313 size_t count,
314 const std::function<std::shared_ptr<Core::VKImage>(size_t)>& frame_at,
315 const std::shared_ptr<Kakshya::SignalSourceContainer>& source,
316 const std::shared_ptr<VisionMatrix>& matrix,
317 Kakshya::TextureCollectionSpec spec)
318 {
319 spec.max_bytes = 0;
320 spec.ring_layers = 0;
321
322 auto collection = std::make_shared<Kakshya::TextureCollection>(spec);
323
324 for (size_t i = 0; i < count; ++i) {
325 const auto image = frame_at(i);
326 if (!image)
327 continue;
328
329 VisionIO frame = source ? VisionIO { source, source } : VisionIO {};
330 frame.metadata["vision_image"] = image;
331 if (const auto extracted = extracted_image(run_chain(matrix, std::move(frame))))
332 collection->append(extracted);
333 }
334
335 return collection;
336 }

374std::shared_ptr<Buffers::TextureBuffer> VisionWorkflow::source_frames(
375 const std::shared_ptr<Kakshya::VideoStreamContainer>& source,
376 std::string_view caller) const
377{
378 if (!source) {
379 MF_WARN(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
380 "VisionWorkflow::{}: no video stream", caller);
381 return nullptr;
382 }
383
384 const auto io = m_io_manager.lock();
385 if (!io || m_scheduler.expired() || m_event_manager.expired()) {
386 MF_ERROR(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
387 "VisionWorkflow::{}: the engine's managers are gone", caller);
388 return nullptr;
389 }
390
391 std::shared_ptr<Buffers::TextureBuffer> input;
392 if (const auto camera = std::dynamic_pointer_cast<Kakshya::CameraContainer>(source)) {
393 input = io->get_camera_buffer(camera);
394 if (!input)
395 input = io->hook_camera_to_buffer(camera);
396 } else {
397 input = io->get_video_buffer(source);
398 if (!input)
399 input = io->hook_video_container_to_buffer(source);
400 }
401
402 if (!input) {
403 MF_ERROR(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
404 "VisionWorkflow::{}: the video stream could not be hooked to a buffer", caller);
405 return nullptr;
406 }
407
408 return input;
409}

532 if (!chimera_pipeline)
533 return true;
534 if (chimera_pipeline->is_running()) {
535 *seen_running = true;
536 return false;
537 }

539 });

544void VisionWorkflow::start(
545 const std::shared_ptr<Kakshya::VideoStreamContainer>& source,
546 const std::shared_ptr<Buffers::TextureBuffer>& frames,
547 const LiveConfig& live,
548 const std::shared_ptr<VisionMatrix>& matrix,
549 const std::string& name,
550 std::function<void(const VisionIO&)> consumer,
551 std::function<bool()> stop_when)
552{
553 auto results = std::make_shared<Vruta::BroadcastSource<VisionIO>>();
554 auto event = Kriya::subscribe(*m_event_manager.lock(), results, std::move(consumer));
555 keep(name, event);
556
557 auto ended = stop_at_end(source, m_scheduler, m_event_manager, event, name);
558 if (stop_when) {
559 ended = [at_end = std::move(ended), stop_when = std::move(stop_when),
560 scheduler = m_scheduler, events = m_event_manager, weak_consumer = std::weak_ptr(event), name]() {
561 if (at_end())
562 return true;
563 if (!stop_when())
564 return false;
565 stop_run(scheduler, events, weak_consumer, name);
566 return true;
567 };
568 }
569
570 start_producer(*m_scheduler.lock(), results, source, frames, live.apply_to, matrix, live.interval_seconds,
571 ended, name);
572}
