32 [[nodiscard]] std::shared_ptr<Nodes::GpuSync::MeshWriterNode> resolve_writer(
33 const std::optional<Kakshya::MeshData>& mesh,
34 const StructureConfig::MeshExpression& expression,
35 const std::shared_ptr<Nodes::GpuSync::MeshWriterNode>& writer,
36 std::string_view context)
37 {
38 const unsigned int source_count = static_cast<unsigned int>(mesh.has_value())
39 + static_cast<unsigned int>(static_cast<bool>(expression))
40 + static_cast<unsigned int>(writer != nullptr);
41
42 if (source_count != 1) {
43 MF_ERROR(Journal::Component::API, Journal::Context::Init,
44 "mint: {} requires exactly one mesh, expression or writer source", context);
45 return nullptr;
46 }
47
48 if (writer) {
49 return writer;
50 }
51
52 Kakshya::MeshData data = mesh ? *mesh : expression();
53 if (!data.is_valid()) {
54 MF_ERROR(Journal::Component::API, Journal::Context::Init,
55 "mint: {} produced invalid MeshData", context);
56 return nullptr;
57 }
58
59 auto resolved = std::make_shared<Nodes::GpuSync::MeshWriterNode>(data.vertex_count());
60 resolved->set_mesh(data);
61 return resolved;
62 }

117 const StructureConfig::Object& config) const
118 {
119 if (!has_target(config.render)) {
120 return nullptr;
121 }
122
123 const unsigned int source_count = static_cast<unsigned int>(config.mesh.has_value())
124 + static_cast<unsigned int>(static_cast<bool>(config.expression))
125 + static_cast<unsigned int>(config.writer != nullptr);

128 "mint: Object requires exactly one mesh, expression or writer source");
129 return nullptr;
130 }
131

134 if (!writer) {
135 return nullptr;
136 }
137
138 auto buffer = std::make_shared<Buffers::MeshBuffer>(writer);
139
140 StructureConfig::RenderConfig render = config.render;
141 bind_mesh_textures(buffer, render, config.textures);
142 register_graphics_buffer(buffer);
143 buffer->setup_rendering(render);

147 [[nodiscard]] std::shared_ptr<Buffers::MeshNetworkBuffer> build(
148 const StructureConfig::Model& config) const
149 {
150 if (!has_target(config.render)) {
151 return nullptr;
152 }
153
154 auto network = config.network
155 ? config.network
156 : std::make_shared<Nodes::Network::MeshNetwork>();
157 const auto first_component_slot = static_cast<uint32_t>(network->slot_count());
158
159 for (size_t index = 0; index < config.components.size(); ++index) {
160 const auto& component = config.components[index];
161 auto writer = resolve_writer(
162 component.mesh, component.expression, component.writer, "Model component");
163 if (!writer) {
164 return nullptr;
165 }
166
167 if (component.parent_component && component.parent_slot) {
168 MF_ERROR(Journal::Component::API, Journal::Context::Init,
169 "mint: Model component {} cannot name both a component and slot parent",
170 index);
171 return nullptr;
172 }
173
174 std::optional<uint32_t> parent;
175 if (component.parent_component) {
176 if (*component.parent_component >= index) {
177 MF_ERROR(Journal::Component::API, Journal::Context::Init,
178 "mint: Model component {} must name an earlier component as parent",
179 index);
180 return nullptr;
181 }
182 parent = first_component_slot + *component.parent_component;
183 } else if (component.parent_slot) {
184 if (*component.parent_slot >= first_component_slot) {
185 MF_ERROR(Journal::Component::API, Journal::Context::Init,
186 "mint: Model component {} parent slot is not in the supplied network",
187 index);
188 return nullptr;
189 }
190 parent = component.parent_slot;
191 }
192
193 const uint32_t slot_index = network->add_slot(
194 "component_" + std::to_string(index), writer, parent);
195 network->get_slot(slot_index).local_transform = component.local_transform;
196 }
197
198 if (network->slot_count() == 0) {
199 MF_ERROR(Journal::Component::API, Journal::Context::Init,
200 "mint: Model requires at least one mesh component");
201 return nullptr;
202 }
203
204 register_network_if_needed(network);
205 auto buffer = std::make_shared<Buffers::MeshNetworkBuffer>(
206 network, config.over_allocate_factor);
207 StructureConfig::RenderConfig render = config.render;
208 bind_mesh_textures(buffer, render, config.textures);
209 register_graphics_buffer(buffer);
210 buffer->setup_rendering(render);
211 return buffer;
212 }

214 [[nodiscard]] std::shared_ptr<Buffers::InstanceNetworkBuffer> build(
215 const StructureConfig::Instances& config) const
216 {
217 if (!has_target(config.render)) {
218 return nullptr;
219 }
220
221 auto network = config.network
222 ? config.network
223 : std::make_shared<Nodes::Network::InstanceNetwork>();
224
225 if (!config.transforms.empty()) {
226 auto writer = resolve_writer(
227 config.prototype_mesh,
228 config.prototype_expression,
229 config.prototype_writer,

231 if (!writer) {
232 return nullptr;
233 }
234
235 for (size_t index = 0; index < config.transforms.size(); ++index) {
236 const uint32_t slot_index = network->add_slot(

239 }
240 }
241
242 if (network->slot_count() == 0) {
243 MF_ERROR(Journal::Component::API, Journal::Context::Init,
244 "mint: Instances requires an existing populated network or transforms");
245 return nullptr;
246 }
247
248 register_network_if_needed(network);
249 auto buffer = std::make_shared<Buffers::InstanceNetworkBuffer>(
250 network, config.over_allocate_factor);
251 StructureConfig::RenderConfig render = config.render;
252 add_all_textures(render, config.textures);
253 register_graphics_buffer(buffer);

258 [[nodiscard]] std::shared_ptr<Buffers::NetworkGeometryBuffer> build(
259 const StructureConfig::Assembly& config) const
260 {
261 if (!has_target(config.render)) {
262 return nullptr;
263 }
264
265 auto network = config.network
266 ? config.network
267 : std::make_shared<Nodes::Network::AssemblyNetwork>();
268
269 for (const auto& source : config.sources) {
270 network->add_geometry(source);
271 }
272
273 if (network->get_node_count() == 0) {
274 MF_ERROR(Journal::Component::API, Journal::Context::Init,
275 "mint: Assembly requires at least one geometry source");
276 return nullptr;
277 }
278
279 register_network_if_needed(network);
280 auto buffer = std::make_shared<Buffers::NetworkGeometryBuffer>(network);
281 StructureConfig::RenderConfig render = config.render;
282 render.triangulate = true;
283 add_all_textures(render, config.textures);
284 register_graphics_buffer(buffer);
285 buffer->setup_rendering(render);
286 buffer->get_render_processor()->set_mill_spec(config.mill);

290 [[nodiscard]] std::shared_ptr<Buffers::ComputeMeshBuffer> build(
291 const StructureConfig::Isosurface& config) const
292 {
293 if (!has_target(config.render)) {
294 return nullptr;
295 }
296
297 const bool has_buffer = config.buffer != nullptr;
298 const bool has_config = config.config.has_value();
299 if (static_cast<unsigned int>(has_buffer)
300 + static_cast<unsigned int>(has_config)
301 != 1) {
302 MF_ERROR(Journal::Component::API, Journal::Context::Init,
303 "mint: Isosurface requires exactly one buffer or config");
304 return nullptr;
305 }
306
307 auto buffer = has_buffer
308 ? config.buffer
309 : std::make_shared<Buffers::ComputeMeshBuffer>(*config.config);
310
311 StructureConfig::RenderConfig render = config.render;
312 if (!config.textures.empty() && config.textures.front()) {
313 const std::string binding = render.default_texture_binding.empty()
314 ? "diffuseTex"
315 : render.default_texture_binding;
316 buffer->set_texture(config.textures.front(), binding);
317 render.default_texture_binding = binding;
318 }
319 add_extra_textures(render, config.textures, 1);
320 register_graphics_buffer(buffer);
321 buffer->setup_rendering(render);
322 return buffer;
