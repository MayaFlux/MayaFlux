3/**
4 * @file Boolean.hpp
5 * @brief Pure decision functions of bits and an array of numbers.
6 *
7 * The boolean counterpart of Kernels.hpp. Every kernel returns a bit and
8 * keeps whatever it remembers in the array it is given, so the behaviour is
9 * the numbers: the array can be retuned live, saved and restored with the
10 * node, and cloned. A signal is a bit when it exceeds the kernel's threshold,
11 * and a kernel driven by an edge counts a signal that starts high as a rise.
12 *
13 * Four shapes, each the function type of the matching Logic constructor:
14 * - history kernels: `bool(span<const bool> history, span<const double>)`,
15 * history[0] the newest bit
16 * - input kernels: `bool(double input, span<double>)`
17 * - time kernels: `bool(double input, double time, span<double>)`
18 * - parallel kernels: `bool(const vector<double>& inputs, span<double>)`, one
19 * value per input slot as documented by each kernel
20 *
21 * `Logic(Kinesis::Discrete::truth_table, 3, truth_table_state(3, rule))` is a
22 * complete node. Every kernel returns false for an array too short to hold
23 * its layout.
24 */
25

28/**
29 * @brief One step of the 32-bit xorshift generator.
30 * @param state Nonzero state; zero stays zero
31 *
32 * Small enough to be held exactly in an element of the array, which is what
33 * the random kernels do.
34 */
35[[nodiscard]] inline uint32_t xorshift32(uint32_t state) noexcept
36{
37 state ^= state << 13U;
38 state ^= state >> 17U;
39 state ^= state << 5U;
40 return state;
41}
