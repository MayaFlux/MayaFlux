29/**
30 * @class Perception
31 * @brief A value of any type, read from the world through the perception context and handed to its targets.
32 *
33 * T is whatever is sensed: a scalar from a node or buffer, a vector, a struct
34 * of readings, a Tendency, a DataVariant. The source reads it, using the
35 * context when it needs position or neighbours. A target is any callable taking
36 * const T& and decides where it lands, such as a cell the perception function
37 * reads. On each invoke the source runs once and every target gets the value,
38 * in the order they were added.
39 *
40 * How many targets a Perception may have is decided by its owner: a Sensor
41 * keeps one, an Agent keeps as many as are added.
42 */
43template <typename T>
44class Perception {
45public:
46 using Source = std::function<T(const PerceptionContext&)>;
47 using Target = std::function<void(const T&)>;
48
49 explicit Perception(Source source)
50 : m_source(std::move(source))
51 {
52 }

54 /** @brief Add a target. Returns this perception for chaining. */

58 return *this;
59 }

62 void clear_targets() { m_targets.clear(); }
