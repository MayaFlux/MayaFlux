9/**
10 * @struct CameraConfig
11 * @brief Configuration for opening a camera device via FFmpeg.
12 *
13 * Platform device string conventions:
14 * - Linux: "/dev/video0", "/dev/video1", …
15 * - macOS: "0" (AVFoundation device index) or a camera name such as
16 * "FaceTime HD Camera"; frame rate must equal a rate range maximum
17 * - Windows: "video=Integrated Camera" (DirectShow filter name)
18 *
19 * Resolution, frame rate and input format are hints. If the device rejects
20 * them, open() retries without the input format, then without frame rate,
21 * then with device defaults. The negotiated values are available from
22 * FFmpegCameraReader after open().
23 */
24struct MAYAFLUX_API CameraConfig {
25 std::string device_name; ///< Platform device string.
26 uint32_t target_width { 1920 }; ///< Requested width in pixels.
27 uint32_t target_height { 1080 }; ///< Requested height in pixels.
28 double target_fps { 30.0 }; ///< Hint only; device may ignore.
29 std::string format_override; ///< Leave empty to use CAMERA_FORMAT for current platform.
30 int pixel_format { -1 }; ///< Target AVPixelFormat as int; negative selects AV_PIX_FMT_RGBA.
31 std::string display_name; ///< Human-readable device name for a chooser.
32 std::string input_format; ///< Capture format or codec (e.g. "mjpeg"); empty selects mjpeg where the platform supports it.
33};

35/**
36 * @struct CameraMode
37 * @brief A requested capture mode offered by a source, for a chooser.
38 *
39 * A mode is a request, not a guarantee: open() falls back to the nearest
40 * mode the device accepts. Modes marked high_end ask for more than most
41 * webcams deliver and are likely to fall back on ordinary cameras.
42 */
43struct MAYAFLUX_API CameraMode {
44 std::string label; ///< Caption for a chooser.
45 uint32_t width {}; ///< Requested width in pixels, 0 leaves it to the device.
46 uint32_t height {}; ///< Requested height in pixels, 0 leaves it to the device.
47 double fps {}; ///< Requested frame rate, 0 leaves it to the device.
48 bool high_end {}; ///< Needs a capable camera.
49
50 /** @brief Write this mode's size and frame rate into @p config. */
51 void apply(CameraConfig& config) const
52 {
53 config.target_width = width;
54 config.target_height = height;
55 config.target_fps = fps;
56 }
57};

59/**
60 * @class CameraSource
61 * @brief Abstract interface for a live camera backend hosted by IOManager.
62 *
63 * Enumerates configurations, opens a selected device, then provides
64 * negotiated parameters, frame delivery, and IOService wiring.
65 *
66 * A camera source is a live, unbounded stream: no ring buffer, no seek, no
67 * batch decode. One frame is pulled per process cycle, demand-driven by
68 * CameraContainer::process_default(), never pushed speculatively ahead of
69 * the consumer.
70 *
71 * Two integration paths apply to any implementation:
72 * - Managed: IOManager owns construction and opening (backend-specific),
73 * then calls register_camera_source(), which assigns a
74 * reader_id and wires the container's IOService callback.
75 * - Standalone: the caller drives the same sequence manually —
76 * create_container() → setup_io_service(id) →
77 * set_container() → close() — without IOManager.
78 *
79 * FFmpegCameraReader is the first concrete implementation.
80 */
81class MAYAFLUX_API CameraSource {
82public:
