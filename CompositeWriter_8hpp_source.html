7/**
8 * @class CompositeWriter
9 * @brief Typed writer contract for schema-bearing Composite files.
10 *
11 * A concrete writer chooses a file format and encodes Composite fields
12 * without exposing unrestricted byte writes. A file is opened with one
13 * layout, then receives one or more bounded slices before close().
14 *
15 * Usage:
16 * @code
17 * bool save_measurements(
18 * CompositeWriter& writer, const Kakshya::CompositeArray& array)
19 * {
20 * return writer.write("measurements.csv", array);
21 * }
22 * @endcode
23 *
24 * @note The source array must remain alive throughout each write_rows()
25 * call. The writer does not retain the borrowed slice.
26 */
27class MAYAFLUX_API CompositeWriter {
28public:
29 virtual ~CompositeWriter() = default;
30
31 /**
32 * @brief Check whether this writer supports the target path.
33 * @param filepath Output path.
34 * @return True for a supported file format.
35 */
36 [[nodiscard]] virtual bool can_write(
37 const std::string& filepath) const = 0;
38
39 /**
40 * @brief Open a new file with the schema all subsequent slices must use.
41 * @param filepath Output path.
42 * @param layout Finalized or declarative Composite field layout.
43 * @return True when the file is ready to receive elements.
44 */
45 virtual bool open(
46 const std::string& filepath,
47 const Kakshya::CompositeLayout& layout) = 0;
48
49 /**
50 * @brief Write a bounded slice of elements.
51 * @param rows Borrowed elements whose field names, types, and order
52 * must match the layout passed to open().
53 * @return True if every element was accepted by the writer. On failure,
54 * elements before the failing one may already have been written.
55 */
56 virtual bool write_rows(const Kakshya::CompositeSlice& rows) = 0;

88
89 /**

91 *
92 * Opens with the array's layout, writes all its elements, then closes

95 *
96 * @param filepath Output path.
97 * @param array Source array.
98 * @return True only when open, write, and close all succeed.

100 bool write(
