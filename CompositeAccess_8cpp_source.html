24std::optional<CompositeAccess> as_composite_access(
25 const DataVariant& elements, const DataVariant& text, const CompositeLayout& layout)
26{
27 const auto* element_bytes = std::get_if<std::vector<uint8_t>>(&elements);
28 const auto* text_bytes = std::get_if<std::vector<uint8_t>>(&text);
29 const size_t stride = layout.stride_bytes();
30 if (!element_bytes || !text_bytes || stride == 0 || element_bytes->size() % stride != 0) {
31 MF_WARN(Journal::Component::Kakshya, Journal::Context::Runtime,
32 "as_composite_access: invalid storage or element stride");
33 return std::nullopt;
34 }
35
36 for (const auto& field : layout.fields()) {
37 if (field.offset_bytes > stride || field.size_bytes > stride - field.offset_bytes
38 || (field.type == typeid(std::string) && field.size_bytes != 2 * sizeof(uint64_t))) {
39 MF_WARN(Journal::Component::Kakshya, Journal::Context::Runtime,
40 "as_composite_access: invalid layout for field '{}'", field.name);
41 return std::nullopt;
42 }
43 }
44
45 return CompositeAccess(elements, text, layout);
46}
