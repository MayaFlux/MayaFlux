203 /**
204 * @brief Register mapped's element for scroll reflow in panel.
205 *
206 * When reposition is empty, one is built from mapped's own buffer,
207 * geometry_fn, and state: it reruns geometry_fn unconditionally
208 * (assumes the Form was built with live_offset wired to panel's own
209 * offset, so it reads the new position itself) and pushes the
210 * resulting bounds_hint/contains to layer. These are the same two
211 * effects Mapped<T>::sync(Layer*) has, replicated rather than
212 * called through, so the closure captures only independently-owned
213 * pieces (buffer and state are shared_ptr, geometry_fn is a
214 * copyable std::function), never a pointer back to mapped itself,
215 * since the closure outlives this call, stored inside panel.
216 *
217 * @param panel Scrollable to track into.
218 * @param layer Layer both panel and mapped's element were
219 * registered on.
220 * @param mapped Element to track. Only buffer/geometry_fn/
221 * state/element.id are read, nothing is stored.
222 * @param base_bounds mapped's current on-screen NDC bounds.
223 * @param reposition Called with the scrolled bounds on every
224 * scroll. Built as described above when empty.
225 */
226 template <typename T>
227 void track(
228 Scrollable& panel,
229 Layer& layer,
230 const Mapped<T>& mapped,
231 Kinesis::AABB2D base_bounds,
232 std::function<void(Kinesis::AABB2D)> reposition = {})
233 {
234 if (!reposition) {
235 auto buffer = mapped.element.buffer;
236 auto geometry_fn = mapped.geometry_fn;
237 auto state = mapped.state;
238 auto id = mapped.element.id;
239 auto* layer_ptr = &layer;
240
241 reposition = [buffer, geometry_fn, state, id, layer_ptr](Kinesis::AABB2D) {
242 if (!buffer || !geometry_fn || !state)
243 return;
244
245 std::vector<uint8_t> bytes;
246 Element el;
247 el.id = id;
248 geometry_fn(state->value, bytes, el);
249 buffer->submit(bytes);
250
251 if (el.bounds_hint)
252 layer_ptr->set_bounds(id, *el.bounds_hint);
253 if (el.contains)
254 layer_ptr->set_contains(id, el.contains);
255 };
256 }
257
258 panel.track(layer, mapped.element.id, base_bounds, std::move(reposition));
259 }
