155void TextureContainer::from_image_array(const std::shared_ptr<Core::VKImage>& image)
156{
157 if (!image || !image->is_initialized()) {
158 MF_ERROR(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
159 "TextureContainer::from_image_array called with uninitialised image");
160 return;
161 }
162
163 const auto n = static_cast<uint32_t>(m_data.size());
164 if (image->get_array_layers() < n) {
165 MF_ERROR(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
166 "TextureContainer::from_image_array image has {} layers, container expects {}",
167 image->get_array_layers(), n);
168 return;
169 }
170
171 const size_t layer_bytes = byte_size();
172 std::vector<uint8_t> combined(layer_bytes * n);
173 TextureLoom::instance().download_data(image, combined.data(), combined.size(), nullptr);
174
175 const size_t element_count = static_cast<size_t>(m_width) * m_height * m_channels;
176 for (uint32_t i = 0; i < n; ++i) {
177 Memory::SeqlockWriteGuard g(m_slot_locks[i]);
178 m_data[i] = make_empty_storage(m_format, element_count);
179 auto [ptr, bytes] = variant_bytes_mutable(m_data[i]);
180 if (ptr && bytes == layer_bytes)
181 std::memcpy(ptr, combined.data() + i * layer_bytes, layer_bytes);
182
183 m_normalised_dirty[i].store(true, std::memory_order_release);
184 }
185
186 update_processing_state(ProcessingState::READY);
187}

237 uint32_t layer, const std::shared_ptr<Buffers::VKBuffer>& staging) const
238{
239 if (layer >= m_data.size()) {
240 MF_ERROR(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
241 "TextureContainer::to_image layer {} out of range ({})", layer, m_data.size());
242 return nullptr;

278std::shared_ptr<Core::VKImage> TextureContainer::cached_array_image(
279 const std::shared_ptr<Buffers::VKBuffer>& staging) const
280{
281 const auto n = static_cast<uint32_t>(m_data.size());
282 if (n == 0)
283 return nullptr;
284
285 if (n == 1)
286 return cached_layer_image(0, staging);
287
288 const size_t layer_bytes = byte_size();
289 std::vector<uint8_t> combined(layer_bytes * n);
290 for (uint32_t i = 0; i < n; ++i) {
291 bool valid = false;
292 const bool copied = seqlock_read_void(m_slot_locks[i], 8, [&] {
293 auto [ptr, bytes] = variant_bytes(m_data[i]);
294 valid = ptr && bytes == layer_bytes;
295 if (valid)
296 std::memcpy(combined.data() + i * layer_bytes, ptr, layer_bytes);
297 });
298 if (!copied || !valid) {
299 MF_ERROR(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
300 "TextureContainer::to_image_array layer {} has invalid pixel data", i);
301 return nullptr;
302 }
303 }
304
305 const Portal::Graphics::ImageKey key {
306 .width = m_width, .height = m_height, .layers = n, .format = m_format, .kind = Portal::Graphics::ImageKey::Kind::SAMPLED_ARRAY
307 };
308 Memory::SerializedSeqlockWriteGuard image_guard(m_image_cache_lock);
309 return TextureLoom::instance().refresh_cached_image(
310 m_array_image_cache, key, combined, staging);
311}

338bool TextureContainer::upload_image_array(
339 const std::shared_ptr<Core::VKImage>& image,
340 const std::shared_ptr<Buffers::VKBuffer>& staging) const
341{
342 const auto n = static_cast<uint32_t>(m_data.size());

544std::vector<DataVariant> TextureContainer::get_region_data(const Region& region) const
545{
546 if (m_data.empty())
547 return {};
548
549 const size_t layer = (m_data.size() > 1 && !region.start_coordinates.empty())
550 ? static_cast<size_t>(region.start_coordinates[0])
551 : 0;
552
553 if (layer >= m_data.size())
554 return {};
555
556 std::optional<std::vector<DataVariant>> result;
557 seqlock_read_void(m_slot_locks[layer], 8, [&] {
558 result = std::visit(
559 [&](const auto& vec) -> std::vector<DataVariant> {
560 using T = typename std::decay_t<decltype(vec)>::value_type;
561 if constexpr (std::is_same_v<T, uint8_t>
562 || std::is_same_v<T, uint16_t>
563 || std::is_same_v<T, float>) {
564 auto extracted = extract_region_data<T>(
565 std::span<const T>(vec.data(), vec.size()),
566 region,
567 m_structure.dimensions);
568 return { DataVariant(std::move(extracted)) };
569 } else {
570 return {};
571 }
572 },
573 m_data[layer]);
574 });
575 return result.value_or(std::vector<DataVariant> {});
576}

591void TextureContainer::set_region_data(
592 const Region& region, const std::vector<DataVariant>& data)
593{
594 if (data.empty())
595 return;
596 if (region.start_coordinates.size() < 2 || region.end_coordinates.size() < 2)
597 return;
598
599 const size_t layer = (m_data.size() > 1 && !region.start_coordinates.empty())
600 ? static_cast<size_t>(region.start_coordinates[0])
601 : 0;
602
603 if (layer >= m_data.size())
604 return;
605
606 const size_t coord_offset = (m_data.size() > 1) ? 1 : 0;
607 const uint64_t y0 = region.start_coordinates[coord_offset];
608 const uint64_t x0 = region.start_coordinates[coord_offset + 1];
609 const uint64_t y1 = std::min(region.end_coordinates[coord_offset], static_cast<uint64_t>(m_height - 1));
610 const uint64_t x1 = std::min(region.end_coordinates[coord_offset + 1], static_cast<uint64_t>(m_width - 1));
611
612 Memory::SeqlockWriteGuard g(m_slot_locks[layer]);
613 std::visit(
614 [&](auto& dst_vec) {
615 using T = typename std::decay_t<decltype(dst_vec)>::value_type;
616 if constexpr (std::is_same_v<T, uint8_t>
617 || std::is_same_v<T, uint16_t>
618 || std::is_same_v<T, float>) {
619 const auto* src = std::get_if<std::vector<T>>(&data[0]);
620 if (!src || src->empty()) {
621 MF_ERROR(Journal::Component::Kakshya, Journal::Context::ContainerProcessing,
622 "TextureContainer::set_region_data source variant does not match "
623 "container element type");
624 return;
625 }
626 size_t src_idx = 0;
627 for (uint64_t y = y0; y <= y1 && src_idx < src->size(); ++y) {

830auto TextureContainer::get_frame_typed(uint64_t frame_index) const -> DataSpanVariant
831{
832 if (frame_index >= m_data.size()) {
833 return { std::span<const uint8_t> {} };
834 }
835
836 const size_t layer_elems = static_cast<size_t>(m_width) * m_height * m_channels;
837 DataSpanVariant out { std::span<const uint8_t> {} };
838 seqlock_read_void(m_slot_locks[frame_index], 8, [&] {
839 out = std::visit(
840 [&](const auto& vec) -> DataSpanVariant {
841 using T = typename std::decay_t<decltype(vec)>::value_type;
842 if constexpr (std::is_same_v<T, uint8_t> || std::is_same_v<T, uint16_t> || std::is_same_v<T, float>) {
843 const size_t nn = std::min(layer_elems, vec.size());
844 return DataSpanVariant(std::span<const T>(vec.data(), nn));
845 } else {
846 return { std::span<const uint8_t> {} };
847 }
848 },
849 m_data[frame_index]);
850 });
851 return out;
852}

854void TextureContainer::get_frames_typed(
855 void* output,
856 size_t count,
857 uint64_t start_frame,
858 uint64_t num_frames,
859 const std::type_info& type) const
860{
861 if (type == typeid(uint8_t)) {
862 get_frames_typed_as<uint8_t>(std::span<uint8_t>(static_cast<uint8_t*>(output), count), start_frame, num_frames);
863 return;
864 }
865 if (type == typeid(uint16_t)) {
866 get_frames_typed_as<uint16_t>(std::span<uint16_t>(static_cast<uint16_t*>(output), count), start_frame, num_frames);
867 return;
868 }
869 if (type == typeid(float)) {
870 get_frames_typed_as<float>(std::span<float>(static_cast<float*>(output), count), start_frame, num_frames);
871 return;
872 }
873
874 error<std::runtime_error>(
875 Journal::Component::Kakshya,
876 Journal::Context::Runtime,
877 std::source_location::current(),
878 "TextureContainer supports only uint8_t, uint16_t, and float for typed frame extraction");
879}

881template <typename T>
882auto TextureContainer::get_frame_typed_as(uint64_t frame_index) const -> std::span<const T>
883{
884 if (frame_index >= m_data.size())
885 return {};
886
887 std::span<const T> result;
888 seqlock_read_void(m_slot_locks[frame_index], 8, [&] {
889 const auto* vec = std::get_if<std::vector<T>>(&m_data[frame_index]);
890 if (!vec || vec->empty())
891 return;
892 const size_t layer_elems = static_cast<size_t>(m_width) * m_height * m_channels;
893 const size_t n = std::min(layer_elems, vec->size());
894 result = std::span<const T>(vec->data(), n);
895 });
896
897 return result;
898}

900template <typename T>
901void TextureContainer::get_frames_typed_as(std::span<T> output, uint64_t start_frame, uint64_t num_frames) const
902{
903 const size_t layer_elems = static_cast<size_t>(m_width) * m_height * m_channels;
904 size_t out_idx = 0;
905
906 for (uint64_t layer = start_frame;
907 layer < start_frame + num_frames && layer < m_data.size() && out_idx < output.size();
908 ++layer) {
909 seqlock_read_void(m_slot_locks[layer], 8, [&] {
910 const auto* vec = std::get_if<std::vector<T>>(&m_data[layer]);
911 if (!vec || vec->empty())
912 return;
913 const size_t copy_n = std::min(layer_elems, std::min(vec->size(), output.size() - out_idx));
914 std::copy_n(vec->begin(), static_cast<std::ptrdiff_t>(copy_n), output.begin() + static_cast<std::ptrdiff_t>(out_idx));
915 out_idx += copy_n;
916 });
917 }
918
919 if (out_idx < output.size()) {
920 std::fill(output.begin() + static_cast<std::ptrdiff_t>(out_idx), output.end(), T {});
921 }
922}
