111bool HookList::mutate(const std::function<bool(std::vector<HookEntry>&)>& edit)
112{
113 for (;;) {
114 m_readers.fetch_add(1);
115
116 const std::vector<HookEntry>* current = m_head.load();
117
118 auto next = std::make_unique<std::vector<HookEntry>>();
119 if (current) {
120 *next = *current;
121 }
122
123 const bool changed = edit(*next);
124 bool swapped = false;
125 if (changed) {
126 const std::vector<HookEntry>* replacement = next->empty() ? nullptr : next.get();
127 const std::vector<HookEntry>* expected = current;
128 swapped = m_head.compare_exchange_strong(expected, replacement);
129 if (swapped && replacement) {
130 next.release();
131 }
132 }
133
134 m_readers.fetch_sub(1);
135
136 if (!changed) {
137 return false;
138 }
139 if (swapped) {
140 if (current) {
141 retire(current);
142 }
143 reclaim();
144 return true;
145 }
146 }
147}

338void update_routing_state(RoutingState& state)
339{
340 state.cycles_elapsed++;
341
342 double progress = 0.;
343 if (state.fade_cycles > 0) {
344 progress = std::min<double>(1.0, static_cast<double>(state.cycles_elapsed) / state.fade_cycles);
345 } else {
346 for (uint32_t ch = 0; ch < 32; ++ch) {
347 bool will_be_in = state.to_channels & (1 << ch);
348 state.amount[ch] = will_be_in ? 1.0 : 0.0;
349 }
350 state.phase = RoutingState::NONE;
351 return;
352 }
353
354 for (uint32_t ch = 0; ch < 32; ch++) {
355 bool was_in = state.from_channels & (1 << ch);
356 bool will_be_in = state.to_channels & (1 << ch);
357
358 if (was_in && will_be_in) {
359 state.amount[ch] = 1.0;
360 } else if (was_in && !will_be_in) {
361 state.amount[ch] = 1.0 - progress;
362 } else if (!was_in && will_be_in) {
363 state.amount[ch] = progress;
364 } else {
365 state.amount[ch] = 0.0;
366 }
367 }
368
369 if (state.cycles_elapsed >= state.fade_cycles) {
370 state.phase = RoutingState::COMPLETED;
371 }
372}
