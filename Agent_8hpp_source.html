15/**
16 * @class Agent
17 * @brief Object that both perceives nearby entities and acts on MayaFlux objects.
18 *
19 * Constructed with a query radius and, optionally, a perception function and an
20 * influence function. Position is optional: call @c set_position before registering with
21 * @c Fabric if spatial behaviour is required. Without a position the spatial
22 * index is not consulted and @c spatial_results will be empty on each commit.
23 *
24 * On each commit the perception function is invoked first, then the influence
25 * function. Both receive contexts populated from the same spatial snapshot.
26 *
27 * The id is assigned by @c Fabric::wire and is stable for the object's lifetime.
28 *
29 * RenderConfig and AttachConfig take designated initializers.
30 *
31 * @code
32 * agent->attach(buf, { .on_move = AttachConfig::OnMove::Carry });
33 * agent->render(mgr, { .target_window = window });
34 * @endcode
35 */

187 * @brief Make an existing, already rendered buffer the look of this object.
188 *
189 * The buffer is neither registered nor written to. Its geometry is read
190 * once and the object takes its position from the centroid, or from the
191 * vertex at config.index, plus config.offset. Every render processor the
192 * buffer has is placed at the object's position and orientation. If the
193 * object is moved from outside, config.on_move decides whether the move
194 * becomes the new offset or carries the buffer. If the object already has

218 /**
219 * @brief Add an influence of any type, with as many targets as are wanted.
220 * @tparam T Type the producer returns; whatever the targets take.
221 * @param producer Builds the value from the influence context.
222 * @return The influence, to add its targets with Influence::add_target.
223 *
224 * Runs on every influence after the influence function. Typed influences
225 * are runtime objects and are not part of encoded state.
226 */
227 template <typename T>
228 std::shared_ptr<Influence<T>> add_influence(typename Influence<T>::Producer producer)
229 {
230 auto influence = std::make_shared<Influence<T>>(std::move(producer));
231 m_influences.emplace_back(influence,
232 [influence](const InfluenceContext& ctx) { influence->invoke(ctx); });
233 return influence;
234 }

236 /** @brief Stop an influence added with add_influence(). */

239 {

241 [&influence](const auto& entry) { return entry.first == influence; });
242 }

248 * @brief Add a perception of any type, with as many targets as are wanted.

251 * @return The perception, to add its targets with Perception::add_target.

347 /**
348 * @brief Invoke the perception function with the supplied context.
349 * @param ctx Populated context for this commit.
350 */
351 virtual void invoke_perception(const PerceptionContext& ctx)
352 {
353 for (const auto& [handle, perception] : m_perceptions) {
354 perception(ctx);
355 }
356 if (m_perception_fn) {
357 m_perception_fn(ctx);
358 }
359 }
360
361 /**
362 * @brief Invoke the influence function with the supplied context.
363 * @param ctx Populated context for this commit.
364 */
365 virtual void invoke_influence(const InfluenceContext& ctx) const
366 {
367 if (m_influence_fn) {
368 m_influence_fn(ctx);
369 }
370 dispatch_audio_sinks(m_audio_sinks, ctx);
371 dispatch_render_sinks(m_render_sinks, ctx);
372 if (m_attachment)
373 apply_attachment(*m_attachment, ctx);
374 for (const auto& [handle, influence] : m_influences) {
375 influence(ctx);
376 }
377 }
