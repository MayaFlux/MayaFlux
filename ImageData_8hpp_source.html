42/**
43 * @struct ImageData
44 * @brief Raw image data loaded from file or read back from a GPU image.
45 *
46 * Pixel storage is a variant over uint8, uint16, and float, chosen by the
47 * loader based on the source format. 8-bit formats (PNG, JPG, BMP, TGA)
48 * populate the uint8 variant. 16-bit PNG populates the uint16 variant.
49 * Floating-point formats (EXR, HDR) populate the float variant.
50 *
51 * The declared ImageFormat must match the active variant. Accessors below
52 * enforce this; direct member access is permitted but callers are
53 * responsible for consistency.
54 */
55struct ImageData {
56 using PixelStorage = std::variant<
57 std::vector<uint8_t>,
58 std::vector<uint16_t>,
59 std::vector<float>>;
60
61 PixelStorage pixels;
62 uint32_t width { 0 };
63 uint32_t height { 0 };
64 uint32_t channels { 0 };
65 ImageFormat format { ImageFormat::RGBA8 };
66
67 /**
68 * @brief Total byte size of pixel storage, dispatched on variant.
69 */
70 [[nodiscard]] size_t byte_size() const
71 {
72 return std::visit(
73 [](const auto& vec) { return vec.size() * sizeof(typename std::decay_t<decltype(vec)>::value_type); },
74 pixels);
75 }
76
77 /**
78 * @brief Raw data pointer, dispatched on variant. For upload paths.
79 */
80 [[nodiscard]] const void* data() const
81 {
82 return std::visit(
83 [](const auto& vec) -> const void* { return vec.data(); },
84 pixels);
85 }
86
87 /**
88 * @brief Writable raw data pointer, dispatched on variant. For download paths.
89 */
90 [[nodiscard]] void* data()
91 {
92 return std::visit(

94 pixels);
95 }
