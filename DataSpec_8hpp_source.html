59 * Valid types:
60 * - Kakshya::DataVariant

65 * - Kakshya::Region
66 * - Kakshya::RegionGroup
67 * - std::vector<Kakshya::RegionSegment>
68 * - Any Eigen matrix type (any scalar)
69 * - Any VariantVector (std::vector<T> where T is constructible into DataVariant)
70 * - Any type constructible from Kakshya::DataVariant

78 std::same_as<T, std::shared_ptr<Kakshya::SignalSourceContainer>> ||
79 std::same_as<T, Kakshya::Region> ||
80 std::same_as<T, Kakshya::RegionGroup> ||

287/**
288 * @struct extraction_traits
289 * @brief Native-type aware extraction traits.
290 *
291 * extraction_traits_d and its aliases are unchanged; this struct is purely
292 * additive. Callers that need native access (GPU staging, container
293 * roundtrips) use extraction_traits<T> or the native_result_t / native_element_t
294 * convenience aliases.
295 */
296template <typename T>
297struct extraction_traits {
298 static constexpr bool is_multi_variant = extraction_traits_d<T>::is_multi_variant;
299 static constexpr bool requires_container = extraction_traits_d<T>::requires_container;
300 static constexpr bool is_region_like = extraction_traits_d<T>::is_region_like;
301
302 using native_element_type = typename detail::native_element<T>::type;
303 using algorithm_result_type = typename extraction_traits_d<T>::result_type;
304
305 using native_result_type = std::conditional_t<
306 std::is_same_v<native_element_type, void>,
307 algorithm_result_type,
308 std::conditional_t<
309 extraction_traits_d<T>::is_multi_variant,
310 std::vector<std::span<native_element_type>>,
311 std::span<native_element_type>>>;
312};
