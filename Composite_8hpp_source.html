23/**
24 * @class CompositeLayout
25 * @brief Shared field description and byte layout for a CompositeArray.
26 *
27 * Declare fields before constructing an array. Construction finalizes the
28 * layout, placing a presence bitmap before the fields in each element.

30 * std::string; text uses the array's separate UTF-8 byte storage.

32 * Usage:
33 * @code

36 * layout.add_field<uint32_t>("age");
37 * layout.add_field<double>("score");

39 * @endcode
40 *
41 * @note stride_bytes() and field offsets are final only after construction
42 * of a CompositeArray from this layout.
43 */

45public:
46 /**
47 * @brief Declare a named arithmetic or UTF-8 text field.
48 * @tparam T Exact stored type; use std::string for text.

51 * the layout has already been finalized.

306template <typename T>
307 requires ArithmeticData<T>
308std::optional<T> Composite::get(std::string_view field_name) const noexcept
309{
310 const auto field_index = m_layout->find_field(field_name);
311 if (!field_index || !has(field_name))
312 return std::nullopt;
313
314 const auto& field = m_layout->fields()[*field_index];
315 if (field.type != std::type_index(typeid(T)))
316 return std::nullopt;
317
318 const auto& rows = std::get<std::vector<uint8_t>>(*m_rows);
319 T value {};
320
321 std::memcpy(&value,
322 rows.data() + m_index * m_layout->stride_bytes() + field.offset_bytes,
323 sizeof(T));
324
325 return value;
326}
