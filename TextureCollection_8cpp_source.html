18 vk::Rect2D fitted(uint32_t src_w, uint32_t src_h, uint32_t dst_w, uint32_t dst_h)
19 {
20 const double scale = std::min(
21 static_cast<double>(dst_w) / static_cast<double>(src_w),
22 static_cast<double>(dst_h) / static_cast<double>(src_h));
23
24 const auto w = std::clamp(static_cast<uint32_t>(std::lround(src_w * scale)), 1U, dst_w);
25 const auto h = std::clamp(static_cast<uint32_t>(std::lround(src_h * scale)), 1U, dst_h);
26
27 return {
28 vk::Offset2D { static_cast<int32_t>((dst_w - w) / 2), static_cast<int32_t>((dst_h - h) / 2) },
29 vk::Extent2D { w, h }
30 };
31 }
