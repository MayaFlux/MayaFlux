30/**
31 * @struct CameraConfig
32 * @brief Configuration for opening a camera device via FFmpeg.
33 *
34 * Platform device string conventions:
35 * - Linux: "/dev/video0", "/dev/video1", …
36 * - macOS: "0" (AVFoundation device index) or "FaceTime HD Camera"
37 * - Windows: "video=Integrated Camera" (DirectShow filter name)
38 *
39 * Resolution and frame rate values are hints passed to the device driver
40 * via AVDictionary options. The device may negotiate different parameters;
41 * the actual negotiated values are available from FFmpegCameraReader after
42 * open().
43 *
44 * This type is FFmpeg-specific by design and does not appear anywhere in
45 * CameraSource. Another backend defines its own config type entirely.
46 */
47struct MAYAFLUX_API CameraConfig {
48 std::string device_name; ///< Platform device string.
49 uint32_t target_width { 1920 }; ///< Requested width in pixels.
50 uint32_t target_height { 1080 }; ///< Requested height in pixels.
51 double target_fps { 30.0 }; ///< Hint only; device may ignore.
52 std::string format_override; ///< Leave empty to use CAMERA_FORMAT for current platform.
53 int pixel_format { -1 }; ///< Target AVPixelFormat as int; negative selects AV_PIX_FMT_RGBA.
54};
