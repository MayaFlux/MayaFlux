8/**
9 * @class Weights
10 * @brief Rows-by-columns table of doubles with atomic cell access
11 *
12 * A plain numeric table with no knowledge of what its rows and columns mean.
13 * Transition probabilities, source-to-output weights, and neighbour strengths
14 * are the same structure read by different code.
15 *
16 * Cells are read and written with relaxed atomics, so any thread can write a
17 * cell while another reads it. Row helpers and row views are not atomic and
18 * belong on the thread that consumes the table. Copying is not synchronised.
19 * Dimensions change only through assign(), which must not run concurrently
20 * with any other access.
21 *
22 * weighted_pick() turns a row into a single stochastic choice, drawing from
23 * a Stochastic instance the caller supplies and owns (one instance per
24 * thread, matching Stochastic's own contract).
25 *
26 * ## Usage
27 *
28 * ```cpp
29 * Weights transitions(5, 5);
30 * transitions.normalize_row(0);
31 *
32 * Stochastic draw(Algorithm::UNIFORM);
33 * size_t next = transitions.weighted_pick(0, draw);
34 * ```
35 */
36class Weights {

185 /**
186 * @brief Every cell uniform except the diagonal, each row normalized
187 * to sum to one
188 * @note Assumes a square table. A row beyond the column count has no
189 * diagonal cell to zero and is filled uniformly as-is.
190 *
191 * The common starting point for a transition table that should never
192 * pick its own row: every other cell equally likely, before any
193 * further shaping.
194 */
195 void uniform_excluding_self() noexcept
196 {

206 /**
207 * @brief Write an adjacency edge list into this table
208 * @param edges Index pairs, the same shape Kinesis::EdgeList is
209 * (Spatial::ProximityGraphs' sequential_chain(),
210 * k_nearest_neighbors(), minimum_spanning_tree(), and so on
211 * all return one)
212 * @param weight Value written at each edge
213 * @param symmetric When true (the default), also writes the reverse
214 * direction for every edge, matching an undirected graph. Some
215 * EdgeList producers (k_nearest_neighbors, nearest_neighbor_graph)
216 * are already directed; pass false for those.
217 * @param normalize When true (the default), every row touched by an
218 * edge is normalized afterward, so the result is directly
219 * usable as a transition table.
220 * @note An index outside this table's bounds is skipped rather than
221 * trusted, since edges typically comes from an external
222 * computation over a possibly different point count.
223 */
224 void apply_edges(std::span<const std::pair<size_t, size_t>> edges,
225 double weight = 1.0, bool symmetric = true, bool normalize = true) noexcept
226 {
227 for (const auto& [a, b] : edges) {
228 if (a < m_rows && b < m_cols) {
229 (*this)(a, b) = weight;
230 }
231 if (symmetric && b < m_rows && a < m_cols) {
232 (*this)(b, a) = weight;
233 }
234 }
235 if (normalize) {
236 for (size_t i = 0; i < m_rows; ++i) {
237 normalize_row(i);
238 }
239 }
240 }

242 /**
243 * @brief Weighted random draw over a row of nonnegative weights
244 * @param weights Row to draw from. Negative entries are treated as zero.
245 * @param generator Entropy source for the draw
246 * @return Index into weights, or weights.size() when every weight is
247 * zero or negative and no draw is possible
248 *
249 * Does not require a Weights instance or normalized weights; any span
250 * of nonnegative values is a valid distribution to draw from.
251 */
252 [[nodiscard]] static size_t weighted_pick(std::span<const double> weights, Stochastic& generator)
253 {
254 double total = 0.0;
255 for (double weight : weights) {
256 total += std::max(0.0, weight);
257 }
258
259 if (total <= 0.0) {
260 return weights.size();
261 }
262
263 const double target = generator(0.0, total);
264 double running = 0.0;
265 for (size_t i = 0; i < weights.size(); ++i) {
266 running += std::max(0.0, weights[i]);
267 if (target <= running) {
268 return i;
269 }
270 }
271 return weights.size() - 1;
272 }

277 * @param generator Entropy source for the draw
278 * @return Index into the row, or cols() when every weight in the row
279 * is zero or negative
280 */
281 [[nodiscard]] size_t weighted_pick(size_t row_index, Stochastic& generator) const
282 {
283 return weighted_pick(row(row_index), generator);
