110void TopologyGeneratorNode::write_path_attributes(
111 std::span<const LineVertex> points,
112 size_t num_points)
113{
114 if (num_points < 2 || m_vertices.empty()) {
115 return;
116 }
117
118 const size_t num_segments = num_points - 1;
119 const size_t count = m_vertices.size() / 2 + 1;
120 const auto span = static_cast<float>(count - 1);
121
122 for (size_t i = 0; i + 1 < count; ++i) {
123 const float t0 = static_cast<float>(i) / span;
124 const float t1 = static_cast<float>(i + 1) / span;
125
126 const auto s0 = std::min<size_t>(
127 static_cast<size_t>(t0 * static_cast<float>(num_segments)), num_segments - 1);
128 const auto s1 = std::min<size_t>(
129 static_cast<size_t>(t1 * static_cast<float>(num_segments)), num_segments - 1);
130
131 LineVertex& v0 = m_vertices[i * 2];
132 LineVertex& v1 = m_vertices[i * 2 + 1];
133
134 v0.color = m_force_uniform_color ? m_line_color : points[s0].color;
135 v1.color = m_force_uniform_color ? m_line_color : points[s1].color;
136
137 v0.thickness = m_force_uniform_thickness ? m_line_thickness : points[s0].thickness;
138 v1.thickness = m_force_uniform_thickness ? m_line_thickness : points[s1].thickness;
139 }

394void TopologyGeneratorNode::build_interpolated_path(
395 std::span<LineVertex> points,
396 size_t num_points)
397{
398 m_evaluator.configure(m_path_interpolation_mode, 0.5);
399
400 m_control_scratch.resize(num_points * 3);
401 for (size_t i = 0; i < num_points; ++i) {
402 m_control_scratch[i * 3 + 0] = points[i].position.x;
403 m_control_scratch[i * 3 + 1] = points[i].position.y;
404 m_control_scratch[i * 3 + 2] = points[i].position.z;
405 }
406
407 const size_t num_segments = num_points - 1;
408 const auto total_samples = static_cast<Eigen::Index>(
409 1 + num_segments * (m_samples_per_segment - 1));
410
411 m_evaluator.evaluate_planar(m_control_scratch, 3, total_samples, m_curve_primary);
412
413 const std::vector<double>* curve = &m_curve_primary;
414
415 if (m_use_arc_length_reparameterization) {
416 m_evaluator.reparameterize_planar(m_curve_primary, 3,
417 total_samples, total_samples, m_curve_secondary);
418 curve = &m_curve_secondary;
419 }
420
421 const auto count = static_cast<size_t>(total_samples);
422 if (count < 2) {
423 return;
424 }
425
426 m_vertices.resize((count - 1) * 2);
427
428 const double* x = curve->data();
429 const double* y = x + count;
430 const double* z = y + count;
431
432 for (size_t i = 0; i + 1 < count; ++i) {
433 m_vertices[i * 2].position = {
434 static_cast<float>(x[i]), static_cast<float>(y[i]), static_cast<float>(z[i])
435 };
436 m_vertices[i * 2 + 1].position = {
437 static_cast<float>(x[i + 1]), static_cast<float>(y[i + 1]), static_cast<float>(z[i + 1])
438 };
439 }
440
441 write_path_attributes(points, num_points);
442}

444void TopologyGeneratorNode::build_direct_connections(
445 std::span<LineVertex> points,
446 size_t num_points)
447{
448 size_t valid_connections = std::ranges::count_if(m_connections,
449 [num_points](const auto& conn) {
450 return conn.first < num_points && conn.second < num_points;
451 });
452
453 m_vertices.clear();
454 m_vertices.reserve(valid_connections * 2);
455
456 for (const auto& [a, b] : m_connections) {
457 if (a >= num_points || b >= num_points) {
458 continue;
459 }
460
461 glm::vec3 color_a = m_force_uniform_color ? m_line_color : points[a].color;
462 glm::vec3 color_b = m_force_uniform_color ? m_line_color : points[b].color;
463
464 float thick_a = m_force_uniform_thickness ? m_line_thickness : points[a].thickness;
465 float thick_b = m_force_uniform_thickness ? m_line_thickness : points[b].thickness;
466
467 m_vertices.emplace_back(LineVertex {
468 .position = points[a].position,
469 .color = color_a,
470 .thickness = thick_a });
471
472 m_vertices.emplace_back(LineVertex {
473 .position = points[b].position,
474 .color = color_b,
475 .thickness = thick_b });
476 }
477}
