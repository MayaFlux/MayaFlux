53void DataWriteProcessor::write_buffer(std::shared_ptr<VKBuffer> target, Kakshya::DataVariant data, size_t dst_offset)
54{
55 auto payload = std::make_shared<Kakshya::DataVariant>(std::move(data));
56 enqueue_write([target = std::move(target), payload, dst_offset](DataWriteProcessor& self, const std::shared_ptr<VKBuffer>&) {
57 const auto bytes = Kakshya::convert_variant<uint8_t>(*payload);
58 if (!target || !target->is_initialized() || bytes.size_bytes() == 0
59 || dst_offset + bytes.size_bytes() > target->get_size_bytes()) {
60 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,

62 return;
63 }

65 ensure_staging(self.m_write_staging, bytes.size_bytes());
66 }
67 upload_to_gpu(bytes.data(), bytes.size_bytes(), target, self.m_write_staging, dst_offset);

71void DataWriteProcessor::write_field(std::string name, Kakshya::DataVariant data)
72{

74 enqueue_write([name = std::move(name), payload](DataWriteProcessor& self, const std::shared_ptr<VKBuffer>& vk) {

76 const size_t size = bytes.size_bytes();

322void DataWriteProcessor::processing_function(const std::shared_ptr<Buffer>& buffer)
323{
324 auto vk = std::dynamic_pointer_cast<VKBuffer>(buffer);
325 if (!vk) {
326 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
327 "DataWriteProcessor attached to non-VKBuffer");
328 return;
329 }
330
331 while (auto write = m_writes->pop()) {
332 (*write)(*this, vk);
333 }
334
335 if (m_texture_dirty.test(std::memory_order_acquire)) {
336 m_texture_dirty.clear(std::memory_order_release);
337 if (m_pending_texture) {
338 if (auto rp = vk->get_render_processor()) {
339 rp->bind_texture(m_pending_texture->binding, m_pending_texture->image);
340 }
341 m_pending_texture.reset();
342 }
343 }
344
345 if (m_pixel_dirty.test(std::memory_order_acquire)) {
346 m_pixel_dirty.clear(std::memory_order_release);
347 std::swap(m_pixel_active, m_pixel_pending);
348 upload_texture(vk, m_pixel_active);
349 }
350
351 if (m_gpu_texture) {
352 if (auto rp = vk->get_render_processor()) {
353 if (!m_tex_binding_confirmed) {
354 rp->bind_texture(m_tex_binding, m_gpu_texture);
355 m_tex_binding_confirmed = true;
356 }
357 }
358 }
359
360 if (!m_data_dirty.test(std::memory_order_acquire)) {
361 return;
362 }
363
364 m_data_dirty.clear(std::memory_order_release);
365 std::swap(m_active, m_data_pending);
366
367 if (m_active.empty()) {
368 return;
369 }
370
371 upload_primary(vk, m_active);
372
373 for (size_t i = 1; i < m_active.size(); ++i) {
374 if (i == 1 && m_tex_width > 0 && is_vertex_modality(m_modality)) {
375 upload_texture(vk, m_active[i]);
376 } else {
377 upload_secondary(vk, m_active[i]);
378 }
379 }
380}

400void DataWriteProcessor::upload_vertex(const std::shared_ptr<VKBuffer>& vk, std::vector<Kakshya::DataVariant>& slots)
401{
402 std::optional<Kakshya::VertexAccess> access;
403
404 switch (m_topology) {
405 case Portal::Graphics::PrimitiveTopology::LINE_LIST:
406 case Portal::Graphics::PrimitiveTopology::LINE_STRIP:
407 access = Kakshya::as_line_vertex_access(slots);
408 break;
409 case Portal::Graphics::PrimitiveTopology::TRIANGLE_LIST:
410 case Portal::Graphics::PrimitiveTopology::TRIANGLE_STRIP:
411 access = Kakshya::as_mesh_vertex_access(slots);
412 break;
413 default:
414 access = Kakshya::as_point_vertex_access(slots);
415 break;
416 }
417
418 if (!access) {
419 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
420 "DataWriteProcessor: vertex access failed for modality {}",
421 Kakshya::modality_to_string(m_modality));
422 return;
423 }
424
425 ensure_capacity(vk, access->byte_count);
426 upload_to_gpu(access->data_ptr, access->byte_count, vk, m_staging);
427 vk->set_vertex_layout(access->layout);
428}

430void DataWriteProcessor::upload_texture(const std::shared_ptr<VKBuffer>& vk, Kakshya::DataVariant& slot)
431{

433 if (!access) {

435 "DataWriteProcessor: as_texture_access failed for modality {}",
436 Kakshya::modality_to_string(m_modality));
437 return;
438 }
439
440 m_last_texture_format = access->format;
441
442 if (m_tex_width == 0 || m_tex_height == 0) {

444 "DataWriteProcessor: setup_pixel_target() not called before texture upload");

446 }
447

462 }
