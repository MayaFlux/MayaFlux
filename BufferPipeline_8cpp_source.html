210void BufferPipeline::end()
211{
212 m_continuous_execution = false;
213
214 const auto routine = std::move(m_routine);
215 const auto self = std::move(m_active_self);
216
217 if (!routine) {
218 return;
219 }
220
221 const bool running = routine->is_active();
222
223 if (m_scheduler) {
224 m_scheduler->cancel_task(routine);
225 }
226
227 if (running && m_on_complete) {
228 m_on_complete();
229 }

252{
253 prepare_displays();
254
255 switch (m_execution_strategy) {
256 case ExecutionStrategy::PHASED:

259 case ExecutionStrategy::STREAMING:

261
262 case ExecutionStrategy::PARALLEL:

264
265 case ExecutionStrategy::REACTIVE:

267

271 std::source_location::current(),
