65 std::optional<std::vector<glm::vec3>> decode_positions(
66 std::span<const uint8_t> bytes,
67 const Kakshya::VertexLayout& layout)
68 {
69 std::vector<Kakshya::DataVariant> channels;
70 Kakshya::VertexInsertion insertion(channels);
71 if (!insertion.insert_interleaved(bytes, layout)) {
72 return std::nullopt;
73 }
74
75 const auto it = std::ranges::find_if(layout.attributes,
76 [](const auto& attribute) {
77 return attribute.component_modality == Kakshya::DataModality::VERTEX_POSITIONS_3D;
78 });
79 if (it == layout.attributes.end()) {
80 return std::nullopt;
81 }
82
83 auto* positions = std::get_if<std::vector<glm::vec3>>(
84 &channels.at(static_cast<size_t>(it - layout.attributes.begin())));
85 return positions ? std::optional<std::vector<glm::vec3>> { std::move(*positions) } : std::nullopt;

181 std::optional<Kinesis::AABB3D> instance_bounds(const std::shared_ptr<Buffers::InstanceNetworkBuffer>& buffer)
182 {
183 const auto network = buffer->get_network();
184 if (!network) {
185 return std::nullopt;
186 }
187
188 std::optional<Kinesis::AABB3D> total;
189 for (uint32_t i = 0; i < network->slot_count(); ++i) {
190 const auto& slot = network->get_slot(i);
191 const auto local = slot.node ? node_bounds(slot.node) : std::nullopt;
192 if (!local) {
193 continue;
194 }
195
196 for (int corner = 0; corner < 8; ++corner) {
197 const glm::vec3 point {
198 (corner & 1) != 0 ? local->max.x : local->min.x,
199 (corner & 2) != 0 ? local->max.y : local->min.y,
200 (corner & 4) != 0 ? local->max.z : local->min.z,
201 };
202 const glm::vec3 placed = glm::vec3(slot.transform * glm::vec4(point, 1.0F));
203 total = total
204 ? Kinesis::AABB3D { .min = glm::min(total->min, placed), .max = glm::max(total->max, placed) }
205 : Kinesis::AABB3D { .min = placed, .max = placed };
206 }
207 }
208 return total;
209 }

256std::optional<Kinesis::AABB3D> read_bounds(
257 const std::shared_ptr<Buffers::VKBuffer>& buf,
258 const std::optional<uint32_t>& collection)
259{
260 if (const auto volume = std::dynamic_pointer_cast<Buffers::VolumeGridBuffer>(buf)) {
261 return volume->get_bounds();
262 }
263
264 if (const auto network_buffer = std::dynamic_pointer_cast<Buffers::NetworkGeometryBuffer>(buf)) {
265 const auto cloud = network_cloud(network_buffer);
266 if (!cloud || cloud->positions.empty()) {
267 return std::nullopt;
268 }
269 if (!collection) {
270 return Kinesis::aabb(cloud->positions);
271 }
272
273 std::vector<glm::vec3> chosen;
274 const size_t count = std::min(cloud->positions.size(), cloud->clusters.size());
275 for (size_t i = 0; i < count; ++i) {
276 if (cloud->clusters.at(i) == *collection) {
277 chosen.push_back(cloud->positions.at(i));
278 }
279 }
280 return chosen.empty() ? std::nullopt : std::optional<Kinesis::AABB3D> { Kinesis::aabb(chosen) };
281 }
282
283 if (const auto geometry = std::dynamic_pointer_cast<Buffers::GeometryBuffer>(buf)) {
284 if (const auto cloud = node_cloud(geometry); cloud && !cloud->positions.empty()) {
285 return Kinesis::aabb(cloud->positions);
286 }
287 }
288
289 if (const auto instances = std::dynamic_pointer_cast<Buffers::InstanceNetworkBuffer>(buf)) {
290 return instance_bounds(instances);
291 }
292
293 const auto mesh = snapshot_mesh(buf);
294 const auto* vertices = mesh ? std::get_if<std::vector<uint8_t>>(&mesh->vertex_variant) : nullptr;
295 if (!vertices) {
296 return std::nullopt;
297 }
298
299 const auto positions = decode_positions(*vertices, mesh->layout);
300 return positions && !positions->empty() ? std::optional<Kinesis::AABB3D> { Kinesis::aabb(*positions) } : std::nullopt;
301}
