28/**
29 * @class Bridge
30 * @brief Two-way binding orchestrator for Forma elements.
31 *
32 * One Bridge instance serves the full application. It owns all inbound
33 * GraphicsRoutine tasks and all outbound FormaBindingsProcessor instances.
34 * Layer and Context are created via the Forma free functions and passed in
35 * — Bridge does not own or create them.
36 *
37 * Inbound (what drives an element's MappedState each frame):
38 * - bind(id, Node) — reads Node::get_last_output() via GraphicsRoutine
39 * - bind(id, lambda) — calls std::function<float()> via GraphicsRoutine
40 *
41 * Outbound (where the element's value goes each frame):
42 * - write(id, ShaderProcessor, offset) — push constant staging
43 * - write(id, descriptor_name, ...) — descriptor binding
44 * - write(id, AudioWriteProcessor) — audio buffer
45 * - write(id, DataWriteProcessor) — vertex buffer
46 * - write(id, Constant) — node graph value carrier
47 *
48 * All overloads also accept shared_ptr<MappedState<T>> in place of the
49 * element id. These resolve to the state->id.
50 * duplicated logic.
51 *
52 * Usage:
53 * @code
54 * auto [layer, ctx] = Forma::create_layer(window, "hud", event_manager);
55 * auto el = Forma::create_element<float>(*layer, window, bm, geom, 0.5f);
56 *
57 * Bridge bridge(scheduler, buffer_manager);
58 * bridge.bind(el.state, envelope);
59 * bridge.write(el.state, compute_proc, offsetof(PC, cutoff));
60 *
61 * ctx->on_press(el.element.id, IO::MouseButtons::LEFT,
62 * [](uint32_t, glm::vec2){});
63 * @endcode
64 */

324 /**
325 * @brief Register a MappedState<T> so that MappedState overloads and
326 * outbound bindings can resolve to the correct element id and reader.
327 *
328 * Called by Forma::create_element. The reader closes over the MappedState
329 * and is stored type-erased as std::function<float()>.
330 *
331 * @tparam T MappedState value type.
332 * @param state MappedState to register.
333 * @param id Element id from Layer::add.
334 * @param buffer FormaBuffer the element renders into.
335 * @param project Optional T -> float projection. Identity cast if empty.
336 */
337 template <typename T>
338 void register_element(
339 std::shared_ptr<MappedState<T>> state,
340 uint32_t id,
341 std::shared_ptr<Buffers::FormaBuffer> buffer,
342 std::function<float(T)> project = {})
343 {
344 std::function<float()> reader;
345 if (project) {
346 reader = [s = state, p = std::move(project)] {
347 return p(s->value);
348 };
349 } else if constexpr (std::is_convertible_v<T, float>) {
350 reader = [s = state] {
351 return static_cast<float>(s->value);
352 };
353 } else {
354 reader = [] { return 0.F; };
355 }
356
357 std::function<std::vector<float>()> bulk_reader;
358 if constexpr (std::is_same_v<T, std::vector<float>>) {
359 bulk_reader = [s = state] { return s->value; };
360 } else if constexpr (std::is_same_v<T, std::vector<double>>) {
361 bulk_reader = [s = state] {
362 std::vector<float> out;
363 out.reserve(s->value.size());
364 for (double v : s->value)
365 out.push_back(static_cast<float>(v));
366 return out;
367 };
368 }
369
370 std::function<void(float)> writer;
371 if constexpr (std::is_convertible_v<float, T>) {
372 writer = [s = state](float v) {
373 s->write(static_cast<T>(v));
374 };
375 } else {
376 writer = [](float) { };
377 }
378
379 state->id = id;
380 m_records[id] = ElementRecord {
381 .buffer = std::move(buffer),
382 .bindings = nullptr,
383 .reader = std::move(reader),
384 .bulk_reader = std::move(bulk_reader),
385 .writer = std::move(writer),
386 .inbound_task = {},
387 .outbound_tasks = {},
388 };
389 }
