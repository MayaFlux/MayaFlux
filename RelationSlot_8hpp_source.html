7/**
8 * @struct RelationSlot
9 * @brief One addressable participant of a RelationNetwork
10 *
11 * A plain record: an optional node, a level, an activity flag, and the last
12 * rendered block. Fields are public and operators read them directly; the
13 * few methods below exist only where a field needs more than a plain
14 * assignment (firing callbacks, enabling a node's hooks).
15 *
16 * @warning A slot's address is not stable: RelationNetwork::add_slot() may
17 * reallocate the whole slot vector. Never hold a pointer or
18 * reference to a RelationSlot past a process_batch() call; read a
19 * slot fresh through RelationNetwork::get_slot(index) or
20 * get_node_output(index) instead.
21 *
22 * Structural edits (assigning a node, registering callbacks) must happen
23 * outside process_batch().
24 *
25 * A plain aggregate: RelationNetwork::add_slot() takes one by value, so it
26 * can be built with a designated initializer instead of a set_node() call
27 * afterward.
28 *
29 * @code
30 * n->add_slot({ .name = "carrier", .node = vega.Sine(220.0, 1.0) });
31 * n->add_slot({ .name = "product" }); // no node
32 * @endcode
33 *
34 * @note .node set this way still gets its hooks enabled: add_slot()
35 * applies that regardless of whether the node arrived through
36 * set_node() or through the .node field directly.
37 */
38struct RelationSlot {
39 uint32_t index {};
40 std::string name;
41 std::shared_ptr<Node> node;
42 double level { 0.0 };
43 bool active { true };
44 std::vector<double> block;

46 std::vector<std::function<void()>> on_enter;

49 /**
50 * @brief Set this slot's node
51 * @param new_node Node whose output becomes this slot's signal. Owned by the slot.
52 *
53 * Enables the node's hooks (on_impulse, on_phase_wrap, on_tick, ...) so
54 * they fire while the slot is advanced. A node already belonging to
55 * another network stays silent until set_in_network(false) is called on
56 * it directly.
57 */
58 void set_node(std::shared_ptr<Node> new_node)
59 {
60 if (new_node) {
61 new_node->m_fire_events_during_snapshot = true;
62 }
63 node = std::move(new_node);
64 }

66 /**
67 * @brief Set this slot's activity, firing on_enter/on_leave on a real transition
68 */
69 void set_active(bool is_active)
70 {
71 if (active == is_active) {
72 return;
73 }
74 active = is_active;
75 for (const auto& callback : is_active ? on_enter : on_leave) {
76 if (callback) {
77 callback();
78 }
79 }
80 }
