62/**
63 * @brief Thin domain wrapper that adds subscript channel-binding syntax.
64 *
65 * Audio[0] and Audio[{0,1}] produce a CreationContext consumed immediately
66 * by operator|. Implicit conversion to Domain keeps all existing call sites
67 * that accept a plain Domain parameter working without change.
68 */
69struct DomainSpec {
70 Domain value;
71
72 /** @brief Bind a single channel to this domain. */
73 [[nodiscard]] CreationContext operator[](uint32_t ch) const
74 {
75 return { value, ch };
76 }
77
78 /** @brief Bind multiple channels to this domain. */
79 [[nodiscard]] CreationContext operator[](std::initializer_list<uint32_t> chs) const
80 {
81 return { value, std::vector<uint32_t>(chs) };
82 }
83
84 /** @brief Implicit decay to Domain for call sites that accept Domain directly. */
85 operator Domain() const { return value; }
86
87 /** @brief Implicit decay to CreationContext for bare ptr | Audio usage. */
88 operator CreationContext() const { return { value }; }
89};

411/**
412 * @brief Global Creator instance.
413 *
414 * Every call on vega is one of three kinds:
415 * - Class-named factories (Sine, AudioBuffer, ModalNetwork, TextureBuffer and
416 * the rest of the registry) construct the class with exactly its
417 * constructor arguments. The pipe registers the result in a domain.
418 * - read_* loaders take a source and return the real object. read_audio,
419 * read_image, read_file, read_mesh and read_mesh_network load from a path
420 * or, with no argument, from the matching choose_* call. read_file
421 * returns a CompositeContainer for supported structured formats.
422 * read_mesh returns a MeshGroupHandle that the pipe registers. read_hid,
423 * read_midi, read_osc, read_tablet and read_input create an input node and
424 * register it against a binding.
425 * - evolve and mint take a config and perform the whole sequence, returning
426 * the real object with rendering attached.
427 *
428 * @code
429 * auto wave = vega.Sine(440.f) | Audio[0];
430 * auto buf = vega.AudioBuffer(0, 512) | Audio[0];
431 * auto net = vega.ModalNetwork(16, 220.0) | Audio[{0, 1}];
432 * auto tex = vega.TextureBuffer(...) | Graphics;
433 *
434 * auto sfx = vega.read_audio("x.wav") | Audio;
435 * auto image = vega.read_image("x.png") | Graphics;
436 * auto records = vega.read_file("x.csv");
437 * auto meshes = vega.read_mesh("x.fbx") | Graphics;
438 * auto pad = vega.read_midi(config, binding);
439 *
440 * auto life = vega.evolve(grid_config, seed, { .target_window = window });
441 * auto smoke = vega.evolve(volume, flow_config, { .target_window = window });
442 * auto form = vega.mint(StructureConfig::Object {
443 * .mesh = mesh,
444 * .render = { .target_window = window } });
445 * @endcode
446 */
