12/**
13 * @struct GlyphMetrics
14 * @brief Per-glyph layout and UV data produced by GlyphAtlas.
15 *
16 * UV coordinates are in normalised [0, 1] atlas space.
17 * Bearing and size are in pixels at the atlas's declared pixel_size.
18 *
19 * face and glyph_index identify which FontFace actually produced this
20 * glyph (the primary face, or a fallback on a primary miss) and its glyph
21 * index within that specific face. Glyph indices are only meaningful
22 * relative to the face that produced them: a consumer doing anything
23 * face-sensitive (kerning) must compare face identity, not just index.
24 *
25 * When HarfBuzz is introduced it will supply glyph_index and position
26 * offsets directly; this struct remains the downstream currency.
27 */
28struct GlyphMetrics {
29 float uv_x0 { 0.F }; ///< Left UV edge in atlas texture.
30 float uv_y0 { 0.F }; ///< Top UV edge in atlas texture.
31 float uv_x1 { 0.F }; ///< Right UV edge in atlas texture.
32 float uv_y1 { 0.F }; ///< Bottom UV edge in atlas texture.
33
34 int32_t bearing_x { 0 }; ///< Horizontal bearing in pixels (from FT_GlyphSlot).
35 int32_t bearing_y { 0 }; ///< Vertical bearing in pixels (from FT_GlyphSlot).
36 uint32_t width { 0 }; ///< Glyph bitmap width in pixels.
37 uint32_t height { 0 }; ///< Glyph bitmap height in pixels.
38 int32_t advance_x { 0 }; ///< Horizontal advance in pixels (26.6 fixed-point >> 6).
39
40 const FontFace* face { nullptr }; ///< Face that produced this glyph (primary or fallback).
41 FT_UInt glyph_index { 0 }; ///< Glyph index within `face` (from FT_Get_Char_Index).
42};

44/**
45 * @class GlyphAtlas
46 * @brief Rasterizes and packs glyphs from a FontFace into a TextureContainer.
47 *
48 * One atlas corresponds to one (FontFace, pixel_size) pair. The atlas
49 * texture is R8 (single-channel coverage); colour is applied in the shader.
50 *
51 * Glyphs are rasterized on first request via get_or_rasterize() and packed
52 * into the atlas using a simple shelf packing algorithm. On shelf overflow
53 * the atlas texture doubles in size and every previously cached glyph is
54 * re-rasterized into the new texture from its owning face; rasterization
55 * never permanently refuses for lack of room.
56 *
57 * The primary face is fixed at construction. A codepoint the primary face
58 * lacks is looked up in the fallback chain (add_fallback()), in order, and
59 * the first face reporting a glyph produces it; a codepoint absent from
60 * every face fails as before. The atlas's cache is keyed on (face,
61 * glyph_index) rather than glyph_index alone, since glyph indices are only
62 * unique within the face that assigned them: the same numeric index from
63 * two different faces names two unrelated glyphs.
64 *
65 * Callers obtain glyph indices via FT_Get_Char_Index on the FontFace. This
66 * keeps the path open for HarfBuzz, which outputs glyph indices directly
67 * (against the primary face; HarfBuzz does not participate in fallback).
68 *
69 * Thread safety: not thread-safe. Rasterization must occur on the thread
70 * that owns the FontFace.
71 */
