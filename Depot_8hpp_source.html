3/**
4 * @file API/Depot.hpp
5 * @brief Audio file loading and container management API
6 *
7 * This header provides the public API for working with IOManager,
8 * container creation, and file type checking within the MayaFlux engine.
9 * It includes:
10 * - `create_container<ContainerType>(args...)`: Template function to create signal source containers.
11 * - `is_audio(filepath)`: Check if a file is an audio file based on its extension.
12 * - `is_image(filepath)`: Check if a file is an image file based on its extension.
13 * - `get_io_manager()`: Access the global IOManager instance for file loading and buffer management.
14 */

49/**
50 * @brief creates a new container of the specified type
51 * @tparam ContainerType Type of container to create (must be derived from SignalSourceContainer)
52 * @tparam Args Constructor argument types
53 * @param args Constructor arguments for the container
54 * @return Shared pointer to the created container
55 */
56template <typename ContainerType, typename... Args>
57 requires std::derived_from<ContainerType, Kakshya::SignalSourceContainer>
58auto create_container(Args&&... args) -> std::shared_ptr<ContainerType>
59{
60 return std::make_shared<ContainerType>(std::forward<Args>(args)...);
61}
