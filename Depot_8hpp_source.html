56/**
57 * @brief creates a new container of the specified type
58 * @tparam ContainerType Type of container to create (must be derived from SignalSourceContainer)
59 * @tparam Args Constructor argument types
60 * @param args Constructor arguments for the container
61 * @return Shared pointer to the created container
62 */

64 requires std::derived_from<ContainerType, Kakshya::SignalSourceContainer>
65auto create_container(Args&&... args) -> std::shared_ptr<ContainerType>
66{
67 return std::make_shared<ContainerType>(std::forward<Args>(args)...);
68}
