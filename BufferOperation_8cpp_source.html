47BufferOperation BufferOperation::capture_input(
48 const std::shared_ptr<Buffers::BufferManager>& buffer_manager,
49 uint32_t input_channel,
50 BufferCapture::CaptureMode mode,
51 uint32_t cycle_count)
52{
53 auto input_buffer = std::make_shared<Buffers::AudioBuffer>(input_channel);
54 buffer_manager->register_input_listener(input_buffer, input_channel);
55 buffer_manager->add_buffer(input_buffer, Buffers::ProcessingToken::AUDIO_BACKEND, input_channel);
56
57 BufferCapture capture(input_buffer, mode, cycle_count);
58 if (mode == BufferCapture::CaptureMode::ACCUMULATE && cycle_count == 0) {

61

593bool BufferOperation::is_capture_phase_operation(const BufferOperation& op)
594{
595 if (op.get_execution_phase() == BufferOperation::ExecutionPhase::CAPTURE) {
596 return true;
597 }
598 if (op.get_execution_phase() == BufferOperation::ExecutionPhase::PROCESS) {
599 return false;
600 }
601
602 switch (op.get_type()) {
603 case BufferOperation::OpType::CAPTURE:
604 return true;
605 case BufferOperation::OpType::MODIFY:
606 return op.is_streaming();
607 default:
608 return false;
609 }
610}

612bool BufferOperation::is_process_phase_operation(const BufferOperation& op)
613{
614 if (op.get_execution_phase() == BufferOperation::ExecutionPhase::PROCESS) {
615 return true;
616 }
617 if (op.get_execution_phase() == BufferOperation::ExecutionPhase::CAPTURE) {
618 return false;
619 }
620
621 switch (op.get_type()) {
622 case BufferOperation::OpType::MODIFY:
623 return !op.is_streaming();
624 case BufferOperation::OpType::TRANSFORM:
625 case BufferOperation::OpType::ROUTE:
626 case BufferOperation::OpType::LOAD:
627 case BufferOperation::OpType::DISPATCH:
628 case BufferOperation::OpType::FUSE:
629 return true;
630 case BufferOperation::OpType::CAPTURE:
631 case BufferOperation::OpType::CONDITION:
632 case BufferOperation::OpType::BRANCH:
633 case BufferOperation::OpType::SYNC:
634 default:
635 return false;
636 }
