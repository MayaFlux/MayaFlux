284std::unique_ptr<DataReadProcessor::PendingRead> DataReadProcessor::collect_result(ReadKind kind)
285{
286 if (!m_result_ready.load(std::memory_order_acquire)) {
287 return nullptr;
288 }
289
290 std::unique_ptr<PendingRead> result;
291 m_result.sweep(
292 [kind](std::unique_ptr<PendingRead>& pending) {
293 switch (kind) {
294 case ReadKind::Bytes:
295 return !pending->mesh;
296 case ReadKind::Vertices:
297 return pending->layout.has_value();
298 case ReadKind::Mesh:
299 return pending->mesh.has_value() || pending->triangle_soup;
300 case ReadKind::Pixels:
301 return pending->image.has_value();
302 }
303 return false;
304 },
305 [this, &result](std::unique_ptr<PendingRead>& pending) {
306 if (pending->transfer) {
307 resolve_back_buffer_read(
308 pending->transfer,
309 generation_slot(pending->handle),
310 pending->bytes.data(),
311 pending->bytes.size(),
312 m_staging);
313 }
314 result = std::move(pending);
315 });
316 if (result) {
317 m_result_ready.store(false, std::memory_order_release);
318 m_result_occupied.store(false, std::memory_order_release);
319 }
320 return result;
321}

416bool DataReadProcessor::submit_read(
417 PendingRead& pending,
418 std::shared_ptr<VKBuffer>& staging,
419 size_t available_bytes,
420 size_t byte_count,
421 bool mapped,
422 bool transferable)
423{
424 if (byte_count == 0) {
425 byte_count = available_bytes;
426 }
427
428 if (byte_count > available_bytes || !pending.handle.buffer) {
429 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
430 "DataReadProcessor requested {} bytes from a {} byte buffer",
431 byte_count, available_bytes);
432 return false;
433 }
434
435 pending.bytes.resize(byte_count);
436
437 if (byte_count != 0 && mapped && pending.handle.mapped_ptr) {
438 std::memcpy(pending.bytes.data(), pending.handle.mapped_ptr, byte_count);
439 } else if (byte_count != 0) {
440 if (!transferable) {
441 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
442 "DataReadProcessor source has no mapped memory or transfer-source usage");
443 return false;
444 }
445
446 pending.transfer = download_back_buffer_async(
447 generation_slot(pending.handle), byte_count, staging);
448
449 if (!pending.transfer) {
450 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
451 "DataReadProcessor failed to submit a device-local read");
452 return false;
453 }
454 }
455
456 return true;
457}

459void DataReadProcessor::processing_function(const std::shared_ptr<Buffer>& buffer)
460{
461 if (!is_compatible_with(buffer)) {
462 MF_RT_ERROR(Journal::Component::Buffers, Journal::Context::BufferProcessing,
463 "DataReadProcessor attached to an incompatible buffer");
464 return;
465 }
466
467 if (m_result_occupied.load(std::memory_order_acquire)) {
468 return;
469 }
470
471 const auto vk_buffer = std::static_pointer_cast<VKBuffer>(buffer);
472 if (!vk_buffer->is_initialized()) {
473 return;
474 }
475
476 std::optional<ReadRequest> request;
477 m_request.drain([&request](ReadRequest& next) {
478 if (!request) {
479 request.emplace(std::move(next));
480 }
481 });
482 if (!request) {
483 return;
484 }
485
486 auto pending = std::make_unique<PendingRead>();
487 pending->source = vk_buffer;
488
489 if (!request->fill(*this, vk_buffer, *pending)) {
490 return;
491 }
492
493 m_result_occupied.store(true, std::memory_order_release);
494 m_result.push(std::move(pending));
495 m_result_ready.store(true, std::memory_order_release);
496}
