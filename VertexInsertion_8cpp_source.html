61 const char* layout_fault(const VertexLayout& layout) noexcept
62 {
63 if (layout.stride_bytes == 0) {
64 return "layout stride_bytes is zero";
65 }
66 if (layout.attributes.empty()) {
67 return "layout has no attributes";
68 }
69 for (const auto& attr : layout.attributes) {
70 const size_t size = element_size(attr.component_modality);
71 if (size == 0) {
72 return "layout attribute has a modality with no vertex channel type";
73 }
74 if (static_cast<size_t>(attr.offset_in_vertex) + size > layout.stride_bytes) {
75 return "layout attribute extends past the stride";
76 }
77 }
78 return nullptr;
79 }

93bool VertexInsertion::insert_interleaved(
94 std::span<const uint8_t> vertex_bytes,
95 const VertexLayout& layout)
96{
97 if (const char* fault = layout_fault(layout)) {
98 MF_ERROR(Journal::Component::Kakshya, Journal::Context::Runtime,
99 "VertexInsertion::insert_interleaved: {}", fault);
100 return false;
101 }
102 if (vertex_bytes.empty() || vertex_bytes.size() % layout.stride_bytes != 0) {
103 MF_ERROR(Journal::Component::Kakshya, Journal::Context::Runtime,
104 "VertexInsertion::insert_interleaved: {} bytes is not a nonzero multiple of stride {}",
105 vertex_bytes.size(), layout.stride_bytes);
106 return false;
107 }
108
109 const size_t count = vertex_bytes.size() / layout.stride_bytes;
110
111 m_channels.clear();
112 m_channels.reserve(layout.attributes.size());
113 for (const auto& attr : layout.attributes) {
114 m_channels.push_back(gather_attribute(
115 attr.component_modality, vertex_bytes,
116 layout.stride_bytes, attr.offset_in_vertex, count));
117 }
118
119 m_layout = layout;
120 m_layout.vertex_count = static_cast<uint32_t>(count);
121
122 MF_DEBUG(Journal::Component::Kakshya, Journal::Context::Runtime,
123 "VertexInsertion::insert_interleaved: {} vertices into {} channels",
124 count, m_channels.size());
125 return true;
126}
