15 /**
16 * The engine is published through a raw atomic pointer and is freed only
17 * after End() has joined every thread that can call a global getter, so a
18 * getter is a single acquire load and never waits on a lifecycle change.
19 */
20 std::atomic<Core::Engine*> engine_ptr { nullptr };
21 std::atomic<bool> initialized { false };
22 std::atomic<bool> terminating { false };
23 std::atomic<bool> ending { false };
24 std::atomic<bool> atexit_registered { false };

27 void cleanup_engine()
28 {
29 if (cleaning_up) {
30 return;
31 }
32
33 if (ending.exchange(true, std::memory_order_acq_rel)) {
34 while (ending.load(std::memory_order_acquire)) {
35 std::this_thread::yield();
36 }
37 return;
38 }
39
40 cleaning_up = true;
41
42 if (std::unique_ptr<Core::Engine> engine { engine_ptr.load(std::memory_order_acquire) }) {
43 if (initialized.load(std::memory_order_acquire)) {
44 engine->End();
45 Journal::Archivist::shutdown();
46 }
47 cleanup_persistent_store();
48 initialized.store(false, std::memory_order_release);
49 engine_ptr.store(nullptr, std::memory_order_release);
50 }
51
52 cleaning_up = false;
53 ending.store(false, std::memory_order_release);
54 }

56 Core::Engine& get_or_create_engine()
57 {
58 Core::Engine* engine = engine_ptr.load(std::memory_order_acquire);
59 if (engine) {
60 return *engine;
61 }
62
63 auto candidate = std::make_unique<Core::Engine>();
64 Core::Engine* expected = nullptr;
65 if (engine_ptr.compare_exchange_strong(expected, candidate.get(),
66 std::memory_order_acq_rel, std::memory_order_acquire)) {
67 engine = candidate.release();
68 if (!atexit_registered.exchange(true, std::memory_order_acq_rel)) {
69 std::atexit(cleanup_engine);
70 }
71 terminating.store(false, std::memory_order_release);
72 initialized.store(true, std::memory_order_release);
73 } else {
74 engine = expected;
75 }
76 return *engine;
77 }
