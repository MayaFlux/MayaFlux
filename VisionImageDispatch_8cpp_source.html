10const std::vector<float>& VisionGpuExecutor::gaussian_kernel_2d(uint32_t radius, float sigma)
11{
12 static std::unordered_map<uint64_t, std::vector<float>> cache;
13 const uint64_t key = (static_cast<uint64_t>(std::bit_cast<uint32_t>(sigma)) << 32)
14 | radius;
15 auto it = cache.find(key);
16 if (it != cache.end())
17 return it->second;
18
19 const uint32_t diam = 2 * radius + 1;
20 std::vector<float> k(static_cast<size_t>(diam) * diam);
21 float sum = 0.0F;
22 for (uint32_t y = 0; y < diam; ++y) {
23 for (uint32_t x = 0; x < diam; ++x) {
24 const float fx = static_cast<float>(x) - static_cast<float>(radius);
25 const float fy = static_cast<float>(y) - static_cast<float>(radius);
26 const float v = std::exp(-(fx * fx + fy * fy) / (2.0F * sigma * sigma));
27 k[y * diam + x] = v;
28 sum += v;
29 }
30 }
31 for (auto& v : k)
32 v /= sum;
33
34 return cache.emplace(key, std::move(k)).first->second;
35}
