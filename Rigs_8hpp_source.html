43/**
44 * @brief Construct a built SamplingPipeline from an audio file.
45 *
46 * Loads the file into a DynamicSoundStream via SoundFileReader::load_bounded,
47 * constructs a SamplingPipeline with engine globals (BufferManager,
48 * TaskScheduler, buffer size), calls build(), and returns the result.
49 *
50 * The returned sampler is ready for play() and play_continuous() calls.
51 * Voice slots are allocated on demand via load().
52 *
53 * @param filepath Path to the audio file (any FFmpeg-supported format).
54 * @param num_samples Number of samples to load from the file (default: 48000 * 5).
55 * @param truncate Truncate stream to num_samples if true (default: true).
56 * @param channel Output channel index (default: 0).
57 * @param max_dur_ms Optional maximum duration to build the pipeline for (in milliseconds).
58 Defaults to 0 which is infinite (the pipeline will run until the sampler is destroyed).
59 * @return Built SamplingPipeline, or nullptr if the file could not be loaded.
60 *
61 * @code
62 * auto kick = MayaFlux::create_sampler("kick.wav");
63 * kick->play(0, kick->slice_from_stream());
64 *
65 * auto pad = MayaFlux::create_sampler("pad.wav");
66 * pad->load(0, pad->slice_from_stream()).speed = 0.5;
67 * pad->play_continuous(0);
68 * @endcode
69 */
70MAYAFLUX_API std::shared_ptr<Kriya::SamplingPipeline> create_sampler(
71 const std::string& filepath, uint32_t num_samples = 48000 * 5, bool truncate = true,
72 uint32_t channel = 0, uint64_t max_dur_ms = 0);

294 * @param source What to capture, for example from capture_file_from, capture_input_from or capture_from.
