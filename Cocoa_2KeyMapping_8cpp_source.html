143}
144
145int to_cocoa_key(IO::Keys key) noexcept
146{
147 if (key == IO::Keys::Unknown)
148 return -1;
149
150 for (std::size_t key_code = 0; key_code < key_map.size(); ++key_code) {
151 if (key_map[key_code] == key)
152 return static_cast<int>(key_code);
153 }
154
155 return -1;
156}

158bool is_valid_cocoa_key(uint16_t key_code) noexcept
