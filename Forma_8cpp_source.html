151std::pair<Mapped<std::shared_ptr<Kakshya::PlotContainer>>, Surface>
152plot(
153 std::string title,
154 uint32_t width,
155 uint32_t height,
156 std::shared_ptr<Kakshya::PlotContainer> container,
157 Plot::SeriesSpec spec)
158{
159 const uint64_t N = container->series_count() > 0
160 ? container->series_size(0)
161 : 0;
162
163 auto& atelier = internal::atelier();
164
165 auto window = atelier.create_window(
166 Core::WindowCreateInfo { .title = std::move(title), .width = width, .height = height });
167
168 auto surface = atelier.create_surface(window, window->get_create_info().title);
169
170 if (spec.background_fn) {
171 auto bg = atelier.create_element<float>(
172 surface.layer(), window,
173 *spec.background_fn,
174 0.F,
175 Graphics::PrimitiveTopology::TRIANGLE_STRIP,
176 static_cast<size_t>(4) * Kakshya::VertexLayout::for_meshes().stride_bytes);
177
178 const auto bg_id = bg.element.id;
179 auto buf = atelier.create_buffer(window, spec.capacity_for(N), spec.topology);
180 auto mapped = Plot::place(surface, std::move(buf), spec, std::move(container));
181 surface.layer().relate(mapped.element.id, bg_id);
182 surface.layer().send_to_back(bg_id);
183
184 internal::atelier().place_adornments(surface, spec, mapped.element.id);
185
186 return { std::move(mapped), std::move(surface) };
187 }
188
189 auto buf = atelier.create_buffer(window, spec.capacity_for(N), spec.topology);
190 auto mapped = Plot::place(surface, std::move(buf), spec, std::move(container));
191
192 internal::atelier().place_adornments(surface, spec, mapped.element.id);
193
194 return { std::move(mapped), std::move(surface) };
195}
