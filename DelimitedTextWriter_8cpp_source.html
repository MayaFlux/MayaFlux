25bool same_layout(
26 const Kakshya::CompositeLayout& left,
27 const Kakshya::CompositeLayout& right)
28{
29 if (left.stride_bytes() != right.stride_bytes()
30 || left.presence_bytes() != right.presence_bytes()
31 || left.fields().size() != right.fields().size())
32 return false;

48std::optional<std::string> format_number(
49 const Kakshya::Composite& element,
50 std::string_view name)
51{
52 const auto value = element.get<T>(name);
53 if (!value)
54 return std::nullopt;
55
56 if constexpr (std::same_as<T, bool>) {
57 return *value ? "true" : "false";
58 } else {
59 std::array<char, 128> buffer {};
60
61 if constexpr (std::integral<T>) {
62 using Formatted = std::conditional_t<
63 std::is_signed_v<T>, long long, unsigned long long>;
64
65 const auto [end, error] = std::to_chars(
66 buffer.data(), buffer.data() + buffer.size(),
67 static_cast<Formatted>(*value));
68
69 if (error != std::errc {})
70 return std::nullopt;
71
72 return std::string(buffer.data(), end);
73 } else {
74 if (!std::isfinite(*value))
75 return std::nullopt;
76
77 const auto [end, error] = std::to_chars(
78 buffer.data(), buffer.data() + buffer.size(), *value);
79
80 if (error != std::errc {})
81 return std::nullopt;
82
83 return std::string(buffer.data(), end);
84 }

192{
193 if (is_open()) {
194 m_last_error = "Close the current file before opening another";
195 return false;
196 }
197
198 m_last_error.clear();
199
200 if (!can_write(filepath)) {
201 m_last_error = "Expected a .csv or .tsv output path";
202 return false;
203 }
204
205 if (layout.fields().empty()) {
206 m_last_error = "Composite output requires at least one field";
207 return false;
208 }
209
210 for (const auto& field : layout.fields()) {
211 if (!supported_type(field.type)) {
212 m_last_error = "Unsupported Composite field: " + field.name;
213 return false;
214 }
215 }
216
217 Kakshya::CompositeArray empty(layout);
218 m_layout = empty.layout();
219 m_active_delimiter = m_delimiter != '\0'
220 ? m_delimiter

222

224 std::ios::binary | std::ios::out | std::ios::trunc);
225
226 if (!m_file) {

228 m_layout.reset();
229 return false;
230 }

232 m_rows_written = 0;

236
237 for (size_t i = 0; i < fields.size(); ++i) {
238 if (i != 0)
239 m_file.put(m_active_delimiter);

319{

322 || text.find('"') != std::string_view::npos
323 || text.find('\r') != std::string_view::npos

339 m_last_error = "Delimited text cell exceeds stream size";

357bool DelimitedTextWriter::write_element(
358 const Kakshya::Composite& element)
359{
360 const auto& fields = m_layout->fields();
361
362 for (size_t i = 0; i < fields.size(); ++i) {
363 if (i != 0)
364 m_file.put(m_active_delimiter);
365
366 const auto& field = fields[i];
367 if (!element.has(field.name))
368 continue;
369
370 if (field.type == typeid(std::string)) {
371 const auto text = element.text(field.name);
372
373 if (!text || !write_cell(*text, text->empty())) {
374 if (m_last_error.empty())
375 m_last_error = "Invalid text field: " + field.name;
376 return false;
377 }
378 } else {
379 const auto text = format_field(element, field);
380
381 if (!text || !write_cell(*text, false)) {
382 if (m_last_error.empty())
383 m_last_error = "Invalid numeric field: " + field.name;
384 return false;
385 }
386 }
387 }
388
389 m_file.put('\n');
390
391 if (!m_file) {
392 m_last_error = "Failed to write delimited text element";
393 return false;
394 }
395
396 return true;
