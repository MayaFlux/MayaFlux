12 /**
13 * @brief Release fences of submissions that were not awaited: the ingest
14 * pair and the flow pyramid build. Waits when one is still pending,
15 * so any descriptor those submissions used is safe to rewrite once
16 * this returns.
17 */
18 void reap_fences(VisionGpuContexts& contexts)
19 {
20 auto& foundry = Portal::Graphics::get_shader_foundry();
21 for (auto* slot : { &contexts.ingest_fence, &contexts.ingest_barrier_fence, &contexts.flow_state.build_fence }) {
22 if (*slot != Portal::Graphics::INVALID_FENCE) {
23 foundry.wait_for_fence(*slot);
24 foundry.release_fence(*slot);
25 *slot = Portal::Graphics::INVALID_FENCE;
26 }
27 }
28 }

30 /**
31 * @brief Convert a non-storage seed frame to an rgba32f storage image.
32 *
33 * Frames already carrying storage usage pass through. Otherwise
34 * vision_ingest.comp samples the frame as a texture (sampler does the
35 * unorm/sRGB decode) into contexts.ingest's rgba32f output. The dispatch
36 * is not awaited — a trailing compute barrier gives the memory
37 * dependency, fences are reaped on the next run.
38 */
39 std::shared_ptr<Core::VKImage> op_ingest(
40 VisionGpuContexts& contexts,
41 const std::shared_ptr<Core::VKImage>& frame,
42 uint32_t w, uint32_t h)
43 {
44 if (!frame || !frame->is_initialized())
45 return frame;
46 if (static_cast<bool>(frame->get_usage_flags() & vk::ImageUsageFlagBits::eStorage))
47 return frame;
48
49 auto& foundry = Portal::Graphics::get_shader_foundry();
50 auto& ingest = contexts.ingest;
51
52 ingest.swap_shader({
53 .shader_path = "vision_ingest.comp.spv",
54 .workgroup_size = k_wg2d,
55 .push_constant_size = sizeof(IngestPC),
56 });
57 ingest.stage_image(frame);
58 ingest.set_push_constants(IngestPC { .width = w, .height = h });
59 ingest.prepare_output_image(w, h);
60 ingest.set_output_dimensions(w, h);
61 contexts.ingest_fence = ingest.dispatch_async({});
62 ingest.clear_output_dimensions();
63
64 auto out = ingest.get_output_image(0);
65
66 if (out) {
67 const auto bcmd = foundry.begin_commands(
68 Portal::Graphics::ShaderFoundry::CommandBufferType::COMPUTE);
69 foundry.image_barrier(bcmd, out->get_image(),
70 vk::ImageLayout::eGeneral, vk::ImageLayout::eGeneral,
71 vk::AccessFlagBits::eShaderWrite, vk::AccessFlagBits::eShaderRead,
72 vk::PipelineStageFlagBits::eComputeShader,
73 vk::PipelineStageFlagBits::eComputeShader);
74 contexts.ingest_barrier_fence = foundry.submit_async(bcmd);
75 }
76
77 return out;
78 }

330void VisionGpuExecutor::reset()
331{
332 if (!m_contexts)
333 return;
334
335 auto& contexts = *m_contexts;
336
337 reap_fences(contexts);
338
339 if (contexts.suspended.is_active()) {
340 auto& foundry = Portal::Graphics::get_shader_foundry();
341 foundry.wait_for_fence(contexts.suspended.fence);
342 foundry.release_fence(contexts.suspended.fence);
343 contexts.suspended.fence = Portal::Graphics::INVALID_FENCE;
344 contexts.suspended.finalize = nullptr;
345 }
346
347 contexts.flow_state.have_prev = false;
348 contexts.flow_state.curr_ready = false;
349 contexts.flow_state.last_flow.reset();
350 contexts.flow_state.last_tracks.clear();
351 contexts.flow_state.last_track_count = 0;
352 contexts.flow_state.last_export.reset();
353 contexts.flow_state.export_pending = false;
354
355 contexts.pass.sequence = nullptr;
356 contexts.pass.index = 0;
357 contexts.pass.result = Kinesis::Vision::VisionResult {};
358 contexts.pass.current.reset();
359 contexts.pass.forget();
360 contexts.pass.completed.clear();
361
362 contexts.source.reset();
363 contexts.bound_staged.reset();
364}

366void VisionGpuExecutor::after_step(VisionGpuContexts& contexts, size_t index)
367{
368 const auto& steps = contexts.pass.sequence->steps;
369 const VisionOp source_op = steps[index].op;
370 if (source_op != VisionOp::RgbaToGray && source_op != VisionOp::RgbaToHsv)
371 return;
372
373 const uint32_t src_channel = source_op == VisionOp::RgbaToHsv ? 2U : 0U;
374
375 uint32_t levels = 0;
376 bool needs_flow = false;
377 bool confine_seen = false;
378 bool warned = false;
379 for (size_t i = index + 1; i < steps.size(); ++i) {
380 if (steps[i].op == VisionOp::Confine)
381 confine_seen = true;
382
383 const bool is_flow = steps[i].op == VisionOp::TrackKeypoints || steps[i].op == VisionOp::OpticalFlowDense;
384 if (is_flow && confine_seen && !warned) {
385 MF_WARN(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
386 "run_gpu: Confine sits between RgbaToGray and a flow step; the flow pyramid is "
387 "built from the unconfined gray frame, so results are wrong or rejected. "
388 "Place Confine before RgbaToGray");
389 warned = true;
390 }
391
392 if (steps[i].op == VisionOp::TrackKeypoints) {
393 if (const auto* p = std::get_if<TrackKeypointsParams>(&steps[i].params)) {
394 needs_flow = true;
395 levels = std::max(levels, p->levels);
396 }
397 } else if (steps[i].op == VisionOp::OpticalFlowDense) {
398 if (const auto* p = std::get_if<OpticalFlowDenseParams>(&steps[i].params)) {
399 needs_flow = true;
400 levels = std::max(levels, p->levels);
401 }
402 }
403 }
404
405 if (needs_flow)
406 build_flow_pyramid(contexts, levels, src_channel);
407}

413VisionResult VisionGpuExecutor::run(
414 VisionGpuContexts& contexts,
415 const VisionSequence& sequence,
416 const std::shared_ptr<Core::VKImage>& image,
417 uint32_t w, uint32_t h)
418{
419 auto& pixel_ctx = contexts.pixel;
420 auto& structured_ctx = contexts.structured;
421 auto& label_ctx = contexts.labels;
422 auto& cc_pipeline = contexts.component_contours;
423
424 auto& foundry = Portal::Graphics::get_shader_foundry();
425 std::unordered_map<size_t, CompletedOp> completed_ops;
426 size_t begin = 0;
427
428 if (contexts.suspended.is_active()) {
429 if (&sequence != contexts.pass.sequence) {
430 MF_WARN(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
431 "run_gpu: polling a suspension with a different sequence; "
432 "the walk continues on the sequence the run started from");
433 }
434
435 if (!foundry.is_fence_signaled(contexts.suspended.fence)) {
436 VisionResult pending;
437 pending.status = VisionStatus::SUSPENDED;
438 pending.suspended_at = contexts.pass.index;
439 return pending;
440 }
441
442 foundry.release_fence(contexts.suspended.fence);
443 contexts.suspended.fence = Portal::Graphics::INVALID_FENCE;
444
445 if (auto finalize = std::move(contexts.suspended.finalize)) {
446 contexts.suspended.finalize = nullptr;
447 finalize(contexts);
448 }
449
450 begin = contexts.pass.index + 1;
451 } else {
452 reap_fences(contexts);
453 contexts.flow_state.curr_ready = false;
454 contexts.pass.begin(sequence, w, h);
455 contexts.bound_staged.reset();
456 const auto seed = op_ingest(contexts, image, w, h);
457 contexts.pass.current = seed;
458 contexts.source = seed;
459 }
460
461 for (contexts.pass.index = begin; contexts.pass.index < sequence.steps.size(); ++contexts.pass.index) {
462 const auto& step = sequence.steps[contexts.pass.index];
463
464 const uint32_t w = contexts.pass.w;
465 const uint32_t h = contexts.pass.h;
466
467 const auto cfg = config(step.op, step.params);
468
469 if (cfg.shader_id == Portal::Graphics::INVALID_SHADER && cfg.shader_path.empty()) {
470 MF_ERROR(Journal::Component::Yantra, Journal::Context::ComputeMatrix,
471 "run_gpu: no GPU implementation for VisionOp {}",
472 static_cast<int>(step.op));
473 return VisionResult {};
474 }
475
476 if (cfg.shader_id != contexts.bound_config.shader_id
477 || cfg.shader_path != contexts.bound_config.shader_path) {
478 pixel_ctx.swap_shader(cfg);
479 contexts.bound_config = cfg;
480 }
481 if (contexts.pass.current != contexts.bound_staged) {
482 pixel_ctx.stage_image(contexts.pass.current);
483 contexts.bound_staged = contexts.pass.current;
484 }
485 if (w != contexts.pass.storage_w || h != contexts.pass.storage_h
486 || (contexts.pass.current && contexts.pass.current == pixel_ctx.get_output_image(0))) {
487 pixel_ctx.prepare_output_image(w, h);
488 contexts.pass.storage_w = w;
489 contexts.pass.storage_h = h;
490 }
491 pixel_ctx.set_output_dimensions(w, h);
492
493 switch (step.op) {
494 case VisionOp::Downsample2x: {
495 const uint32_t new_w = std::max(1U, w / 2);
496 const uint32_t new_h = std::max(1U, h / 2);
497
498 pixel_ctx.prepare_output_image(new_w, new_h);
499 pixel_ctx.set_output_dimensions(new_w, new_h);
500 {
501 const auto f = pixel_ctx.dispatch_async({});
502 foundry.wait_for_fence(f);
503 foundry.release_fence(f);
504 }
505 pixel_ctx.clear_output_dimensions();
506
507 auto downsampled = pixel_ctx.get_output_image(0);
508 completed_ops[Kinesis::Vision::hash_vision_step(step.op, step.params)] = { .output = downsampled, .input = contexts.pass.current };
509 contexts.pass.result.images.downsample_2x = downsampled;
510 contexts.pass.current = downsampled;
511 contexts.bound_staged.reset();
512
513 contexts.pass.set_geometry(new_w, new_h);
514 contexts.pass.storage_w = new_w;
515 contexts.pass.storage_h = new_h;
516
517 continue;
518 }
519 case VisionOp::Threshold: {
520 const auto& p = std::get<ThresholdParams>(step.params);
521 if (p.channels == ChannelMask::NONE) {
522 pixel_ctx.set_push_constants(ThresholdPC { .value = p.value });
523 } else {
524 pixel_ctx.set_push_constants(ThresholdBandsPC {
525 .lo0 = p.bands[0].lo, .lo1 = p.bands[1].lo, .lo2 = p.bands[2].lo,
526 .hi0 = p.bands[0].hi, .hi1 = p.bands[1].hi, .hi2 = p.bands[2].hi,
527 .channels = static_cast<uint32_t>(p.channels) });
528 }
529 break;
530 }
531 case VisionOp::NormalizeRange: {
532 const auto& p = std::get<NormalizeRangeParams>(step.params);
533 const float scale = (p.hi > p.lo) ? 1.0F / (p.hi - p.lo) : 1.0F;
534 const float off = (p.hi > p.lo) ? -p.lo / (p.hi - p.lo) : 0.0F;
535 pixel_ctx.set_push_constants(NormalizePC { .scale = scale, .offset = off });
536 break;
537 }
538 case VisionOp::RgbaToGray:
539 pixel_ctx.set_push_constants(RgbaToGrayPC {
540 .wr = 0.299F, .wg = 0.587F, .wb = 0.114F, .wa = 0.0F });
541 break;
542 case VisionOp::GaussianBlur: {
543 const auto& p = std::get<GaussianBlurParams>(step.params);
544 const auto radius = static_cast<uint32_t>(std::ceil(p.sigma * 3.0F));
545 const auto& weights = gaussian_kernel_2d(radius, p.sigma);
546 pixel_ctx.set_binding_data(2, std::span<const float>(weights));
547 pixel_ctx.set_push_constants(GaussianPC { .radius = radius, .width = w, .height = h });
548 break;
549 }
550 case VisionOp::Erode:
551 case VisionOp::Dilate:
552 case VisionOp::MorphGradient:
553 pixel_ctx.set_push_constants(MorphPC {
554 .radius = std::get<MorphParams>(step.params).radius });
555 break;
556 case VisionOp::ThresholdAdaptive: {
557 const auto& p = std::get<ThresholdAdaptiveParams>(step.params);
558 pixel_ctx.set_push_constants(ThresholdAdaptivePC { .block_size = p.block_size, .offset = p.offset, .channels = static_cast<uint32_t>(p.channels) });
559 break;
560 }
561 case VisionOp::ThresholdOtsu: {
562 const auto* otsu = std::get_if<OtsuParams>(&step.params);
563 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = op_threshold_otsu(contexts, otsu ? *otsu : OtsuParams {});
564 continue;
565 }
566 case VisionOp::Confine: {
567 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = op_confine(contexts, std::get<ConfineParams>(step.params));
568 continue;
569 }
570 case VisionOp::Open:
571 case VisionOp::Close: {
572 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = op_open_close(contexts, step.op, std::get<MorphParams>(step.params));
573 continue;
574 }
575 case VisionOp::Canny: {
576 auto done = op_canny(contexts, step.params, std::get<CannyParams>(step.params));
577 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = done;
578 continue;
579 }
580 case VisionOp::HarrisResponse: {
581 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = op_harris_response(contexts, std::get<HarrisParams>(step.params));
582 continue;
583 }
584 case VisionOp::ExtractPeaks: {
585 op_extract_peaks(contexts, std::get<ExtractPeaksParams>(step.params));
586 continue;
587 }
588 case VisionOp::ConnectedComponents: {
589 if (!contexts.pass.current) {
590 continue;
591 }
592 op_connected_components(contexts, std::get<Kinesis::Vision::ConnectedComponentsParams>(step.params));
593 continue;
594 }
595 case VisionOp::FindContours: {
596 if (!op_find_contours(contexts,
597 std::get<Kinesis::Vision::FindContoursParams>(step.params)))
598 return VisionResult {};
599 continue;
600 }
601 case VisionOp::TrackKeypoints: {
602 if (op_track_keypoints(contexts, step)) {
603 VisionResult pending;
604 pending.status = VisionStatus::SUSPENDED;
605 pending.suspended_at = contexts.pass.index;
606 return pending;
607 }
608 continue;
609 }
610 case VisionOp::OpticalFlowDense: {
611 if (op_dense_flow(contexts, step)) {
612 VisionResult pending;
613 pending.status = VisionStatus::SUSPENDED;
614 pending.suspended_at = contexts.pass.index;
615 return pending;
616 }
617 continue;
618 }
619 default:
620 break;
621 }
622
623 auto dispatch_input = contexts.pass.current;
624 const auto fence = pixel_ctx.dispatch_async({});
625
626 if (step.deferred) {
627 pixel_ctx.clear_output_dimensions();
628 contexts.pass.current = pixel_ctx.get_output_image(0);
629 contexts.bound_staged.reset();
630
631 const Kinesis::Vision::GpuVisionPass::Completed done {
632 .output = contexts.pass.current,
633 .input = dispatch_input
634 };
635 contexts.pass.completed[Kinesis::Vision::hash_vision_step(step.op, step.params)] = done;
636
637 contexts.suspended.fence = fence;
638 contexts.suspended.finalize = [](VisionGpuContexts& c) { after_step(c, c.pass.index); };
639
640 VisionResult pending;
641 pending.status = VisionStatus::SUSPENDED;
642 pending.suspended_at = contexts.pass.index;
643 return pending;
644 }
645
646 foundry.wait_for_fence(fence);
647 foundry.release_fence(fence);
648
649 pixel_ctx.clear_output_dimensions();
650 contexts.pass.current = pixel_ctx.get_output_image(0);
651 contexts.bound_staged.reset();
652 completed_ops[Kinesis::Vision::hash_vision_step(step.op, step.params)] = { .output = contexts.pass.current, .input = dispatch_input };
653
654 switch (step.op) {
655 case VisionOp::RgbaToGray:
656 contexts.pass.result.gray = contexts.pass.current;
657 break;
658 case VisionOp::RgbaToHsv:
659 contexts.pass.result.images.rgba_to_hsv = contexts.pass.current;
660 break;
661 case VisionOp::GrayToRgba:
662 contexts.pass.result.images.gray_to_rgba = contexts.pass.current;
663 break;
664 case VisionOp::Threshold:
665 contexts.pass.result.images.threshold = contexts.pass.current;
666 break;
667 case VisionOp::ThresholdAdaptive:
668 contexts.pass.result.images.threshold_adaptive = contexts.pass.current;
669 break;
670 case VisionOp::NormalizeInplace:
671 contexts.pass.result.images.normalize_inplace = contexts.pass.current;
672 break;
673 case VisionOp::NormalizeRange:
674 contexts.pass.result.images.normalize_range = contexts.pass.current;
675 break;
676 case VisionOp::GaussianBlur:
677 contexts.pass.result.images.gaussian_blur = contexts.pass.current;
678 break;
679 case VisionOp::FilterSeparable:
680 contexts.pass.result.images.filter_separable = contexts.pass.current;
681 break;
682 case VisionOp::Sobel:
683 contexts.pass.result.images.sobel = contexts.pass.current;
684 break;
685 case VisionOp::Scharr:
686 contexts.pass.result.images.scharr = contexts.pass.current;
687 break;
688 case VisionOp::Erode:
689 contexts.pass.result.images.erode = contexts.pass.current;
690 break;
691 case VisionOp::Dilate:
692 contexts.pass.result.images.dilate = contexts.pass.current;
693 break;
694 case VisionOp::MorphGradient:
695 contexts.pass.result.images.morph_gradient = contexts.pass.current;
696 break;
697 default:
698 break;
699 }
700
701 after_step(contexts, contexts.pass.index);
702 }
703
704 return std::move(contexts.pass.result);
705}
