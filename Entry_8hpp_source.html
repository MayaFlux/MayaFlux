65 /**
66 * @brief Union of the header strip and every row region.
67 *
68 * Collapses to the header region alone when the group has no rows.
69 */
70 [[nodiscard]] Kinesis::AABB2D bounds() const noexcept
71 {
72 Kinesis::AABB2D b = header.bounds();
73 for (const auto& r : rows) {
74 const auto rb = r.bounds();
75 b.min = glm::min(b.min, rb.min);
76 b.max = glm::max(b.max, rb.max);
77 }
78 return b;
79 }
