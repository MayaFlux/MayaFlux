13 float uniform_scale(const glm::vec3& from, const glm::vec3& to, bool largest)
14 {
15 const std::array<std::pair<float, float>, 3> axes { {
16 { from.x, to.x },
17 { from.y, to.y },
18 { from.z, to.z },
19 } };
20
21 bool any = false;
22 float scale = 1.0F;
23 for (const auto& [source, target] : axes) {
24 if (source <= k_epsilon) {
25 continue;
26 }
27 const float ratio = target / source;
28 scale = !any ? ratio : (largest ? std::max(scale, ratio) : std::min(scale, ratio));
29 any = true;
30 }
31 return scale;
32 }

34 Kinesis::AABB3D resolve(
35 const Kinesis::AABB3D& source,
36 const Kinesis::AABB3D& target,
37 Portal::Graphics::FitMode mode)
38 {
39 using Portal::Graphics::FitMode;
40
41 const glm::vec3 from = source.extent();
42 const glm::vec3 to = target.extent();
43
44 switch (mode) {
45 case FitMode::CONTAIN:
46 return Kinesis::AABB3D::from_center(target.center(), from * uniform_scale(from, to, false) * 0.5F);
47 case FitMode::COVER:
48 return Kinesis::AABB3D::from_center(target.center(), from * uniform_scale(from, to, true) * 0.5F);
49 case FitMode::CENTER:
50 return Kinesis::AABB3D::from_center(target.center(), from * 0.5F);
51 case FitMode::STRETCH:
52 case FitMode::TILE:
53 case FitMode::TILE_MIRRORED:
54 break;
55 }
56 return target;
57 }
