9 float decode_half(uint16_t h)
10 {
11 const uint32_t sign = static_cast<uint32_t>(h & 0x8000U) << 16;
12 const uint32_t exp = (h >> 10) & 0x1FU;
13 const uint32_t mant = h & 0x3FFU;
14
15 uint32_t bits {};
16
17 if (exp == 0) {
18 if (mant == 0) {
19 bits = sign;
20 } else {
21 uint32_t e = 0;
22 uint32_t m = mant;
23 while ((m & 0x400U) == 0) {
24 m <<= 1;
25 ++e;
26 }
27 m &= 0x3FFU;
28 bits = sign | ((113U - e) << 23) | (m << 13);
29 }
30 } else if (exp == 0x1FU) {
31 bits = sign | 0x7F800000U | (mant << 13);
32 } else {
33 bits = sign | ((exp - 15U + 127U) << 23) | (mant << 13);
34 }
35
36 float out {};
37 std::memcpy(&out, &bits, sizeof(out));
38 return out;
39 }

41 uint16_t encode_half(float f)
42 {
43 uint32_t bits {};
44 std::memcpy(&bits, &f, sizeof(bits));
45
46 const auto sign = static_cast<uint16_t>((bits >> 16) & 0x8000U);
47 const int32_t exp = static_cast<int32_t>((bits >> 23) & 0xFFU) - 127 + 15;
48 uint32_t mant = bits & 0x7FFFFFU;
49
50 if (exp >= 0x1F)
51 return static_cast<uint16_t>(sign | 0x7C00U);
52
53 if (exp <= 0) {
54 if (exp < -10)
55 return sign;
56 mant |= 0x800000U;
57 return static_cast<uint16_t>(sign | (mant >> static_cast<uint32_t>(14 - exp)));
58 }
59
60 return static_cast<uint16_t>(
61 sign | (static_cast<uint32_t>(exp) << 10) | (mant >> 13));
62 }

119 void blend_typed(std::span<const uint8_t> a, std::span<const uint8_t> b, uint8_t* out, double weight)
120 {
121 const auto* from = reinterpret_cast<const T*>(a.data());
122 const auto* to = reinterpret_cast<const T*>(b.data());
123 auto* dest = reinterpret_cast<T*>(out);
124 const size_t count = a.size() / sizeof(T);
125 const double rounding = std::is_integral_v<T> ? 0.5 : 0.0;
126
127 for (size_t i = 0; i < count; ++i) {
128 const auto x = static_cast<double>(from[i]);
129 dest[i] = static_cast<T>(x + (static_cast<double>(to[i]) - x) * weight + rounding);
130 }
131 }

166double read_normalized_at(const DataVariant& v,
167 ImageFormat format,
168 const std::optional<DataDimension::ValueRange>& range,
169 size_t elem_index)
170{
171 return std::visit(
172 [&](const auto& vec) -> double {
173 using T = typename std::decay_t<decltype(vec)>::value_type;
174
175 if constexpr (std::is_same_v<T, uint8_t>
176 || std::is_same_v<T, uint16_t>
177 || std::is_same_v<T, float>) {
178
179 if (elem_index >= vec.size())
180 return 0.0;
181
182 double raw {};
183 double type_max = 1.0;
184
185 if constexpr (std::is_same_v<T, uint8_t>) {
186 raw = static_cast<double>(vec[elem_index]);
187 type_max = 255.0;
188 } else if constexpr (std::is_same_v<T, uint16_t>) {
189 if (is_float_format(format)) {
190 raw = static_cast<double>(decode_half(vec[elem_index]));
191 } else {
192 raw = static_cast<double>(vec[elem_index]);
193 type_max = 65535.0;
194 }
195 } else {
196 raw = static_cast<double>(vec[elem_index]);
197 }
198
199 if (!range)
200 return raw / type_max;
201
202 if (range->invalid && raw == *range->invalid)
203 return std::numeric_limits<double>::quiet_NaN();
204
205 const double span = range->max - range->min;
206 if (span <= 0.0)
207 return 0.0;
208
209 return std::clamp((raw - range->min) / span, 0.0, 1.0);
210 } else {
211 return 0.0;
212 }
213 },
214 v);
215}
