108std::vector<std::shared_ptr<Buffers::SoundContainerBuffer>> create_audio_container_buffers(
109 const std::shared_ptr<Buffers::BufferManager>& buffer_manager,
110 const std::shared_ptr<Kakshya::SoundFileContainer>& container)
111{
112 std::vector<std::shared_ptr<Buffers::SoundContainerBuffer>> created_buffers;
113
114 const uint32_t num_channels = container->get_num_channels();
115 created_buffers.reserve(num_channels);
116
117 for (uint32_t channel = 0; channel < num_channels; ++channel) {
118 auto container_buffer = buffer_manager->create_audio_buffer<Buffers::SoundContainerBuffer>(
119 Buffers::ProcessingToken::AUDIO_BACKEND,
120 channel,
121 container,
122 channel);
123
124 container_buffer->initialize();
125 created_buffers.push_back(std::move(container_buffer));
126 }
127
128 return created_buffers;
129}

131std::shared_ptr<Buffers::VolumeGridBuffer> create_volume_grid_buffer(
132 const std::shared_ptr<Buffers::BufferManager>& buffer_manager,
133 const Kakshya::VolumeData& volume_data)
134{
135 std::vector<Buffers::VolumeGridBuffer::FieldDecl> decls;
136 decls.reserve(volume_data.fields.size());
137 for (const auto& field : volume_data.fields) {
138 decls.push_back({
139 .name = field.name,
140 .stride_bytes = field.is_vector() ? sizeof(glm::vec4) : sizeof(float),
141 .double_buffered = true,
142 .semantics = field.semantics,
143 });
144 }
145
146 auto buffer = buffer_manager->create_graphics_buffer<Buffers::VolumeGridBuffer>(
147 Buffers::ProcessingToken::GRAPHICS_BACKEND,
148 volume_data.lattice, decls);
149
150 if (!buffer)
151 return nullptr;
152
153 if (!IO::upload_volume(volume_data, buffer))
154 return nullptr;
155
156 return buffer;
157}

164 auto writer = ModelWriterRegistry::instance().create_writer(filepath);

168 return false;

177 "save_mesh: wrote '{}'", filepath);
