96 /**
97 * @brief Core transformation implementation
98 * @param input Input data to transform
99 * @return Transformed output data
100 *
101 * Performs the mathematical operation specified by m_operation on the input data.
102 * Supports both in-place and out-of-place transformations based on transformer settings.
103 */
104 output_type transform_implementation(input_type& input) override
105 {
106 switch (m_operation) {
107
108 case MathematicalOperation::GAIN: {
109 const auto a = get_parameter_or<double>("gain_factor", 1.0);
110 return apply_per_channel(input, [a](std::span<double> ch) {
111 D::linear(ch, a, 0.0);
112 });
113 }
114
115 case MathematicalOperation::OFFSET: {
116 const auto b = get_parameter_or<double>("offset_value", 0.0);
117 return apply_per_channel(input, [b](std::span<double> ch) {
118 D::linear(ch, 1.0, b);
119 });
120 }
121
122 case MathematicalOperation::POWER: {
123 const auto exp = get_parameter_or<double>("exponent", 2.0);
124 return apply_per_channel(input, [exp](std::span<double> ch) {
125 D::power(ch, exp);
126 });
127 }
128
129 case MathematicalOperation::LOGARITHMIC: {
130 const auto a = get_parameter_or<double>("scale", 1.0);
131 const auto b = get_parameter_or<double>("input_scale", 1.0);
132 const auto c = get_parameter_or<double>("offset", 1.0);
133 const auto base = get_parameter_or<double>("base", std::numbers::e);
134 return apply_per_channel(input, [a, b, c, base](std::span<double> ch) {
135 D::logarithmic(ch, a, b, c, base);
136 });
137 }
138
139 case MathematicalOperation::EXPONENTIAL: {
140 const auto a = get_parameter_or<double>("scale", 1.0);
141 const auto b = get_parameter_or<double>("rate", 1.0);
142 const auto base = get_parameter_or<double>("base", std::numbers::e);
143 return apply_per_channel(input, [a, b, base](std::span<double> ch) {
144 D::exponential(ch, a, b, base);
145 });
146 }
147
148 case MathematicalOperation::TRIGONOMETRIC: {
149 const auto fn = get_parameter_or<std::string>("trig_function", "sin");
150 const auto freq = get_parameter_or<double>("frequency", 1.0);
151 const auto amp = get_parameter_or<double>("amplitude", 1.0);
152 const auto ph = get_parameter_or<double>("phase", 0.0);
153 if (fn == "cos")
154 return apply_per_channel(input, [freq, amp, ph](std::span<double> ch) {
155 D::apply_trig(ch, [](double x) { return std::cos(x); }, freq, amp, ph);
156 });
157 if (fn == "tan")
158 return apply_per_channel(input, [freq, amp, ph](std::span<double> ch) {
159 D::apply_trig(ch, [](double x) { return std::tan(x); }, freq, amp, ph);
160 });
161 return apply_per_channel(input, [freq, amp, ph](std::span<double> ch) {
162 D::apply_trig(ch, [](double x) { return std::sin(x); }, freq, amp, ph);
163 });
164 }
165
166 case MathematicalOperation::QUANTIZE: {
167 const auto bits = get_parameter_or<uint8_t>("bits", 16);
168 return apply_per_channel(input, [bits](std::span<double> ch) {
169 D::quantize(ch, bits);
170 });
171 }
172
173 case MathematicalOperation::NORMALIZE: {
174 const auto peak = get_parameter_or<double>("target_peak", 1.0);
175 return apply_per_channel(input, [peak](std::span<double> ch) {
176 D::normalize(ch, -peak, peak);
177 });
178 }
179
180 case MathematicalOperation::POLYNOMIAL: {
181 const auto coeffs = get_parameter_or<std::vector<double>>(
182 "coefficients", std::vector<double> { 0.0, 1.0 });
183 auto poly = std::make_shared<Nodes::Generator::Polynomial>(coeffs);
184 return apply_per_channel(input, [&poly](std::span<double> ch) {
185 std::ranges::transform(ch, ch.begin(),
186 [&poly](double x) { return poly->process_sample(x); });
187 });
188 }
189
190 default:
191 return create_output(input);
192 }
193 }

225 /**
226 * @brief Extracts per-channel spans, applies an in-place func to each, and reconstructs.
227 * @tparam Func Callable matching void(std::span<double>)
228 *
229 * In-place: results are copied back into the original channel spans of @p input before
230 * reconstruction. Out-of-place: input is not mutated.
231 */
232 template <typename Func>
233 output_type apply_per_channel(input_type& input, Func&& func)
234 {
235 auto [channels, structure_info] = OperationHelper::extract_structured_double(input);
236 m_working_buffer.resize(channels.size());
237 for (size_t i = 0; i < channels.size(); ++i) {
238 m_working_buffer[i].assign(channels[i].begin(), channels[i].end());
239 func(std::span<double> { m_working_buffer[i] });
240 }
241 if (this->is_in_place())
242 for (size_t i = 0; i < channels.size(); ++i)
243 std::ranges::copy(m_working_buffer[i], channels[i].begin());
244 return create_output(
245 OperationHelper::reconstruct_from_double<InputType>(m_working_buffer, structure_info));
246 }

280 template <typename T>
281 T get_parameter_or(const std::string& name, const T& default_value) const
