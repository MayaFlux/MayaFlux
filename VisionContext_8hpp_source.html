214/**
215 * @brief State threaded through one execution of a VisionSequence.
216 *
217 * Shared by the CPU and GPU executors so op functions have one signature on
218 * both paths. Three bands: the walk, working storage, and redundancy caches.
219 *
220 * Cross-run retained state is not here. It is owned vectors on the CPU path
221 * and owned images on the GPU path, neither of which is a working storage
222 * handle, so each executor holds its own in its own types.
223 *
224 * @tparam Handle Working storage handle. Slot index on the CPU path,
225 * shared_ptr<VKImage> on the GPU path.
226 *
227 * sequence is non-owning and valid only for the run that constructed the
228 * pass. Storage and caches outlive a single run once the pass is held by
229 * the executor; the walk band is reset per run by begin().
230 */
231template <typename Handle>
232struct VisionPass {
233 // -------------------------------------------------------------------------
234 // Walk
235 // -------------------------------------------------------------------------
236
237 const VisionSequence* sequence { nullptr };
238 size_t index { 0 };
239 uint32_t w { 0 };
240 uint32_t h { 0 };
241 uint32_t channels { 4 };
242 VisionResult result;
243
244 // -------------------------------------------------------------------------
245 // Working storage
246 // -------------------------------------------------------------------------
247
248 Handle current {};
249 uint32_t storage_w { 0 };
250 uint32_t storage_h { 0 };
251
252 // -------------------------------------------------------------------------
253 // Retained across runs
254 // -------------------------------------------------------------------------
255
256 Handle prev {};
257 Handle prev_cache {};
258 std::vector<Keypoint> prev_keypoints;
259
260 // -------------------------------------------------------------------------
261 // Redundancy caches
262 // -------------------------------------------------------------------------
263
264 struct Completed {
265 Handle output {};
266 Handle input {};
267 };
268
269 std::unordered_map<size_t, Completed> completed;
270
271 /**
272 * @brief Reset the walk band for a fresh run.
273 *
274 * Clears the memo unconditionally: its keys compare image handles, which
275 * proxy for content only within one walk. Across runs the caller's input
276 * and the contexts' output images are reused in place, so an equal handle
277 * does not mean equal pixels.
278 *
279 * Retained cross-run state is dropped only when the geometry changes,
280 * since surviving between runs is what it is for.
281 */
282 void begin(const VisionSequence& seq, uint32_t width, uint32_t height)
283 {
284 completed.clear();
285
286 if (width != w || height != h)
287 forget();
288
289 sequence = &seq;
290 index = 0;
291 channels = 4;
292 result = VisionResult {};
293 set_geometry(width, height);
294 }
295
296 /**
297 * @brief Discard retained cross-run state. Storage and caches are kept.
298 */
299 void forget()
300 {
301 prev = Handle {};
302 prev_cache = Handle {};
303 prev_keypoints.clear();
304 }
305
306 [[nodiscard]] const VisionStep& step() const noexcept
307 {
308 return sequence->steps[index];
309 }
310
311 [[nodiscard]] size_t plane_size() const noexcept
312 {
313 return static_cast<size_t>(w) * h;
314 }
315
316 /**
317 * @brief Op at @p offset steps ahead, or nullptr past the end.
318 *
319 * Replaces the adjacency booleans derived by VisionSequence::Builder.
320 * Correct under any mutation of steps because it is evaluated at the
321 * point of use.
322 */
323 [[nodiscard]] const VisionStep* ahead(size_t offset = 1) const noexcept
324 {
325 const auto& steps = sequence->steps;
326 const size_t at = index + offset;
327 return at < steps.size() ? &steps[at] : nullptr;
328 }
329
330 /**
331 * @brief Op at @p offset steps back, or nullptr before the start.
332 *
333 * Counterpart to ahead(), for ops validating what produced their input.
334 */
335 [[nodiscard]] const VisionStep* behind(size_t offset = 1) const noexcept
336 {
337 if (offset > index)
338 return nullptr;
339 return &sequence->steps[index - offset];
340 }
341
342 void set_geometry(uint32_t width, uint32_t height) noexcept
343 {
344 w = width;
345 h = height;
346 result.w = width;
347 result.h = height;
348 }
349
350 /**
351 * @brief Memoised output for @p key when it was produced from @p input.
352 */
353 [[nodiscard]] const Handle* memo(size_t key, const Handle& input) const
354 {
355 auto it = completed.find(key);
356 if (it == completed.end() || !(it->second.input == input))
357 return nullptr;
358 return &it->second.output;
359 }
360};
