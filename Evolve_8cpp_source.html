43 /**
44 * @brief Resolve a scalar ref against a volume, declaring the field if absent.
45 * @param volume Volume being wired.
46 * @param ref Ref supplied by the caller.
47 * @param fallback Name given to an unnamed ref. Empty leaves it unnamed.
48 * @param single_slot True to declare a single-slot field, false for a
49 * double-buffered one.
50 * @return A ref issued by the volume, or the input unchanged when it cannot
51 * be resolved.
52 */
53 Buffers::ScalarRef resolve_scalar(
54 const std::shared_ptr<Buffers::VolumeGridBuffer>& volume,
55 Buffers::ScalarRef ref,
56 std::string_view fallback,
57 bool single_slot)
58 {
59 if (ref.name.empty()) {
60 if (fallback.empty()) {
61 return ref;
62 }
63 ref.name = std::string(fallback);
64 }
65
66 if (owned_elsewhere(ref.owner, volume)) {
67 return ref;
68 }
69
70 if (!has_field(volume, ref.name)) {
71 return single_slot ? volume->declare_scratch(ref.name) : volume->declare_scalar(ref.name);
72 }
73
74 ref.owner = volume;
75 return ref;
76 }

133 /**
134 * @brief Fill unresolved refs of a flow config from the volume.
135 * @param volume Volume being wired.
136 * @param config Flow config, edited in place.
137 *
138 * Velocity, pressure and divergence are always required, so an unnamed one
139 * is declared as "velocity", "pressure" and "divergence". Scratch is
140 * required only when viscosity is above zero. Carried and buoyancy refs
141 * carry the caller's own names and are declared only if the volume does
142 * not hold them.
143 */
144 void resolve(
145 const std::shared_ptr<Buffers::VolumeGridBuffer>& volume,
146 Buffers::VolumeGridBuffer::FlowConfig& config) const
147 {
148 config.velocity = resolve_vector(volume, config.velocity, "velocity");
149 config.pressure = resolve_scalar(volume, config.pressure, "pressure", false);
150 config.divergence = resolve_scalar(volume, config.divergence, "divergence", true);
151
152 if (config.viscosity > 0.0F) {
153 config.scratch = resolve_vector(volume, config.scratch, "scratch");
154 }
155
156 for (auto& carried : config.carried) {
157 carried.field = resolve_scalar(volume, carried.field, {}, false);
158 }
159
160 if (config.buoyancy) {
161 config.buoyancy->temperature = resolve_scalar(volume, config.buoyancy->temperature, {}, false);
162 config.buoyancy->density = resolve_scalar(volume, config.buoyancy->density, {}, false);
163 }
164 }

183 return nullptr;

186 }
187
188 register_graphics_buffer(grid);
189 grid->setup_rendering(render);
190 return grid;

217 [[nodiscard]] bool build_field(
218 const std::shared_ptr<Nodes::Network::NodeNetwork>& network,
219 const Nodes::Network::SpatialFieldConfig& spatial,
220 const FieldBinder& bind,
221 const RenderConfig& render) const
222 {
223 if (!network) {
224 MF_ERROR(Journal::Component::API, Journal::Context::Init,
225 "evolve: null NodeNetwork");
226 return false;
227 }
228
229 if (!has_target(render)) {
230 return false;
231 }
232
233 if (!network->has_operator()) {
234 auto particles = std::dynamic_pointer_cast<Nodes::Network::ParticleNetwork>(network);
235 if (particles && particles->default_operator_enabled()) {
236 particles->create_operator<Nodes::Network::PhysicsOperator>();
237 }
238 }
239
240 const auto* primary = dynamic_cast<const Nodes::Network::GraphicsOperator*>(network->get_operator());
241 if (!primary) {
242 MF_ERROR(Journal::Component::API, Journal::Context::Init,
243 "evolve: network has no primary GraphicsOperator to read a vertex layout from");
244 return false;
245 }
246
247 auto chain = network->get_operator_chain();
248 auto field = chain->find<Nodes::Network::GpuFieldOperator>();
249 if (!field) {
250 field = chain->emplace<Nodes::Network::GpuFieldOperator>(primary->get_vertex_layout(), spatial);
251 } else {
252 MF_WARN(Journal::Component::API, Journal::Context::Init,
253 "evolve: network already carries a GpuFieldOperator, its SpatialFieldConfig is kept");
254 }
255
256 if (bind) {
257 bind(field);
258 }
259
260 register_node_network(network, Nodes::ProcessingToken::VISUAL_RATE);
261
262 auto buffer = std::make_shared<Buffers::NetworkGeometryBuffer>(network);
263 register_graphics_buffer(buffer);
264 buffer->setup_rendering(render);
265 return true;
