53 const float width = x_max - x_min;
54 const float y_min = advanced.min.y;
55 const float y_max = advanced.max.y;

59 const Kinesis::AABB2D hue_rect { .min = { x_min + width * 0.66F, y_min }, .max = { x_min + width * 0.78F, y_max } };

63 glm::vec3 value, std::vector<uint8_t>& out, Element&) {
64 float aspect = 1.F;
65 if (const auto locked = window.lock()) {
66 const auto& state = locked->get_state();
67 if (state.current_width > 0)
68 aspect = static_cast<float>(state.current_height) / static_cast<float>(state.current_width);
69 }
70 const glm::vec2 marker { k_marker_half_height * aspect, k_marker_half_height };
71
72 std::vector<Kakshya::MeshVertex> verts;
73 verts.reserve(static_cast<size_t>(k_square_cells * k_square_cells + k_hue_segments + 6) * 6);
74
75 append_quad(verts, swatch_rect, Kinesis::hsv_to_rgb(value));
76
77 const float cells = static_cast<float>(k_square_cells);
78 for (uint32_t i = 0; i < k_square_cells; ++i) {
79 for (uint32_t j = 0; j < k_square_cells; ++j) {
80 const float s0 = static_cast<float>(i) / cells;
81 const float s1 = static_cast<float>(i + 1) / cells;
82 const float v0 = static_cast<float>(j) / cells;
83 const float v1 = static_cast<float>(j + 1) / cells;
84 const Kinesis::AABB2D cell {
85 .min = { square_rect.min.x + s0 * square_rect.width(), square_rect.min.y + v0 * square_rect.height() },
86 .max = { square_rect.min.x + s1 * square_rect.width(), square_rect.min.y + v1 * square_rect.height() },
87 };
88 append_quad(verts, cell,
89 Kinesis::hsv_to_rgb({ value.x, s0, v0 }),
90 Kinesis::hsv_to_rgb({ value.x, s1, v0 }),
91 Kinesis::hsv_to_rgb({ value.x, s0, v1 }),
92 Kinesis::hsv_to_rgb({ value.x, s1, v1 }));
93 }
94 }
95
96 const float segments = static_cast<float>(k_hue_segments);
97 for (uint32_t k = 0; k < k_hue_segments; ++k) {
98 const float h0 = static_cast<float>(k) / segments;
99 const float h1 = static_cast<float>(k + 1) / segments;
100 const Kinesis::AABB2D segment {
101 .min = { hue_rect.min.x, hue_rect.min.y + h0 * hue_rect.height() },
102 .max = { hue_rect.max.x, hue_rect.min.y + h1 * hue_rect.height() },
103 };
104 const glm::vec3 low = Kinesis::hsv_to_rgb({ h0, 1.F, 1.F });
105 const glm::vec3 high = Kinesis::hsv_to_rgb({ h1, 1.F, 1.F });
106 append_quad(verts, segment, low, low, high, high);
107 }
108
109 const float s = std::clamp(value.y, 0.F, 1.F);
110 const float v = std::clamp(value.z, 0.F, 1.F);
111 append_marker(verts,
112 { square_rect.min.x + s * square_rect.width(), square_rect.min.y + v * square_rect.height() },
113 marker);
114
115 const float hue_y = hue_rect.min.y + std::clamp(value.x, 0.F, 1.F) * hue_rect.height();
116 const glm::vec2 hue_half { hue_rect.width() * 0.5F + marker.x, marker.y * 0.5F };
117 append_marker(verts, { hue_rect.min.x + hue_rect.width() * 0.5F, hue_y }, hue_half);
118
119 Geometry::write_verts(out, verts);
120 };
121
122 const glm::vec3 initial_hsv = Kinesis::rgb_to_hsv(m_initial);
123 auto drawn = internal::atelier().create_element<glm::vec3>(
124 surface, std::move(geometry), initial_hsv, Graphics::PrimitiveTopology::TRIANGLE_LIST);
125
126 auto color_state = std::make_shared<MappedState<glm::vec3>>();
127 color_state->write(m_initial);
128
129 const uint32_t square_id = surface.add(Element {}.with_name("color_square").with_bounds(square_rect));
130 const uint32_t hue_id = surface.add(Element {}.with_name("color_hue").with_bounds(hue_rect));
131 layer_ptr->relate(drawn.element.id, square_id);
132 layer_ptr->relate(drawn.element.id, hue_id);
133
134 auto commit = [hsv_state = drawn.state, color_state, callback = m_on_change](glm::vec3 next) {
135 hsv_state->write(next);
136 const glm::vec3 rgb = Kinesis::hsv_to_rgb(next);
137 color_state->write(rgb);
138 if (callback)
139 callback(rgb);
140 };
141
142 const auto square_fraction = Kinesis::unit_square(square_rect);
143 auto pick_square = [commit, hsv_state = drawn.state, square_fraction](glm::vec2 ndc) {
144 const glm::vec2 fraction = square_fraction(ndc);
145 commit({ hsv_state->value.x, fraction.x, fraction.y });
146 };
147 auto pick_hue = [commit, hsv_state = drawn.state, hue_rect](glm::vec2 ndc) {
148 if (hue_rect.height() <= 0.F)
149 return;
150 const float hue = std::clamp((ndc.y - hue_rect.min.y) / hue_rect.height(), 0.F, 1.F);
151 commit({ hue, hsv_state->value.y, hsv_state->value.z });
152 };
153
154 ctx.on_press(square_id, IO::MouseButtons::Left,
155 [pick_square](uint32_t, glm::vec2 ndc) { pick_square(ndc); });
156 ctx.wire_drag(square_id, pick_square, IO::MouseButtons::Left);
157 ctx.on_press(hue_id, IO::MouseButtons::Left,
158 [pick_hue](uint32_t, glm::vec2 ndc) { pick_hue(ndc); });
159 ctx.wire_drag(hue_id, pick_hue, IO::MouseButtons::Left);
160
161 internal::atelier().bridge().register_element<glm::vec3>(
162 ElementRef { layer_ptr, square_id }, color_state, nullptr);
163
164 picker_id = drawn.element.id;
165 color = std::move(color_state);
166 hsv = drawn.state;
167 picker_bounds = whole;
168 cursor_out = cursor;
169 return *this;
170}
