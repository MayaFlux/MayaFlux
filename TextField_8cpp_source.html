19 return params.atlas ? *params.atlas : Portal::Text::get_default_atlas();

22 /**
23 * @brief Caret's NDC rect for the current cursor, spanning one full
24 * line height on whatever line it falls on.
25 *
26 * Mirrors the pen origin press()/repress() rasterize text at
27 * internally (pen_x=0, pen_y=ascender(), wrap_w=render_bounds.x) so the
28 * caret lines up with the actually-rasterized glyphs, then maps that
29 * pixel rect into field_bounds the same way the text quad's own UV
30 * (0,0)-(1,1) already maps render_bounds onto it.
31 */
32 Kinesis::AABB2D caret_ndc(
33 const Portal::Text::EditableText& edit,
34 const Portal::Text::PressParams& params,
35 Kinesis::AABB2D field_bounds)
36 {
37 auto& atlas = resolve_atlas(params);
38
39 const auto pen = Portal::Text::x_at(
40 edit.text, atlas, edit.cursor,
41 0.F, static_cast<float>(atlas.ascender()),
42 params.render_bounds.x, 4);
43
44 const float top_px = pen.y - static_cast<float>(atlas.ascender());
45 const float bottom_px = top_px + static_cast<float>(atlas.line_height());
46 const glm::vec2 render_bounds(params.render_bounds);
47
48 return Kinesis::AABB2D {
49 .min = {
50 field_bounds.min.x + (pen.x / render_bounds.x) * field_bounds.width(),
51 field_bounds.max.y - (bottom_px / render_bounds.y) * field_bounds.height() },
52 .max = { field_bounds.min.x + ((pen.x + k_caret_width_px) / render_bounds.x) * field_bounds.width(), field_bounds.max.y - (top_px / render_bounds.y) * field_bounds.height() }
53 };
54 }

56 /**
57 * @brief Resubmit the text quad and the caret quad together.
58 *
59 * FormaBuffer::submit() replaces the whole buffer and has no
60 * partial-range write, so every call - keystroke or reposition() -
61 * sends both quads (12 MeshVertex, 720 bytes) even though usually only
62 * one of them actually moved. Caret color reuses params.color (the
63 * text tint) rather than adding a new config field.
64 */
65 void render_geometry(
66 Buffers::FormaBuffer& buf,
67 const Portal::Text::EditableText& edit,
68 const Portal::Text::PressParams& params,
69 Kinesis::AABB2D field_bounds)
70 {
71 const auto text_quad = Kinesis::textured_mesh_rect(field_bounds, 1.F);
72 const auto caret_quad = Kinesis::textured_mesh_rect(
73 caret_ndc(edit, params, field_bounds), 0.F, glm::vec3(params.color));
74
75 std::array<Kakshya::MeshVertex, 12> combined {};
76 for (size_t i = 0; i < text_quad.size(); ++i)
77 combined[i] = text_quad[i];
78 for (size_t i = 0; i < caret_quad.size(); ++i)
79 combined[text_quad.size() + i] = caret_quad[i];
80
81 buf.submit(combined);
82 }

107 /**
108 * @brief Move the cursor to the nearest codepoint boundary one line
109 * above (line_delta = -1) or below (+1) its current line, at
110 * the same x position it currently sits at.
111 *
112 * x_at() gives the cursor's current pen position; index_at() resolves
113 * a point back to a byte offset. Asking it for (current x, current y
114 * shifted by one line_height()) is the whole trick - both already
115 * exist in Portal::Text for other reasons (copy()'s round-trip,
116 * caret_ndc() itself), so nothing new was needed there. A target y
117 * past the first/last line still resolves to the nearest real line,
118 * since that is index_at()'s own behavior for an out-of-range y.
119 */
120 void move_vertical(
121 Portal::Text::EditableText& edit,
122 const Portal::Text::PressParams& params,
123 int line_delta)
124 {
125 auto& atlas = resolve_atlas(params);
126 const auto pen = Portal::Text::x_at(
127 edit.text, atlas, edit.cursor,
128 0.F, static_cast<float>(atlas.ascender()),
129 params.render_bounds.x, 4);
130
131 const float target_y = pen.y + static_cast<float>(line_delta) * static_cast<float>(atlas.line_height());
132 const auto layout = layout_for(edit, params, atlas);
133
134 edit.cursor = Portal::Text::index_at(
135 edit.text, layout, atlas, pen.x, target_y,
136 0.F, static_cast<float>(atlas.ascender()),
137 params.render_bounds.x, 4);
138 }

140 /// @brief Inverse of caret_ndc()'s pixel-to-NDC mapping: an NDC point
141 /// (a mouse click) back into the same pixel space x_at()/
142 /// index_at() work in.
143 glm::vec2 pixel_at(glm::vec2 ndc, Kinesis::AABB2D field_bounds, glm::uvec2 render_bounds)
144 {
145 return {
146 (ndc.x - field_bounds.min.x) / field_bounds.width() * static_cast<float>(render_bounds.x),
147 (field_bounds.max.y - ndc.y) / field_bounds.height() * static_cast<float>(render_bounds.y)
148 };
149 }

151 /// @brief Move the cursor to the codepoint boundary nearest an NDC
152 /// click position.
153 void move_to_click(
154 Portal::Text::EditableText& edit,
155 const Portal::Text::PressParams& params,
156 Kinesis::AABB2D field_bounds,
157 glm::vec2 ndc)
158 {
159 auto& atlas = resolve_atlas(params);
160 const glm::vec2 px = pixel_at(ndc, field_bounds, params.render_bounds);
161 const auto layout = layout_for(edit, params, atlas);
162
163 edit.cursor = Portal::Text::index_at(
164 edit.text, layout, atlas, px.x, px.y,
165 0.F, static_cast<float>(atlas.ascender()),
166 params.render_bounds.x, 4);
167 }

171TextField& TextField::place(
172 std::shared_ptr<Buffers::FormaBuffer> in_buf,
173 Surface& surface,
174 Kinesis::AABB2D bounds,
175 std::shared_ptr<Portal::Text::PressParams> in_params,
176 std::string initial_text)
177{
178 buf = std::move(in_buf);
179 field_bounds = bounds;
180 m_live_bounds = std::make_shared<Kinesis::AABB2D>(bounds);
181
182 params = std::move(in_params);
183 if (!params)
184 params = std::make_shared<Portal::Text::PressParams>();
185
186 m_element = Element {}
187 .with_name("text_field")
188 .with_bounds(bounds)
189 .with_buffer(buf)
190 .with_text(initial_text, *params, bounds);
191
192 element_id = surface.layer().add(m_element);
193 m_element.id = element_id;
194
195 state = std::make_shared<MappedState<Portal::Text::EditableText>>();
196 state->value.cursor = initial_text.size();
197 state->value.text = std::move(initial_text);
198
199 render_geometry(*buf, state->value, *params, bounds);
200
201 wire(surface.ctx());
202
203 return *this;
204}

285TextField& TextField::scrollable(
286 std::shared_ptr<Buffers::FormaBuffer> in_buf,
287 Surface& surface,
288 Scrollable& scroller,
289 std::shared_ptr<Portal::Text::PressParams> in_params,
290 std::string initial_text,
291 float content_multiplier)
292{
293 const Kinesis::AABB2D viewport = scroller.bounds();
294 const Kinesis::AABB2D content_bounds {
295 .min = glm::vec2(viewport.min.x, viewport.max.y - viewport.height() * content_multiplier),
296 .max = glm::vec2(viewport.max.x, viewport.max.y)
297 };
298
299 place(std::move(in_buf), surface, content_bounds, std::move(in_params), std::move(initial_text));
300
301 TextField field = *this;
302 scroller.track(
303 surface.layer(), element_id, content_bounds,
304 [field](Kinesis::AABB2D shifted) mutable { field.reposition(shifted); },
305 buf);
306
307 surface.layer().relate(element_id, scroller.viewport_id);
308 surface.ctx().on_scroll(element_id, scroller.wheel_handler(surface));
309
310 return *this;
311}
