21/**
22 * @brief Target that feeds an influence of fields into an operator's vertex attributes.
23 * @tparam Field Kinesis::VectorField, SpatialField or UVField, matching @p target.
24 * @param op FieldOperator, as a raw pointer from create_operator or a shared_ptr.
25 * @param target Vertex attribute mask the field drives.
26 *
27 * Binds one forwarding field to @p op now, and every time the influence runs the
28 * target replaces what that field forwards to. Other fields bound to the same
29 * target are left alone, which unbinding and rebinding each tick would not do.
30 * The operator is not retained. The influence and the operator must run on the
31 * same tick domain.
32 */
33template <typename Field, typename Operator>
34[[nodiscard]] typename Influence<Field>::Target bind_field(
35 const Operator& op,
36 Kinesis::FieldTarget target)
37{
38 auto cell = std::make_shared<Field>();
39 op->bind(target, detail::forwarding(cell));
40 return [cell](const Field& field) { *cell = field; };
41}

87/**
88 * @brief Target that feeds an influence of transform functions into one slot.
89 * @param op InstanceFieldOperator or MeshTransformOperator, raw pointer or shared_ptr.
90 * @param slot Slot the function drives.
91 *
92 * The influence's value is the operator's own TransformField: for an
93 * InstanceFieldOperator a function from the slot's current transform to its next
94 * one, for a MeshTransformOperator a function from accumulated seconds to the
95 * slot's local transform.
96 */
97template <typename Operator>
98[[nodiscard]] auto bind_transform(const Operator& op, uint32_t slot)
99{
100 using Transform = typename std::remove_cvref_t<decltype(*op)>::TransformField;
101
102 auto cell = std::make_shared<Transform>();
103 if constexpr (requires { op->bind_transform(slot, std::declval<Transform>()); }) {
104 op->bind_transform(slot, [cell](const glm::mat4& current) {
105 return *cell ? (*cell)(current) : current;
106 });
107 } else {
108 op->bind(slot, [cell](float seconds) {
109 return *cell ? (*cell)(seconds) : glm::mat4(1.0F);
110 });
111 }
112 return typename Influence<Transform>::Target { [cell](const Transform& field) { *cell = field; } };
113}

138/**
139 * @brief Target that pushes an influence's value to a GPU executor as push constants.
140 * @tparam T Trivially copyable struct matching the shader's push constant layout.
141 * @param target An operator with push_constants (MeshFieldOperator,
142 * InstanceFieldOperator) or a shader processor with set_push_constant_data;
143 * raw pointer or shared_ptr, retained by the target.
144 *
145 * For an operator this arms the next dispatch of its executor. This is how an
146 * Agent's state reaches a slot-transform kernel, for example one assembled from
147 * a DualField of mat4 through ShaderSpec::Assemble::function(). For a shader
148 * processor the push block is grown to hold T at @p offset here, so a shader
149 * that already declares it is not drawn before the block exists. A shader that
150 * keeps push constants of its own passes the byte offset after them and
151 * declares both in one block.
152 */
153template <typename T, typename Receiver>
154[[nodiscard]] typename Influence<T>::Target bind_push_constants(const Receiver& target, size_t offset = 0)
155{
156 if constexpr (requires { target->get_push_constant_data(); }) {
157 if (target->get_push_constant_data().size() < offset + sizeof(T)) {
158 target->set_push_constant_size(offset + sizeof(T));
159 }
160 }
161
162 return [target, offset](const T& data) {
163 if constexpr (requires { target->push_constants(data); }) {
164 target->push_constants(data);
165 } else {
166 if (target->get_push_constant_data().size() < offset + sizeof(T)) {
167 target->set_push_constant_size(offset + sizeof(T));
168 }
169 std::memcpy(target->get_push_constant_data().data() + offset, &data, sizeof(T));
170 }
171 };
172}
