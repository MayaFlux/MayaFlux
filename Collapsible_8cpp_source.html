22 [[nodiscard]] GeometryFn<bool> collapsible_header_geom(
23 float y_top,
24 float x_min,
25 float x_max,
26 float row_h,
27 glm::vec3 color_closed,
28 glm::vec3 color_open)
29 {
30 return [y_top, x_min, x_max, row_h, color_closed, color_open](
31 bool open, std::vector<uint8_t>& out, Element& el) {
32 const float bot = y_top - row_h;
33 const glm::vec3 col = open ? color_open : color_closed;
34
35 const size_t stride = Kakshya::VertexLayout::for_meshes().stride_bytes;
36 out.assign(static_cast<size_t>(6) * stride, 0);
37
38 const glm::vec3 bl { x_min, bot, 0.F };
39 const glm::vec3 br { x_max, bot, 0.F };
40 const glm::vec3 tl { x_min, y_top, 0.F };
41 const glm::vec3 tr { x_max, y_top, 0.F };
42
43 auto write = [&](size_t idx, glm::vec3 pos) {
44 auto* v = out.data() + idx * stride;
45 std::memcpy(v, &pos, 12);
46 std::memcpy(v + 12, &col, 12);
47 };
48
49 write(0, bl);
50 write(1, br);
51 write(2, tl);
52 write(3, br);
53 write(4, tr);
54 write(5, tl);
55
56 el.bounds_hint = Kinesis::AABB2D {
57 .min = { x_min, bot },
58 .max = { x_max, y_top },
59 };
60 };
61 }

63 /// @brief A fully-textured TRIANGLE_LIST quad, weight=1 so forma_multi.frag
64 /// samples textures[0]. Matches Entry.cpp's own helper of the same
65 /// shape; not shared across files, same as Element.cpp's private
66 /// textured_mesh_rect isn't either.
67 std::array<Kakshya::MeshVertex, 6> textured_quad(Kinesis::AABB2D region)

91 LayoutCursor& cursor,

95
96 const Kinesis::AABB2D advanced = cursor.advance(row_h);

101 open_state->write(m_initially_open);
102
103 uint32_t hid = 0;
104
105 if (m_label_text.empty()) {
106 Mapped<bool> mapped;
107 mapped.state = open_state;
108 mapped.geometry_fn = collapsible_header_geom(y_top, x_min, x_max, row_h, m_color_closed, m_color_open);
109 mapped.element.buffer = buf;
110 mapped.element.bounds_hint = row_rect;
111
112 hid = surface.layer().add(mapped.element);
113 mapped.element.id = hid;
114 open_state->id = hid;
115 mapped.sync();
116
117 surface.ctx().on_press(hid, IO::MouseButtons::Left,
118 [m = std::move(mapped), open = open_state, surface, hid](uint32_t, glm::vec2) mutable {
119 const bool next = !open->value;
120 open->write(next);
121 for (auto rel_id : surface.layer().related_ids(hid))
122 surface.layer().set_visible(rel_id, next);
123 m.sync();
124 });
125 } else {
126 buf->submit(textured_quad(row_rect));
127
128 const glm::uvec2 dims = row_pixel_dims(surface.window(), x_min, x_max, row_h);
129 auto staging = Buffers::create_image_staging_buffer(
130 static_cast<size_t>(dims.x) * dims.y * 4 * k_staging_margin);
131
132 auto image_slot = Portal::Text::press(m_label_text, dims,
133 { .color = m_label_color,
134 .background = { m_initially_open ? m_color_open : m_color_closed, 1.F } },
135 staging);
136 buf->bind_texture(0, image_slot);
137
138 Element el;
139 el.buffer = buf;
140 el.bounds_hint = row_rect;
141
142 hid = surface.layer().add(el);
143 open_state->id = hid;
144
145 const std::string label_text = m_label_text;
146 const glm::vec4 label_color = m_label_color;
147 const glm::vec3 color_closed = m_color_closed;
148 const glm::vec3 color_open = m_color_open;
149
150 surface.ctx().on_press(hid, IO::MouseButtons::Left,
151 [buf = buf, open = open_state, surface, hid, staging,
152 label_text, label_color, color_closed, color_open](uint32_t, glm::vec2) mutable {
153 const bool next = !open->value;
154 open->write(next);
155 for (auto rel_id : surface.layer().related_ids(hid))
156 surface.layer().set_visible(rel_id, next);
157 });
158
159 surface.ctx().on_resize(hid,
160 [buf = buf, surface, staging, open = open_state,
161 label_text, label_color, color_closed, color_open,
162 x_min, x_max, row_h](uint32_t, uint32_t) {
163 const glm::uvec2 new_dims = row_pixel_dims(surface.window(), x_min, x_max, row_h);
164 const size_t required_bytes = static_cast<size_t>(new_dims.x) * new_dims.y * 4;
165
166 if (required_bytes > staging->get_size_bytes()) {
167 staging->resize(required_bytes * k_staging_margin, false);
168 }
169
170 auto image = Portal::Text::press(
171 label_text, new_dims,
172 { .color = label_color,
173 .background = { open->value ? color_open : color_closed, 1.F } },
174 staging);
175 buf->bind_texture(0, image);
176 });
177 }
178
179 header_bounds = row_rect;
180 header_id = hid;
181 open = open_state;
182 cursor_out = cursor;
183 return *this;
184}
