11/**
12 * @class CompositeContainer
13 * @brief Owns composite data and a bounded, independently owned window.
14 *
15 * Elements share one CompositeLayout and remain packed in the two NDData-backed
16 * stores of CompositeArray. Insertion, slices, and traversals operate directly
17 * on that array. materialize_next() copies a bounded selection into a second
18 * CompositeArray, rebasing selected text offsets into its own text store.
19 * Each traversal owns its own position and window size.
20 *
21 * Specializes NDDataContainer<CompositeArray> directly, not
22 * SignalSourceContainer, so it carries no ProcessingState and no
23 * DataProcessor attach/detach lifecycle. On-demand, non-recurring access
24 * goes through DataFieldTraversal<CompositeArray> instead, obtained via
25 * traverse().
26 *
27 * Usage:
28 * @code
29 * CompositeLayout layout;
30 * layout.add_field<std::string>("name");
31 * layout.add_field<double>("score");
32 * CompositeContainer container(std::move(layout), 64);
33 * auto insertion = container.insert();
34 * auto row = insertion.append();
35 * insertion.set_text(row, "name", "Ada");
36 * insertion.set(row, "score", 0.95);
37 *
38 * auto traversal = container.traverse(32);
39 * auto scores = traversal.window_to_nddata<double>("score");
40 * container.materialize_next();
41 * auto first = container.get_materialized_data().at(0);
42 * @endcode
43 *
44 * @note Views and traversal objects borrow the source array. Changes to
45 * source storage can invalidate them according to CompositeArray's rules.
46 */
