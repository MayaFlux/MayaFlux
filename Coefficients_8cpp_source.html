34 [[nodiscard]] BiquadTerms terms(double frequency, double q, double sample_rate)

72 [[nodiscard]] std::vector<double> lowpass_kernel(double frequency, double sample_rate, size_t taps)
73 {
74 const double nyquist = sample_rate * 0.5;
75 const double f = std::clamp(frequency, 1.0, nyquist - 1.0);
76 const double fc = f / sample_rate;
77 const double centre = static_cast<double>(taps - 1) * 0.5;
78
79 std::vector<double> h(taps);
80 for (size_t i = 0; i < taps; ++i)
81 h[i] = 2.0 * fc * sinc(2.0 * fc * (static_cast<double>(i) - centre));
82
83 apply_hann(h);
84
85 double total = 0.0;
86 for (double v : h)
87 total += v;
88 if (std::abs(total) > 1e-12) {
89 for (double& v : h)
90 v /= total;
91 }
92 return h;
93 }

323std::vector<double> savitzky_golay(size_t window, size_t poly_order, size_t derivative)
324{
325 const size_t n = force_odd(window);
326 if (n <= poly_order || derivative > poly_order)
327 return {};
328
329 const auto rows = static_cast<Eigen::Index>(n);
330 const auto cols = static_cast<Eigen::Index>(poly_order) + 1;
331 const double half = static_cast<double>(n - 1) * 0.5;
332
333 Eigen::MatrixXd vandermonde(rows, cols);
334 for (Eigen::Index i = 0; i < rows; ++i) {
335 const double z = static_cast<double>(i) - half;
336 double p = 1.0;
337 for (Eigen::Index j = 0; j < cols; ++j) {
338 vandermonde(i, j) = p;
339 p *= z;
340 }
341 }
342
343 const Eigen::MatrixXd pseudo = vandermonde
344 .completeOrthogonalDecomposition()
345 .pseudoInverse();
346
347 double factorial = 1.0;
348 for (size_t k = 2; k <= derivative; ++k)
349 factorial *= static_cast<double>(k);
350
351 const Eigen::VectorXd row = pseudo.row(static_cast<Eigen::Index>(derivative)) * factorial;
352
353 std::vector<double> h(n);
354 for (size_t i = 0; i < n; ++i)
355 h[i] = row(static_cast<Eigen::Index>(n - 1 - i));
356
357 return h;
358}

367void lagrange_weights(double delay, std::span<double> weights) noexcept
368{
369 const size_t taps = weights.size();
370 if (taps == 0)
371 return;
372
373 const double d = std::clamp(delay, 0.0, static_cast<double>(taps - 1));
374
375 if (taps > reciprocal_factorial.size()) {
376 for (size_t k = 0; k < taps; ++k) {
377 weights[k] = 1.0;
378 for (size_t j = 0; j < taps; ++j) {
379 if (j == k)
380 continue;
381 weights[k] *= (d - static_cast<double>(j))
382 / (static_cast<double>(k) - static_cast<double>(j));
383 }
384 }
385 return;
386 }
387
388 double before = 1.0;
389 for (size_t k = 0; k < taps; ++k) {
390 weights[k] = before;
391 before *= d - static_cast<double>(k);
392 }
393
394 double after = 1.0;
395 for (size_t k = taps; k-- > 0;) {
396 const size_t rest = taps - 1 - k;
397 const double scale = reciprocal_factorial[k] * reciprocal_factorial[rest];
398 weights[k] *= after * (rest % 2 == 0 ? scale : -scale);
399 after *= d - static_cast<double>(k);
400 }
401}

431std::vector<double> cascade(std::span<const double> lhs, std::span<const double> rhs)
432{
433 if (lhs.empty() || rhs.empty())
434 return {};
435
436 std::vector<double> out(lhs.size() + rhs.size() - 1, 0.0);
437 for (size_t i = 0; i < lhs.size(); ++i) {
438 for (size_t j = 0; j < rhs.size(); ++j)
439 out[i + j] += lhs[i] * rhs[j];

444double max_pole_magnitude(std::span<const double> a)
445{
446 size_t order = a.size();
447 while (order > 1 && a[order - 1] == 0.0)
448 --order;
449
450 if (order <= 1)
451 return 0.0;
452
453 const double a0 = a[0];
454
455 if (order == 2)
456 return std::abs(a[1] / a0);
457
458 if (order == 3) {
459 const double p = a[1] / a0;
460 const double q = a[2] / a0;
461 const double disc = p * p - 4.0 * q;
462 if (disc >= 0.0) {
463 const double r = std::sqrt(disc);
464 return std::max(std::abs((-p + r) * 0.5), std::abs((-p - r) * 0.5));
465 }
466 return std::sqrt(q);
467 }
468
469 const Eigen::Index n = static_cast<Eigen::Index>(order) - 1;
470 Eigen::MatrixXd companion = Eigen::MatrixXd::Zero(n, n);
471 for (Eigen::Index i = 0; i < n; ++i)
472 companion(0, i) = -a[static_cast<size_t>(i) + 1] / a0;
473 for (Eigen::Index i = 1; i < n; ++i)
474 companion(i, i - 1) = 1.0;
475
476 Eigen::EigenSolver<Eigen::MatrixXd> solver(companion, false);
477 double largest = 0.0;
478 for (Eigen::Index i = 0; i < solver.eigenvalues().size(); ++i)
479 largest = std::max(largest, std::abs(solver.eigenvalues()(i)));
480 return largest;
481}
