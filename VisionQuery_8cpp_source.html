10 components.export_boxes = true;

12 VisionSequence::Builder builder;
13 switch (r.segmentation.space) {
14 case SegmentationSpace::Gray:
15 builder.rgba_to_gray();
16 break;
17 case SegmentationSpace::Hsv:
18 builder.rgba_to_hsv();
19 break;
20 case SegmentationSpace::Rgba:
21 break;
22 }
23
24 std::visit([&builder](const auto& method) {
25 using T = std::decay_t<decltype(method)>;
26 if constexpr (std::is_same_v<T, OtsuParams>) {
27 builder.threshold_otsu(method);
28 } else if constexpr (std::is_same_v<T, ThresholdAdaptiveParams>) {
29 builder.threshold_adaptive(method);
30 } else {
31 builder.threshold(method);
32 }
33 },
34 r.segmentation.method);
35
36 return builder
37 .connected_components(components)
38 .find_contours(r.contours)
39 .build();

112 /**
113 * @brief True when an op writes the executor's shared output images.
114 *
115 * Those images rotate between two per size, so anything a later op of this
116 * kind writes can land on an image an earlier step handed out. Only the
117 * ops below run on other contexts.
118 */
119 bool overwrites_shared_images(VisionOp op)
120 {
121 switch (op) {
122 case VisionOp::ExtractPeaks:
123 case VisionOp::TrackKeypoints:
124 case VisionOp::OpticalFlowDense:
125 case VisionOp::Snapshot:
126 return false;
127 default:
128 return true;
129 }

132 /** @brief True when no step from index from onward overwrites the shared output images. */
133 bool preserves_shared_images(const std::vector<VisionStep>& steps, size_t from)
134 {
135 return std::none_of(steps.begin() + static_cast<std::ptrdiff_t>(std::min(from, steps.size())), steps.end(),
136 [](const VisionStep& s) { return overwrites_shared_images(s.op); });
137 }

139 /**
140 * @brief Leading steps identical to reference's own, up to the first
141 * divergence or unseedable step.
142 *
143 * Sharing hands a later sequence the image the reference produced at the
144 * end of the prefix, so it is only allowed when that image survives:
145 * neither sequence may run a step that overwrites the shared images after
146 * the prefix, or the seed would already be gone. Otherwise nothing is
147 * shared and the candidate recomputes its own prefix.
148 */
149 size_t shared_prefix_length(
150 const std::vector<VisionStep>& reference,
151 const std::vector<VisionStep>& candidate)
152 {
153 size_t n = 0;
154 while (n < reference.size() && n < candidate.size()) {
155 if (!seed_accessor_for(candidate[n].op))
156 break;
157 if (hash_vision_step(reference[n].op, reference[n].params)
158 != hash_vision_step(candidate[n].op, candidate[n].params))
159 break;
160 ++n;
161 }
162
163 if (n > 0 && !(preserves_shared_images(reference, n) && preserves_shared_images(candidate, n)))
164 return 0;
165 return n;
166 }

170std::vector<ResolvedSequence> resolve(const VisionQuery& query)
171{
172 std::vector<VisionStep> flow_lane;
173 std::vector<std::vector<VisionStep>> other_chains;
174
175 auto append_to_flow_lane = [&](VisionSequence chain) {
176 auto steps = std::move(chain.steps);
177 if (!flow_lane.empty() && !steps.empty() && steps.front().op == VisionOp::RgbaToGray)
178 steps.erase(steps.begin());
179 flow_lane.insert(flow_lane.end(),
180 std::make_move_iterator(steps.begin()),
181 std::make_move_iterator(steps.end()));
182 };
183
184 if (has_flag(query.intents, VisionIntent::TrackObjects) && query.track_objects)
185 append_to_flow_lane(track_objects_chain(*query.track_objects));
186
187 if (has_flag(query.intents, VisionIntent::EstimateMotion) && query.estimate_motion)
188 append_to_flow_lane(estimate_motion_chain(*query.estimate_motion));
189
190 if (has_flag(query.intents, VisionIntent::FindElements) && query.find_elements)
191 other_chains.push_back(find_elements_chain(*query.find_elements).steps);
192
193 if (has_flag(query.intents, VisionIntent::DetectFeatures) && query.detect_features)
194 other_chains.push_back(detect_features_chain(*query.detect_features).steps);
195
196 if (has_flag(query.intents, VisionIntent::MeasureAppearance) && query.measure_appearance)
197 other_chains.push_back(measure_appearance_chain(*query.measure_appearance).steps);
198
199 if (has_flag(query.intents, VisionIntent::DetectEdges) && query.detect_edges)
200 other_chains.push_back(detect_edges_chain(*query.detect_edges).steps);
201
202 std::vector<ResolvedSequence> sequences;
203 sequences.reserve(1 + other_chains.size());
204 std::optional<size_t> reference_index;
205
206 if (!flow_lane.empty()) {
207 sequences.push_back({ .sequence = VisionSequence { .steps = std::move(flow_lane) } });
208 reference_index = 0;
209 }
210
211 std::vector<ResolvedSequence> seeded;
212 std::vector<ResolvedSequence> unseeded;
213
214 for (auto& steps : other_chains) {
215 if (!reference_index) {
216 sequences.push_back({ .sequence = VisionSequence { .steps = std::move(steps) } });
217 reference_index = 0;
218 continue;
219 }
220
221 const auto& reference = sequences[*reference_index].sequence.steps;
222 ResolvedSequence resolved;
223 const size_t shared = shared_prefix_length(reference, steps);
224 if (shared > 0) {
225 resolved.seed_lane = *reference_index;
226 resolved.seed_field = seed_accessor_for(reference[shared - 1].op);
227 steps.erase(steps.begin(), steps.begin() + static_cast<std::ptrdiff_t>(shared));
228 }
229 resolved.sequence = VisionSequence { .steps = std::move(steps) };
230 (resolved.seed_field ? seeded : unseeded).push_back(std::move(resolved));
231 }
232
233 sequences.insert(sequences.end(), std::make_move_iterator(seeded.begin()), std::make_move_iterator(seeded.end()));
234 sequences.insert(sequences.end(), std::make_move_iterator(unseeded.begin()), std::make_move_iterator(unseeded.end()));
235
236 return sequences;
237}
