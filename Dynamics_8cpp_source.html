7double pulse_response(
8 double t, double onset, double duration, double rise, double fall, double period) noexcept
9{
10 double local = t - onset;
11 if (local < 0.0)
12 return 0.0;
13
14 double pulse = std::max(duration, 0.0);
15 if (period > 0.0) {
16 local -= period * std::floor(local / period);
17 pulse = std::min(pulse, period);
18 }
19
20 const auto peak_at = [rise](double s) {
21 return rise > 0.0 ? 1.0 - std::exp(-s / rise) : 1.0;
22 };
23
24 if (local <= pulse)
25 return peak_at(local);
26
27 if (fall <= 0.0)
28 return 0.0;
29 return peak_at(pulse) * std::exp(-(local - pulse) / fall);
30}
