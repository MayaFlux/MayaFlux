10 /**
11 * @brief Build the compute shader that evaluates @p field at every grid corner.
12 *
13 * Every push constant is a 4-byte scalar, so the block has no padding and
14 * matches the host struct byte for byte.
15 */
16 [[nodiscard]] Portal::Graphics::ShaderSpec field_spec(const Kinesis::FieldSource& field)
17 {
18 if (!field.valid() || field.parameter_count() > 3) {
19 error<std::invalid_argument>(
20 Journal::Component::Buffers,
21 Journal::Context::BufferProcessing,
22 std::source_location::current(),
23 "SDFFieldProcessor: the field did not parse into a function of "
24 "one to three parameters. It needs a trailing return type and a body.");
25 }
26
27 using Kakshya::GpuDataFormat;
28
29 Portal::Graphics::ShaderSpec::Assemble assemble;
30 assemble.ssbo("sdf_grid", Portal::Graphics::BindingDirection::Output, GpuDataFormat::FLOAT32);
31
32 for (const char* name : { "bounds_min_x", "bounds_min_y", "bounds_min_z", "time", "step_x", "step_y", "step_z" }) {
33 assemble.pc(name);
34 }
35 for (const char* name : { "res_x", "res_y", "res_z" }) {
36 assemble.pc(name, GpuDataFormat::UINT32);
37 }
38 for (const char* name : { "p0", "p1", "p2", "p3" }) {
39 assemble.pc(name);
40 }
41
42 assemble.workgroup(64)
43 .function(field.return_type, field.name, field.params, field.body);
44
45 std::string body;
46 body += " uint nx = res_x + 1u;\n";
47 body += " uint ny = res_y + 1u;\n";
48 body += " if (i >= nx * ny * (res_z + 1u)) { return; }\n";
49 body += " uint ix = i % nx;\n";
50 body += " uint iy = (i / nx) % ny;\n";
51 body += " uint iz = i / (nx * ny);\n";
52 body += " vec3 p = vec3(bounds_min_x + float(ix) * step_x,"
53 " bounds_min_y + float(iy) * step_y,"
54 " bounds_min_z + float(iz) * step_z);\n";
55
56 switch (field.parameter_count()) {
57 case 1:
58 body += " sdf_grid[i] = " + field.name + "(p);\n";
59 break;
60 case 2:
61 body += " sdf_grid[i] = " + field.name + "(p, time);\n";
62 break;
63 default:
64 body += " vec4 params = vec4(p0, p1, p2, p3);\n";
65 body += " sdf_grid[i] = " + field.name + "(p, time, params);\n";
66 break;
67 }
68
69 assemble.kernel(Portal::Graphics::KernelSource { .body = std::move(body) });
70 return assemble.build();
71 }
