50void Chimera::cut(size_t layer, double seconds)
51{
52 if (!m_state) {
53 return;
54 }
55
56 auto& target = m_state->layers.at(layer);
57 if (!target.ring && !target.collection) {
58 return;
59 }
60
61 if (target.lag) {
62 const double elapsed = std::max(
63 static_cast<double>(m_state->ticks) / m_state->frame_rate - target.delay, 0.0);
64 target.lag_offset = seconds - (*target.lag)(elapsed);
65 } else if (target.ring) {
66 target.position = static_cast<double>(target.ring->get_write_head())
67 - seconds * target.ring->get_frame_rate();
68 target.entered = true;
69 } else {
70 const double rate = target.collection_rate > 0.0 ? target.collection_rate : m_state->frame_rate;
71 target.position = static_cast<double>(target.collection->get_write_head()) - seconds * rate;
72 target.entered = true;
73 }

146void Chimera::tick(State& state, uint64_t frame)
147{
148 state.ticks = frame + 1;
149 const double now = static_cast<double>(frame) / state.frame_rate;
150
151 uint32_t index = 0;
152 for (auto& layer : state.layers) {
153 const uint32_t current = index++;
154
155 if (now < layer.delay) {
156 continue;
157 }
158
159 const auto age = static_cast<float>(now - layer.delay);
160
161 if (layer.picture) {
162 if (layer.dirty && state.buffer->submit_layer(current, *layer.picture)) {
163 layer.dirty = false;
164 stamp(state, current, glm::vec4(age, 0.0F, 0.0F, 1.0F));
165 }
166 continue;
167 }
168
169 if (frame % layer.interval != 0) {
170 continue;
171 }
172
173 if (layer.image) {
174 if (state.buffer->submit_layer(current, layer.image)) {
175 stamp(state, current, glm::vec4(age, 0.0F, 0.0F, 1.0F));
176 }
177 } else if (layer.ring) {
178 feed(state, layer, current, now - layer.delay);
179 } else if (layer.collection) {
180 collect(state, layer, current, now - layer.delay);
181 }
182 }

185void Chimera::place(const State& state, Layer& layer, uint64_t head, double rate, double nearest, double elapsed)
186{
187 const double latest = static_cast<double>(head) - nearest;
188
189 if (layer.lag) {
190 const double behind = ((*layer.lag)(elapsed) + layer.lag_offset) * rate;
191 layer.position = static_cast<double>(head) - std::max(behind, nearest);
192 } else if (!layer.entered) {
193 layer.position = latest;
194 } else {
195 layer.position = std::min(layer.position + *layer.ratio * rate / state.frame_rate, latest);
196 }
197 layer.entered = true;
198}

200void Chimera::collect(State& state, Layer& layer, uint32_t index, double elapsed)
201{
202 const auto& collection = layer.collection;
203 const uint64_t head = collection->get_write_head();
204 if (head == 0) {
205 return;
206 }
207
208 const double rate = layer.collection_rate > 0.0 ? layer.collection_rate : state.frame_rate;
209 place(state, layer, head, rate, 1.0, elapsed);
210
211 const auto frame = static_cast<uint64_t>(std::max(std::floor(layer.position), 0.0));
212 const uint64_t ring = collection->get_spec().ring_layers;
213 if (frame >= head || (ring > 0 && frame + ring < head)) {
214 if (!layer.missed) {
215 MF_WARN(Journal::Component::Kriya, Journal::Context::CoroutineScheduling,
216 "Chimera layer {} is outside the collection, holding its last frame", index);
217 layer.missed = true;
218 }
219 return;
220 }
221 layer.missed = false;
222
223 const auto slot = static_cast<uint32_t>(ring > 0 ? frame % ring : frame);
224 if (state.buffer->submit_layer(index, collection->to_image(slot))) {
225 stamp(state, index,
226 glm::vec4(static_cast<float>(elapsed), static_cast<float>(static_cast<double>(head) - layer.position),
227 static_cast<float>(layer.position), 1.0F));
228 }

231void Chimera::feed(State& state, Layer& layer, uint32_t index, double elapsed)
232{
233 const auto& ring = layer.ring;
234 const uint64_t head = ring->get_write_head();
235 if (head < 3) {
236 return;
237 }
238
239 place(state, layer, head, ring->get_frame_rate(), 2.0, elapsed);
240
241 const double whole = std::max(std::floor(layer.position), 0.0);
242 const auto frame = static_cast<uint64_t>(whole);
243 if (!ring->is_frame_available(frame)) {
244 if (!layer.missed) {
245 MF_WARN(Journal::Component::Kriya, Journal::Context::CoroutineScheduling,
246 "Chimera layer {} is outside the ring, holding its last frame", index);
247 layer.missed = true;
248 }
249 return;
250 }
251 layer.missed = false;
252
253 std::span<const uint8_t> pixels = ring->get_frame_pixels(frame);
254 if (pixels.empty()) {
255 return;
256 }
257
258 const double fraction = layer.position - whole;
259 if (layer.smooth && fraction > 0.0 && ring->is_frame_available(frame + 1)) {
260 const auto next = ring->get_frame_pixels(frame + 1);
261 layer.scratch.resize(pixels.size());
262 if (!next.empty()
263 && Kakshya::blend_frames(ring->get_format(), pixels, next, layer.scratch.data(), fraction)) {
264 pixels = std::span<const uint8_t>(layer.scratch);
265 }
266 }
267
268 const bool same_extent = ring->get_width() == state.buffer->get_width()
269 && ring->get_height() == state.buffer->get_height()
270 && ring->get_format() == state.buffer->get_format();
271
272 bool stored = false;
273 if (same_extent) {
274 stored = state.buffer->submit_layer(index, pixels);
275 } else if (auto image = Kakshya::ImageData::from_bytes(
276 ring->get_width(), ring->get_height(),
277 Portal::Graphics::TextureLoom::get_channel_count(ring->get_format()),
278 ring->get_format(), pixels)) {
279 stored = state.buffer->submit_layer(index, *image);
280 }
281
282 if (stored) {
283 stamp(state, index,
284 glm::vec4(static_cast<float>(elapsed), static_cast<float>(static_cast<double>(head) - layer.position),
285 static_cast<float>(layer.position), 1.0F));
286 }

470Chimera ChimeraBuilder::start()
471{
472 Chimera set;
473
474 if (!m_buffer || m_layers.empty()) {
475 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
476 "ChimeraBuilder::start needs a buffer and at least one layer");
477 return set;
478 }
479
480 if (m_layers.size() > m_buffer->get_layer_count()) {
481 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
482 "ChimeraBuilder::start declared {} layers, the buffer holds {}",
483 m_layers.size(), m_buffer->get_layer_count());
484 return set;
485 }
486
487 if (!m_pipeline) {
488 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
489 "ChimeraBuilder::start needs a pipeline to feed the layers");
490 return set;
491 }
492
493 for (auto& layer : m_layers) {
494 if (!layer.piped) {
495 continue;
496 }
497
498 layer.ring = m_pipeline->get_graphics_stream();
499 if (!layer.ring) {
500 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
501 "ChimeraBuilder::start found no stream in the pipeline for from_pipeline");
502 return set;
503 }
504 }
505
506 auto state = std::make_shared<Chimera::State>();
507
508 state->buffer = m_buffer;
509 state->pipeline = m_pipeline;
510 state->layers = m_layers;
511 state->frame_rate = std::max(static_cast<double>(Vruta::s_registered_frame_rate), 1.0);
512
513 uint32_t index = 0;
514 for (const auto& layer : state->layers) {
515 m_buffer->set_weight(index, layer.level);
516 if (layer.fit) {
517 m_buffer->set_layer_fit(index, layer.fit);
518 }
519 if (layer.params) {
520 m_buffer->set_layer_params(index, *layer.params);
521 }
522 ++index;
523 }
524
525 if (m_render) {
526 m_buffer->setup_rendering(*m_render);
527 m_render.reset();
528 }
529
530 const std::weak_ptr<Chimera::State> weak = state;
531 m_pipeline >> BufferOperation::dispatch_to(
532 [weak](Kakshya::DataVariant&, uint32_t cycle) {
533 if (const auto running = weak.lock()) {
534 Chimera::tick(*running, cycle);
535 }
536 },
537 Buffers::ProcessingToken::GRAPHICS_BACKEND);
538
539 for (const auto& action : m_actions) {
540 const auto cycles = static_cast<uint32_t>(std::max(std::round(action.seconds * state->frame_rate), 1.0));
541
542 auto timed = BufferOperation::dispatch_to(
543 [weak, run = action.run](Kakshya::DataVariant&, uint32_t cycle) {
544 const auto running = weak.lock();
545 if (!running || cycle == 0) {
546 return;
547 }
548
549 Chimera view;
550 view.m_state = running;
551 run(view);
552 },
553 Buffers::ProcessingToken::GRAPHICS_BACKEND);
554 timed.every_n_cycles(cycles);
555
556 m_pipeline >> std::move(timed);
557 }
558
559 m_pipeline->execute_frame_rate();
560
561 set.m_guard = std::shared_ptr<void>(nullptr, [state](void*) { state->halt(); });
562 set.m_state = std::move(state);
563 return set;
564}
