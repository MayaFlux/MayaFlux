8/**
9 * @struct CompositeProjection
10 * @brief Owning homogeneous NDData projection of one Composite field.
11 *
12 * Presence contains one uint8_t per element: 1 for present, 0 for absent.
13 * Values for absent elements are zero-filled and must be interpreted using
14 * the presence variant. The dimension describes the projected field as a
15 * one-dimensional TENSOR_ND value sequence.
16 *
17 * Usage:
18 * @code
19 * auto projection = items.to_nddata<double>("score");
20 * const auto& scores = std::get<std::vector<double>>(projection->values);
21 * const auto& present = std::get<std::vector<uint8_t>>(projection->presence);
22 * @endcode
23 */
24struct CompositeProjection {
25 DataVariant values; ///< Homogeneous values of the requested field type.
26 DataVariant presence { std::vector<uint8_t> {} }; ///< One byte per value.
27 std::vector<DataDimension> dimensions; ///< One CUSTOM axis named for the field.
28 DataModality modality { DataModality::TENSOR_ND }; ///< Projection modality.
29};

207template <typename T>
208 requires(ArithmeticData<T> && DataVariantElement<T>)
209std::optional<CompositeProjection> CompositeSlice::to_nddata(std::string_view field_name) const
210{
211 const auto field_index = layout().find_field(field_name);
212 if (!field_index || layout().fields()[*field_index].type != std::type_index(typeid(T)))
213 return std::nullopt;
214
215 std::vector<T> values(m_count);
216 std::vector<uint8_t> presence(m_count, 0);
217 for (size_t i = 0; i < m_count; ++i) {
218 const auto element = at(i);
219 if (!element)
220 return std::nullopt;
221 if (auto value = element->get<T>(field_name)) {
222 values[i] = *value;
223 presence[i] = 1;
224 }
225 }
226
227 CompositeProjection result;
228 result.values = std::move(values);
229 result.presence = std::move(presence);
230 result.dimensions.emplace_back(std::string(field_name), m_count, 1, DataDimension::Role::CUSTOM);
231 return result;
232}
