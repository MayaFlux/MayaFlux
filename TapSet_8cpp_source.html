151TapSet TapSetBuilder::start()
152{
153 TapSet taps;
154
155 if (!m_stream) {
156 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
157 "TapSetBuilder::start called with a null stream");
158 return taps;
159 }
160
161 if (m_specs.empty()) {
162 m_specs.emplace_back();
163 }
164
165 const uint64_t frames = m_stream->get_num_frames();
166 const uint64_t end_frame = (m_end_frame == 0 || m_end_frame > frames) ? frames : m_end_frame;
167
168 if (m_start_frame + 1 >= end_frame) {
169 MF_ERROR(Journal::Component::Kriya, Journal::Context::Configuration,
170 "TapSetBuilder::start region [{}, {}) is too short", m_start_frame, end_frame);
171 return taps;
172 }
173
174 const auto rate = static_cast<double>(m_stream->get_sample_rate());
175 const auto first = static_cast<double>(m_start_frame);
176 const auto last = static_cast<double>(end_frame - 1);
177 const uint32_t source_count = std::max<uint32_t>(m_stream->get_num_channels(), 1);
178 const double shortest = 2.0 * static_cast<double>(m_buf_size) / rate;
179
180 std::vector<uint32_t> channels;
181 std::vector<size_t> slot_counts;
182
183 for (const auto& spec : m_specs) {
184 TapSet::Tap built;
185 built.rate = rate;
186 built.velocity = std::make_shared<double>((spec.backward ? -spec.ratio : spec.ratio) * rate);
187
188 Kinesis::TimeMap lag_frames;
189 if (spec.lag) {
190 lag_frames = Kinesis::TimeMap { .fn = [lag = spec.lag, rate, shortest](const double& t) -> double {
191 return std::max((*lag)(t), shortest) * rate;
192 } };
193 }
194 const double origin = static_cast<double>(m_stream->get_write_head(0)) + rate * spec.delay;
195
196 for (const uint32_t output : spec.channels) {
197 auto found = std::ranges::find(channels, output);
198 if (found == channels.end()) {
199 auto made = std::make_shared<SamplingPipeline>(
200 m_stream, m_mgr, m_scheduler, output, m_buf_size);
201 made->build();
202 taps.m_pipelines.push_back(std::move(made));
203 channels.push_back(output);
204 slot_counts.push_back(0);
205 found = std::prev(channels.end());
206 }
207
208 const auto pipeline = static_cast<size_t>(std::distance(channels.begin(), found));
209 const auto& sampler = taps.m_pipelines[pipeline];
210 const size_t slot = slot_counts[pipeline]++;
211
212 auto slice = sampler->slice_from_range(m_start_frame, end_frame, static_cast<uint8_t>(slot));
213 slice.looping = m_looping;
214 slice.scale = spec.level;
215 slice.source_channel = spec.source.value_or(output % source_count);
216 if (!spec.lag) {
217 slice.repeat = spec.repeat;
218 }
219 slice.with_time_map(spec.lag
220 ? Kinesis::TimeMaps::lagged(origin, rate, lag_frames)
221 : Kinesis::TimeMaps::integrated(spec.backward ? last : first, built.velocity));
222 sampler->load(slot, std::move(slice));
223
224 built.seats.push_back({ .sampler = sampler, .slot = slot });
225 }
226
227 taps.m_taps.push_back(std::move(built));
228 }
229
230 std::vector<size_t> order(m_specs.size());
231 for (size_t i = 0; i < order.size(); ++i) {
232 order[i] = i;
233 }
234 std::ranges::stable_sort(order, {}, [this](size_t i) { return m_specs[i].delay; });
235
236 auto entries = std::make_shared<EventChain>(m_scheduler);
237 double elapsed = 0.0;
238
239 for (const size_t i : order) {
240 const auto& seats = taps.m_taps[i].seats;
241 const double delay = m_specs[i].delay;
242
243 if (delay <= 0.0) {
244 for (const auto& seat : seats) {
245 seat.sampler->play(seat.slot);
246 }
247 continue;
248 }
249
250 entries->then([seats] {
251 for (const auto& seat : seats) {
252 seat.sampler->play(seat.slot);
253 }
254 },
255 delay - elapsed);
256 elapsed = delay;
257 }
258
259 if (entries->event_count() > 0) {
260 entries->start();
261 taps.m_entries = std::move(entries);
262 }
263
264 return taps;
265}
