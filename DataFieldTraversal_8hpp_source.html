5/**
6 * @brief Array with optional-like element and contiguous slice access.
7 * @tparam Array Array type whose elements and slices can be borrowed.
8 */
9template <typename Array>
10concept FieldArray = requires(const Array& array, size_t index, size_t count) {
11 { array.size() } -> std::convertible_to<size_t>;
12 { static_cast<bool>(array.at(index)) } -> std::same_as<bool>;
13 *array.at(index);
14 { static_cast<bool>(array.slice(index, count)) } -> std::same_as<bool>;
15 { array.slice(index, count)->size() } -> std::convertible_to<size_t>;
16 *array.slice(index, count)->at(index);
17};

19/**
20 * @class DataFieldTraversal
21 * @brief On-demand windowing and field operations over an indexed array.
22 *
23 * A traversal maintains an independent position and batch size without
24 * owning the array. Field projection is available when the array and its
25 * slices provide to_nddata<T>(field_name). The same traversal works with any
26 * FieldArray; neither it nor the array needs to name a concrete record format.
27 *
28 * Usage:
29 * @code
30 * DataFieldTraversal traversal(items, 64);
31 * while (!traversal.at_end()) {
32 * auto scores = traversal.window_to_nddata<double>("score");
33 * traversal.for_each_in_window([](const auto& item) {
34 * auto name = item.text("name");
35 * });
36 * traversal.advance();
37 * }
38 * @endcode
39 *
40 * @tparam Array Indexed array satisfying FieldArray.
41 * @note Borrowed elements and windows remain subject to the array's own
42 * lifetime and invalidation rules. Keep the array at a stable address while
43 * using a traversal.
44 */

139 /**
140 * @brief Apply a callable to each available element in the array.
141 * @tparam Fn Callable accepting a borrowed element.
142 * @param fn Operation to apply in element order.
143 * @note An index where at() returns disengaged is skipped entirely, not
144 * visited with a default value. fn may be called fewer than size() times.

156 * @brief Apply a callable to each available element in the current window.
157 * @tparam Fn Callable accepting a borrowed element.
158 * @param fn Operation to apply in slice order.
159 * @note An index where at() returns disengaged is skipped entirely, not
160 * visited with a default value. fn may be called fewer than the window's
161 * size() times.
162 */
163 template <typename Fn>
164 void for_each_in_window(Fn&& fn) const
165 {
166 auto window = current();
167 if (!window)
168 return;
169
170 for (size_t i = 0; i < window->size(); ++i) {
171 if (auto element = window->at(i))
172 std::invoke(fn, *element);
173 }
174 }

176 /**
177 * @brief Fold all available elements into an accumulator.
178 * @tparam Acc Accumulator type.
179 * @tparam Fn Callable returning the next accumulator from the current
180 * accumulator and a borrowed element.
181 * @param init Initial accumulator.
182 * @param fn Fold operation applied in element order.
183 * @return Final accumulator.
184 * @note An index where at() returns disengaged is skipped entirely; fn is
185 * not invoked for it, not even with a default-constructed element.
186 */
187 template <typename Acc, typename Fn>
188 [[nodiscard]] Acc reduce(Acc init, Fn&& fn) const
189 {
190 for (size_t i = 0; i < m_array->size(); ++i) {
191 if (auto element = m_array->at(i))
192 init = std::invoke(fn, std::move(init), *element);
193 }
194 return init;
195 }
