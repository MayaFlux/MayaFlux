70/**
71 * @brief Selects where a VisionSequence executes.
72 *
73 * GPU uses MayaFlux::Yantra::VisionGpuExecutor, the Vulkan implementation in
74 * Yantra. It accepts a GPU image, runs the vision steps through GPU dispatches,
75 * and can keep image results on the device. This is the default because it
76 * suits GPU image pipelines and avoids transferring intermediate images to
77 * the host. Host input may still require an upload, and requesting host data
78 * from a GPU result may require a readback and synchronization. GPU resource
79 * setup and dispatch also have costs, particularly for small workloads.
80 *
81 * CPU uses MayaFlux::Kinesis::Vision::VisionExecutor, the host implementation
82 * in Kinesis. It accepts normalized floating-point pixels in host memory and
83 * produces host-side results. Choose it when the input and desired output are
84 * already on the host, when a GPU is unavailable, or when a sequence needs an
85 * operation implemented only by the CPU executor. Supplying a GPU image to a
86 * CPU pipeline requires a transfer to host memory.
87 *
88 * The selected backend applies to the whole sequence. VisionProcessor and
89 * ImageCVProcessor do not switch backend for individual steps or fall back to
90 * the other executor when an operation is unsupported. The default expresses
91 * the preferred path for GPU image processing, not a guarantee that it will
92 * be faster for every input size or transfer pattern.
93 */
94enum class VisionBackend : uint8_t {
95 GPU, ///< Execute the sequence with the Yantra Vulkan executor.
96 CPU, ///< Execute the sequence with the Kinesis host executor.
97};

104 * @brief Selects channels of the current image for a multichannel threshold.
105 *
106 * Ch0, Ch1 and Ch2 are the first three channels as stored: red, green, blue
107 * on an RGBA image, hue, saturation, value after RgbaToHsv. NONE selects the

209/**
210 * @brief Parameters for TrackKeypoints.
211 *
212 * eigen_threshold and error_threshold are normalised by the window area, so
213 * they do not depend on window_radius. levels, max_points and min_distance
214 * apply to the persistent GPU tracker: levels is the pyramid depth including
215 * full resolution, max_points caps live tracks, and min_distance is the grid
216 * cell size in pixels that admits at most one new track. A positive
217 * forward_backward_threshold rejects a track when its backward estimate
218 * misses the source point by more than that many pixels; zero disables it.
219 * host_tracks false skips the host readback entirely (structured stays
220 * empty) when export_tracks is set; it has no effect on the CPU executor,
221 * which always produces the host result.
222 */
223struct TrackKeypointsParams {
224 uint32_t window_radius = 7;
225 uint32_t max_iterations = 20;
226 float eigen_threshold = 1e-4F;
227 float error_threshold = 0.3F;
228 uint32_t levels = 4;
229 uint32_t max_points = 512;
230 float min_distance = 8.0F;
231 float forward_backward_threshold = 0.0F;
232 bool export_tracks = false;
233 bool host_tracks = true;
234};

236/**
237 * @brief Parameters for OpticalFlowDense.
238 *
239 * Dense coarse-to-fine Lucas-Kanade. window_radius is the half size of the
240 * Gaussian weighted window in pixels of the level being solved, iterations the
241 * refinement steps per level, and levels the pyramid depth including full
242 * resolution (lowered for small images). eigen_threshold damps the solve so
243 * untextured pixels take small steps, and max_step caps one increment in
244 * pixels. With visualize set, a hue and brightness rendering of the flow is
245 * delivered in VisionResult::images.flow_visualization, at full brightness
246 * from visual_range pixels. visual_min_motion keeps smaller displacements dark in
247 * the visualization, in pixels of the flow image.
248 */
249struct OpticalFlowDenseParams {
250 uint32_t window_radius = 5;
251 uint32_t iterations = 3;
252 uint32_t levels = 4;
253 float eigen_threshold = 1e-3F;
254 float max_step = 4.0F;
255 bool visualize = false;
256 float visual_range = 2.0F;
257 float visual_min_motion = 0.0F;
258};

260/**
261 * @brief Parameters for Confine.
262 *
263 * bounds is a normalised [0, 1] rectangle anywhere in the current image,
264 * not limited to a found region. resize_to_source false (the default)
265 * leaves the working image at bounds' own pixel size; true nearest-resamples
266 * the confined region back to the frame's own size before this step, e.g.
267 * to keep a fixed working resolution for whatever follows.
268 *
269 * Every later step, and every result coordinate, is in the confined image's
270 * space; the caller remaps to the full frame. In a sequence with
271 * TrackKeypoints or OpticalFlowDense, place Confine before RgbaToGray: the
272 * flow pyramid is built from the gray frame at that point, so a later Confine
273 * is rejected, or with resize_to_source silently mismatches the pyramid. Flow
274 * state resets when the confined size changes, and bounds that move at a
275 * constant size compare frames of different content.
276 */
277struct ConfineParams {
278 BoundingBox bounds;
279 bool resize_to_source { false };
280};

591/**
592 * @brief Combine a hash into an existing seed, FNV-style.
593 */
594inline void hash_combine(size_t& seed, size_t value)
595{
596 seed ^= value + 0x9e3779b9U + (seed << 6) + (seed >> 2);
597}
598
599/**
600 * @brief Combine an optional region into a hash: absence and presence hash
601 * differently, presence by its four bounds.
602 */
603inline void hash_region(size_t& seed, const std::optional<BoundingBox>& region)
604{
605 hash_combine(seed, std::hash<bool> {}(region.has_value()));
606 if (!region)
607 return;
608 hash_combine(seed, std::hash<float> {}(region->x));
609 hash_combine(seed, std::hash<float> {}(region->y));
610 hash_combine(seed, std::hash<float> {}(region->w));
611 hash_combine(seed, std::hash<float> {}(region->h));
612}

614/**
615 * @brief Hash a VisionStep's op and parameters together.
616 *
617 * Keys GPU dispatch memoization on VisionPass::completed, which spans one
618 * walk including any suspensions. Two steps hashing equal are treated as
619 * interchangeable, so every field that changes the output must be hashed.
620 */
621inline size_t hash_vision_step(VisionOp op, const VisionParams& params)
622{
623 size_t seed = std::hash<std::string_view> {}(Reflect::enum_to_string(op));
624
625 std::visit([&seed](const auto& p) {
626 using T = std::decay_t<decltype(p)>;
627 if constexpr (std::is_same_v<T, std::monostate>) {
628 } else if constexpr (std::is_same_v<T, ThresholdParams>) {
629 hash_combine(seed, std::hash<float> {}(p.value));
630 hash_combine(seed, std::hash<uint8_t> {}(static_cast<uint8_t>(p.channels)));
631 for (const auto& band : p.bands) {
632 hash_combine(seed, std::hash<float> {}(band.lo));
633 hash_combine(seed, std::hash<float> {}(band.hi));
634 }
635 } else if constexpr (std::is_same_v<T, ThresholdAdaptiveParams>) {
636 hash_combine(seed, std::hash<uint32_t> {}(p.block_size));
637 hash_combine(seed, std::hash<float> {}(p.offset));
638 hash_combine(seed, std::hash<uint8_t> {}(static_cast<uint8_t>(p.channels)));
639 } else if constexpr (std::is_same_v<T, OtsuParams>) {
640 hash_combine(seed, std::hash<uint8_t> {}(static_cast<uint8_t>(p.channels)));
641 } else if constexpr (std::is_same_v<T, NormalizeRangeParams>) {
642 hash_combine(seed, std::hash<float> {}(p.lo));
643 hash_combine(seed, std::hash<float> {}(p.hi));
644 } else if constexpr (std::is_same_v<T, GaussianBlurParams>) {
645 hash_combine(seed, std::hash<float> {}(p.sigma));
646 } else if constexpr (std::is_same_v<T, FilterSeparableParams>) {
647 for (float v : p.kernel_x)
648 hash_combine(seed, std::hash<float> {}(v));
649 for (float v : p.kernel_y)
650 hash_combine(seed, std::hash<float> {}(v));
651 } else if constexpr (std::is_same_v<T, CannyParams>) {
652 hash_combine(seed, std::hash<float> {}(p.sigma));
653 hash_combine(seed, std::hash<float> {}(p.low_threshold));
654 hash_combine(seed, std::hash<float> {}(p.high_threshold));
655 } else if constexpr (std::is_same_v<T, MorphParams>) {
656 hash_combine(seed, std::hash<uint32_t> {}(p.radius));
657 } else if constexpr (std::is_same_v<T, HarrisParams>) {
658 hash_combine(seed, std::hash<float> {}(p.k));
659 hash_combine(seed, std::hash<float> {}(p.sigma));
660 hash_region(seed, p.region);
661 } else if constexpr (std::is_same_v<T, ExtractPeaksParams>) {
662 hash_combine(seed, std::hash<float> {}(p.threshold));
663 hash_combine(seed, std::hash<uint32_t> {}(p.nms_radius));
664 hash_combine(seed, std::hash<bool> {}(p.export_keypoints));
665 hash_region(seed, p.region);
666 } else if constexpr (std::is_same_v<T, TrackKeypointsParams>) {
667 hash_combine(seed, std::hash<uint32_t> {}(p.window_radius));
668 hash_combine(seed, std::hash<uint32_t> {}(p.max_iterations));
669 hash_combine(seed, std::hash<float> {}(p.eigen_threshold));
670 hash_combine(seed, std::hash<float> {}(p.error_threshold));
671 hash_combine(seed, std::hash<uint32_t> {}(p.levels));
672 hash_combine(seed, std::hash<uint32_t> {}(p.max_points));
673 hash_combine(seed, std::hash<float> {}(p.min_distance));
674 hash_combine(seed, std::hash<float> {}(p.forward_backward_threshold));
675 hash_combine(seed, std::hash<bool> {}(p.export_tracks));
676 hash_combine(seed, std::hash<bool> {}(p.host_tracks));
677 } else if constexpr (std::is_same_v<T, FindContoursParams>) {
678 hash_combine(seed, std::hash<float> {}(p.min_area));
679 hash_combine(seed, std::hash<uint32_t> {}(p.max_contours));
680 hash_combine(seed, std::hash<uint32_t> {}(p.max_points_per_contour));
681 hash_combine(seed, std::hash<bool> {}(p.as_image));
682 hash_combine(seed, std::hash<bool> {}(p.export_contours_buffer));
683 } else if constexpr (std::is_same_v<T, ConnectedComponentsParams>) {
684 hash_combine(seed, std::hash<bool> {}(p.export_labels));
685 hash_combine(seed, std::hash<bool> {}(p.with_colors));
686 hash_combine(seed, std::hash<bool> {}(p.export_boxes));
687 hash_combine(seed, std::hash<bool> {}(p.export_label_buffer));
688 } else if constexpr (std::is_same_v<T, OpticalFlowDenseParams>) {
689 hash_combine(seed, std::hash<uint32_t> {}(p.window_radius));
690 hash_combine(seed, std::hash<uint32_t> {}(p.iterations));
691 hash_combine(seed, std::hash<uint32_t> {}(p.levels));
692 hash_combine(seed, std::hash<float> {}(p.eigen_threshold));
693 hash_combine(seed, std::hash<float> {}(p.max_step));
694 hash_combine(seed, std::hash<bool> {}(p.visualize));
695 hash_combine(seed, std::hash<float> {}(p.visual_range));
696 hash_combine(seed, std::hash<float> {}(p.visual_min_motion));
697 } else if constexpr (std::is_same_v<T, ConfineParams>) {
698 hash_combine(seed, std::hash<float> {}(p.bounds.x));
699 hash_combine(seed, std::hash<float> {}(p.bounds.y));
700 hash_combine(seed, std::hash<float> {}(p.bounds.w));
701 hash_combine(seed, std::hash<float> {}(p.bounds.h));
702 hash_combine(seed, std::hash<bool> {}(p.resize_to_source));
703 }
704 },
705 params);
706
707 return seed;
708}

710/**
711 * @brief True when any step tracks keypoints.
712 *
713 * Whole-sequence, not index-local: the gray-frame capture this gates happens
714 * at RgbaToGray, arbitrarily far ahead of the TrackKeypoints step that needs
715 * it. Evaluate once per run; sequences are short.
716 */
717[[nodiscard]] inline bool tracks_keypoints(const VisionSequence& seq)
718{
719 return std::ranges::any_of(seq.steps,
720 [](const VisionStep& s) { return s.op == VisionOp::TrackKeypoints; });
721}
722
723/**
724 * @brief True when an ExtractPeaks step is immediately followed by TrackKeypoints.
725 *
726 * Whole-sequence for the same reason: the capture gate at RgbaToGray needs the
727 * answer before either step is reached. Ops sitting at the pair itself should
728 * use VisionPass::ahead() instead.
729 */
730[[nodiscard]] inline bool track_follows_peaks(const VisionSequence& seq)
731{
732 for (size_t i = 0; i + 1 < seq.steps.size(); ++i) {
733 if (seq.steps[i].op == VisionOp::ExtractPeaks
734 && seq.steps[i + 1].op == VisionOp::TrackKeypoints)
735 return true;
736 }
737 return false;
738}
